15class stationCoherentlySummedWaveforms: 

16 """ 

17 Generates a coherently-summed waveform (CSW) and calculates its signal-to-noise-ratio (SNR). 

18 

19 When multiple waveforms and a referance waveform are given, 

20 one can find the cross correlation between each waveform and the referance waveform, 

21 then each waveform will be rolled to line up the signal position with the referance 

22 based on the time lag when both waveforms are most correlated, 

23 and then all waveforms including the reference will be summed up 

24 to become a coherently-summed waveform (CSW). 

25 Thermal noise is random so it scales with the square root of added waveforms while coherent signals 

26 will added up linearly. Once summed the CSW is treated as a regular waveform and different analysis variables 

27 are calculated. 

28 """ 

29

34 def begin(self, coincidence_window_size=6 * units.ns, pad_length=500, channel_ids=[0, 1, 2, 3]):

36 Parameters 

37 ---------- 

38 coincidence_window_size: float (default: 6 * units.ns) 

39 Window size used for calculating the maximum peak to peak amplitude in nanoseconds 

40 

41 pad_length: int (default: 500) 

42 Padding length used for calculating the coherent sum 

43 

44 channel_ids: array of int (default: [0, 1, 2, 3]) 

45 Channels for which to calculate the variables 

46 """

52 @register_run() 

53 def run(self, evt, station, det, ref_ch_id=0, use_envelope=True): 

54 """ 

55 Calculate the SNR of the coherently-summed waveform and add to the station object. 

56 

57 Parameters 

58 ---------- 

59 evt, station, det 

60 Event, Station, Detector 

61 ref_ch_id: int (default: 0) 

62 Reference channel for the coherent sum 

63 use_envelope: bool (default: True) 

64 If use Hilbert envelopes to find the cross correlation or not 

65 """ 

66 

67 ref_ch = station.get_channel(ref_ch_id) 

68 ref_trace = ref_ch.get_trace() 

69 trace_set = [ch.get_trace() for ch in station.iter_channels(use_channels = self.__channel_ids) if ch.get_id() != ref_ch.get_id()] 

70 

71 coincidence_window_size_bins_ref = int(round(self.__coincidence_window_size * ref_ch.get_sampling_rate())) 

72 if coincidence_window_size_bins_ref < 2: 

73 logger.warning(f"Coincidence window size of {coincidence_window_size_bins_ref} samples is too small for channel {ref_ch.get_id()}.") 

74 

75 sum_trace = trace_utilities.get_coherent_sum(trace_set, ref_trace, use_envelope) 

76 rms = trace_utilities.get_split_trace_noise_RMS(sum_trace, segments=4, lowest=2) 

77 snr = trace_utilities.get_signal_to_noise_ratio(sum_trace, rms, window_size=coincidence_window_size_bins_ref) 

78 impulsivity = trace_utilities.get_impulsivity(sum_trace) 

79 entropy = trace_utilities.get_entropy(sum_trace) 

80 kurtosis = trace_utilities.get_kurtosis(sum_trace) 

81 

82 station.set_parameter(stpRNOG.coherent_snr, snr) 

83 station.set_parameter(stpRNOG.coherent_impulsivity, impulsivity) 

84 station.set_parameter(stpRNOG.coherent_entropy, entropy) 

85 station.set_parameter(stpRNOG.coherent_kurtosis, kurtosis)

91 def coherent_sum_step_by_step(self, station): 

92 """ Plot the four original waveforms before any alignment and save """ 

93 

94 matplotlib.rcParams.update({"font.size": 20}) 

95 plt.figure(figsize = (10, 6)) 

96 for channel in station.iter_channels(use_channels = self.__channel_ids): 

97 plt.plot(channel.get_trace(), label = f'Original wf[{channel.get_id()}]') 

98 plt.title('Original Waveforms') 

99 plt.xlabel('Sample') 

100 plt.ylabel('Amplitude') 

101 plt.legend() 

102 plt.savefig('original_waveforms.png') 

103 plt.show() 

104 

105 # Iterate over each channel as a potential reference 

106 for ref_ch in station.iter_channels(use_channels = self.__channel_ids): 

107 fig, axs = plt.subplots(2, 2, figsize = (12, 10)) # Create 2x2 grid for each step with a single reference 

108 fig.suptitle(f'Coherent Sum Steps with Reference: wf[{ref_ch.get_id()}]', fontsize = 16) 

109 

110 sum_chan = np.pad(ref_ch.get_trace(), self.__pad_length, mode = 'constant') # Pad reference waveform 

111 channels = [ch for ch in station.iter_channels(use_channels = self.__channel_ids) if ch.get_id() != ref_ch.get_id()] # Exclude the reference channel 

112 

113 # Initialize the coherent sum with the padded reference waveform 

114 current_sum = sum_chan.copy() 

115 

116 for step, ch in enumerate(channels): 

117 ax = axs[step // 2, step % 2] # Select subplot for each step 

118 

119 # Perform full-range cross-correlation to find the necessary shift 

120 cor = signal.correlate(current_sum[self.__pad_length:-self.__pad_length], ch.get_trace(), mode = 'full') 

121 shift = np.argmax(cor) - (len(ch.get_trace()) - 1) # Calculate shift based on maximum correlation 

122 

123 # Pad and shift the waveform to align with the reference sum 

124 padded_wf = np.pad(ch.get_trace(), self.__pad_length, mode = 'constant') 

125 aligned_wf = np.roll(padded_wf, shift) 

126 

127 # Plot the current coherent sum before adding the next waveform 

128 ax.plot(current_sum[self.__pad_length:-self.__pad_length], label = 'Current Coherent Sum', linestyle = '--') 

129 ax.plot(aligned_wf[self.__pad_length:-self.__pad_length], label = f'Next wf[{ch.get_id()}]', alpha = 0.7) 

130 

131 # Add the aligned waveform to the current sum 

132 current_sum += aligned_wf 

133 

134 # Customize plot 

135 ax.legend(loc = 'upper right') 

136 ax.set_xlabel('Sample') 

137 ax.set_ylabel('Amplitude') 

138 ax.set_title(f'Step {step + 1}: Adding wf[{ch.get_id()}]') 

139 

140 # Save each coherent summing figure for the current reference 

141 plt.tight_layout(rect = [0, 0, 1, 0.96]) 

142 plt.savefig(f'coherent_sum_steps_reference_{ref_ch.get_id()}.png') 

143 plt.show()
