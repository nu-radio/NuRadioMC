12class channelTimeOffsetCalculator: 

13 """ 

14 Determines the difference in signal arrival times between channels, the ray types and the signal 

15 arrival direction from the vertex position. 

16 

17 The module uses either the vertex position stored in one of the sim showers or it assumes that 

18 some vertex reconstruction module was already run and stored the vertex position either in the 

19 vertex_2D_fit (for the neutrino2DVertexReconstructor module) of the nu_vertex station parameter. 

20  

21 The module then calculates the expected difference in travel times between channels for this 

22 vertex position, shifts the channel voltages by that time difference and calculates the correlation 

23 with a template. By adding up the correlations for all channels, we can determine the correct 

24 raytracing solution: If the solution is correctd, the correlations will have their maxima at the same 

25 position, resulting in a larger maximum of their sum. Thus, we can determine the correct raytracing 

26 solutions and store the corresponding properties in the channel parameters. 

27  

28 This module assumes that the ray path type for all channels is the same, e.g. each channel sees a direct 

29 ray. Therefore it should only be used for channels that are relatively close to each other. 

30  

31 """ 

32 def __init__(self): 

33 self.__use_sim = False 

34 self.__raytracing_types = ['direct', 'refracted', 'reflected'] 

35 self.__electric_field_template = None 

36 self.__medium = None 

37 self.__antenna_provider = NuRadioReco.detector.antennapattern.AntennaPatternProvider() 

38 

39 def begin(self, electric_field_template, medium, use_sim=False): 

40 """ 

41 Setup method for the module 

42 

43 Parameters 

44 ---------- 

45 electric_field_template: NuRadioReco.framework.base_trace.BaseTrace object (or child of BaseTrace class) 

46 An electric field waveform to be used as a template. It will be folded with the antenna and amplifier 

47 response to get a voltage template, which is then used to estimate timing differences between channels. 

48 medium: NuRadioMC.utilities.medium.Medium object 

49 The index of refraction profile of the ice. 

50 use_sim: Boolean (default: False) 

51 If true, the simulated vertex is used, otherwise it is assumed that the vertex is stored on either the 

52 vertex_2D_fit (for the neutrino2DVertexReconstructor) or the nu_vertex station parameter 

53 """ 

54 self.__use_sim = use_sim 

55 self.__electric_field_template = electric_field_template 

56 self.__medium = medium 

57 

58 @register_run() 

59 def run(self, event, station, det, channel_ids, passband): 

60 """ 

61 Run the module on a station 

62 

63 Parameters 

64 ---------- 

65 event: NuRadioReco.framework.event.Event object 

66 The event the module should be run on 

67 station: NuRadioReco.framework.station.Station object 

68 The station the module should be run on 

69 det: NuRadioReco.detector.detector.Detector object of child object 

70 The detector description 

71 channel_ids: array of int 

72 IDs of the channels the module should be run on 

73 passband: array of float 

74 Lower and upper bound of the bandpass filter that is applied to the 

75 channel waveforms when determining correlation to the template. Can 

76 be used to filter out frequencies that are dominated by noise. 

77 """ 

78 

79 # Create data structured to store pulse properties 

80 propagation_times = np.zeros((len(channel_ids), 3)) 

81 receive_angles = np.zeros((len(channel_ids), 3)) 

82 found_solutions = np.zeros((len(channel_ids), 3)) 

83 # Get vertex position 

84 vertex_position = None 

85 if self.__use_sim: 

86 for sim_shower in event.get_sim_showers(): 

87 if sim_shower.has_parameter(shp.vertex): 

88 vertex_position = sim_shower.get_parameter(shp.vertex) 

89 break 

90 else: 

91 if station.has_parameter(stnp.nu_vertex): 

92 vertex_position = station.get_parameter(stnp.nu_vertex) 

93 elif station.has_parameter(stnp.vertex_2D_fit): 

94 vertex_2d = station.get_parameter(stnp.vertex_2D_fit) 

95 vertex_position = np.array([vertex_2d[0], 0, vertex_2d[1]]) 

96 if vertex_position is None: 

97 raise RuntimeError('Could not find vertex position') 

98 correlation_size = 0 

99 for i_channel, channel_id in enumerate(channel_ids): 

100 channel = station.get_channel(channel_id) 

101 if channel.get_sampling_rate() != self.__electric_field_template.get_sampling_rate(): 

102 raise RuntimeError( 

103 'The channels and the electric field remplate need to have the same sampling rate.' 

104 ) 

105 # Calculate size of largest autocorrelation 

106 if channel.get_number_of_samples() + self.__electric_field_template.get_number_of_samples() - 1 > correlation_size: 

107 correlation_size = channel.get_number_of_samples() + self.__electric_field_template.get_number_of_samples() - 1 

108 channel_position = det.get_relative_position(station.get_id(), channel_id) 

109 raytracer = NuRadioMC.SignalProp.analyticraytracing.ray_tracing( 

110 x1=vertex_position, 

111 x2=channel_position, 

112 medium=self.__medium 

113 ) 

114 raytracer.find_solutions() 

115 # Loop through all 3 ray path types and store the properties into the data structure 

116 for i_solution, solution in enumerate(raytracer.get_results()): 

117 solution_type = raytracer.get_solution_type(i_solution) 

118 found_solutions[i_channel, solution_type - 1] += 1 

119 propagation_times[i_channel, solution_type - 1] = raytracer.get_travel_time(i_solution) 

120 receive_vector = raytracer.get_receive_vector(i_solution) 

121 receive_angles[i_channel, solution_type - 1] = radiotools.helper.cartesian_to_spherical(receive_vector[0], receive_vector[1], receive_vector[2])[0] 

122 # We only want the relative time differences between channels, so we subtract the mean propagation time 

123 for i_solution in range(3): 

124 if len(propagation_times[:, i_solution][propagation_times[:, i_solution] > 0]) > 0: 

125 propagation_times[:, i_solution][propagation_times[:, i_solution] > 0] -= np.mean(propagation_times[:, i_solution][propagation_times[:, i_solution] > 0]) 

126 correlation_sum = np.zeros((3, correlation_size)) 

127 # Now we check which ray path results in the best correlation between channels 

128 for i_channel, channel_id in enumerate(channel_ids): 

129 channel = station.get_channel(channel_id) 

130 antenna_pattern = self.__antenna_provider.load_antenna_pattern(det.get_antenna_model(station.get_id(), channel_id)) 

131 antenna_orientation = det.get_antenna_orientation(station.get_id(), channel_id) 

132 for i_solution in range(3): 

133 if found_solutions[i_channel, i_solution] > 0: 

134 # We calculate the voltage template from the electric field template using the receiving angles 

135 # for that raytracing solution 

136 antenna_response = antenna_pattern.get_antenna_response_vectorized( 

137 self.__electric_field_template.get_frequencies(), 

138 receive_angles[i_channel, i_solution], 

139 0., 

140 antenna_orientation[0], 

141 antenna_orientation[1], 

142 antenna_orientation[2], 

143 antenna_orientation[3] 

144 ) 

145 # For simplicity, we assume equal contribution on the E_theta and E_phi component 

146 channel_template_spec = fft.time2freq(self.__electric_field_template.get_filtered_trace(passband), self.__electric_field_template.get_sampling_rate()) * \ 

147 det.get_amplifier_response(station.get_id(), channel_id, self.__electric_field_template.get_frequencies()) * (antenna_response['theta'] + antenna_response['phi']) 

148 channel_template_trace = fft.freq2time(channel_template_spec, self.__electric_field_template.get_sampling_rate()) 

149 # We apply the expected time shift for the raytracing solution and calculate the correlation with the template 

150 channel.apply_time_shift(-propagation_times[i_channel, i_solution], True) 

151 correlation = radiotools.helper.get_normalized_xcorr(channel_template_trace, channel.get_filtered_trace(passband)) 

152 correlation = np.abs(correlation) 

153 correlation_sum[i_solution][:len(correlation)] += correlation 

154 channel.apply_time_shift(propagation_times[i_channel, i_solution], True) 

155 

156 correlation_sum_max = np.max(correlation_sum, axis=1) 

157 # We look for the raytracing solution that yielded the best correlation and store the corresponding properties 

158 # in the channel parameters 

159 for i_channel, channel_id in enumerate(channel_ids): 

160 channel = station.get_channel(channel_id) 

161 channel.set_parameter(chp.signal_time_offset, propagation_times[i_channel, np.argmax(correlation_sum_max)]) 

162 channel.set_parameter(chp.signal_receiving_zenith, receive_angles[i_channel, np.argmax(correlation_sum_max)]) 

163 channel.set_parameter(chp.signal_ray_type, self.__raytracing_types[np.argmax(correlation_sum_max)])
