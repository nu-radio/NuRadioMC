16class channelGlitchDetector: 

17 """ 

18 This module detects scrambled data (digitizer "glitches") in the channels. 

19 

20 RNO-G data (in particular the RNO-G data recorded with the LAB4D digitizers on the 

21 first-generation radiants) is known to have "glitches" in the channels. 

22 When a glitch is present in a channel, the 64-sample readout blocks were scrambled 

23 by the readout electronics which results in sudden unphyiscal jumps in the signal. 

24 These jumps are detected with this module (but not corrected). 

25 """ 

26 

27 def __init__(self, cut_value=0.0, glitch_fraction_warn_level=0.1, log_level=logging.NOTSET): 

28 """ 

29 Parameters 

30 ---------- 

31 cut_value : float 

32 This module calculates a test statistic that is sensitive to the presence 

33 of digitizer glitches. This parameter marks the critical value at which the test is 

34 said to have detected a glitch. This is a free parameter; increasing its value results 

35 in a lower false-positive rate (i.e. healthy events are incorrectly marked as glitching), 

36 but also a lower true-positive rate (i.e. glitching events are correctly marked as such). 

37 The default value of 0.0 is a good starting point. 

38 

39 glitch_fraction_warn_level : float 

40 Print warning messages at the end of a run if a channel shows glitching in more than a fraction of 

41 `glitch_fraction_warn_level` of all events. 

42 

43 log_level: enum 

44 Set verbosity level of logger. If logging.DEBUG, set mattak to verbose (unless specified in mattak_kwargs). 

45 (Default: logging.NOTSET, ie adhere to general log level) 

46 

47 """ 

48 

49 self.logger = logging.getLogger('NuRadioReco.RNO_G.channelGlitchDetector') 

50 

51 self.ts_cut_value = cut_value 

52 self.glitch_fraction_warn_level = glitch_fraction_warn_level 

53 

54 # Total sampling buffer of the LAB4D: 2048 samples 

55 self._lab4d_readout_size = LAB4D_READOUT_SIZE 

56 

57 # Length of a sampling block in the LAB4D: 64 samples 

58 self._lab4d_sampling_blocksize = LAB4D_SAMPLING_BLOCK_SIZE

60 def begin(self): 

61 # Per-run glitching statistics 

62 self.events_checked = 0 

63 self.events_glitching_per_channel = collections.defaultdict(int) 

64 

65 def end(self): 

66 # Print glitch statistic summary 

67 for ch_id, events_glitching in self.events_glitching_per_channel.items(): 

68 glitching_fraction = events_glitching / self.events_checked 

69 if glitching_fraction > self.glitch_fraction_warn_level: 

70 self.logger.warning(f"Channel {ch_id} shows glitches in {events_glitching} " 

71 f"/ {self.events_checked} = {100*glitching_fraction:.2f}% of events!") 

72 else: 

73 self.logger.info(f"Channel {ch_id} shows glitches in {events_glitching} " 

74 f"/ {self.events_checked} = {100*glitching_fraction:.2f}% of events!")

76 @register_run() 

77 def run(self, event, station, det=None): 

78 """ Run over channel traces and sets `channelParameter.glitch`. 

79 

80 Parameters 

81 ---------- 

82 event : `NuRadioReco.framework.event.Event` 

83 The event object 

84 station : `NuRadioReco.framework.station.Station` 

85 The station object 

86 det : `NuRadioReco.detector.detector.Detector` (default: None) 

87 Detector object, not used! 

88 """ 

89 

90 # update event counter 

91 self.events_checked += 1 

92 

93 for ch in station.iter_channels(): 

94 ch_id = ch.get_id() 

95 

96 trace = ch.get_trace() 

97 trace_us = unscramble(trace, block_size=self._lab4d_sampling_blocksize, readout_size=self._lab4d_readout_size) 

98 

99 # glitching test statistic and boolean discriminate 

100 glitch_ts = (diff_sq(trace, block_size=self._lab4d_sampling_blocksize) - 

101 diff_sq(trace_us, block_size=self._lab4d_sampling_blocksize)) / np.var(trace) 

102 glitch_disc = glitch_ts > self.ts_cut_value 

103 

104 ch.set_parameter(chp.glitch, glitch_disc) 

105 ch.set_parameter(chp.glitch_test_statistic, glitch_ts) 

106 

107 # update glitching statistics 

108 self.events_glitching_per_channel[ch_id] += glitch_disc

110def unscramble(trace, block_size=LAB4D_SAMPLING_BLOCK_SIZE, readout_size=LAB4D_READOUT_SIZE): 

111 """ 

112 Applies an unscrambling operation to the passed `trace`. 

113 Note: the first and last sampling block are unusable and hence replaced by zeros in the returned waveform. 

114 

115 Parameters 

116 ---------- 

117 trace: np.ndarray 

118 The channel waveform 

119 

120 Returns 

121 ------- 

122 new_trace: np.ndarray 

123 The unscrambled trace 

124 """ 

125 

126 twice_block_size = 2 * block_size 

127 

128 new_trace = np.zeros_like(trace) 

129 

130 for i_section in range(len(trace) // block_size): 

131 section_start = i_section * block_size 

132 section_end = i_section * block_size + block_size 

133 if i_section % 2 == 0: 

134 new_trace[(section_start + twice_block_size) % readout_size :\ 

135 (section_end + twice_block_size) % readout_size] = trace[section_start:section_end] 

136 elif i_section > 1: 

137 new_trace[(section_start - twice_block_size) % readout_size :\ 

138 (section_end - twice_block_size) % readout_size] = trace[section_start:section_end] 

139 new_trace[0:block_size] = 0 

140 

141 return new_trace

143def diff_sq(eventdata, block_size=LAB4D_SAMPLING_BLOCK_SIZE): 

144 """ 

145 Returns sum of squared differences of samples across seams of 128-sample chunks. 

146 Ignores the first and last `block_size` samples, which `unscramble` cannot reconstruct. 

147 

148 Parameters 

149 ---------- 

150 eventdata: np.ndarray 

151 The channel waveform 

152 """ 

153 twice_block_size = 2 * block_size 

154 

155 runsum = 0.0 

156 for chunk in range(1, len(eventdata) // twice_block_size - 1): 

157 runsum += (eventdata[chunk * twice_block_size + block_size - 1] - eventdata[chunk * twice_block_size + block_size]) ** 2 

158 return np.sum(runsum)

161 """ 

162 Returns true if any channel in any station has a "glitch". 

163 

164 Requires the RNO_G.channelGlitchDetector module to have ran on the event. 

165 

166 Parameters 

167 ---------- 

168 event_or_station : Event or Station 

169 The event or station to check for glitches. If an event is given, the first station is used 

170 (if multiple stations exist in the event an error is raised). 

171 

172 Returns 

173 ------- 

174 has_glitch : bool 

175 True if any channel has a glitch, False otherwise. 

176 

177 See Also 

178 -------- 

179 NuRadioReco.modules.RNO_G.channelGlitchDetector 

180 """ 

181 if isinstance(event_or_station, Event): 

182 # This will throw an error if the event has more than one station 

183 station = event_or_station.get_station() 

184 else: 

185 station = event_or_station 

186 

187 for channel in station.iter_channels():
