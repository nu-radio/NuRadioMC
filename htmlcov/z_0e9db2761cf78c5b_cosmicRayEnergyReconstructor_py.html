1""" 

2Module to reconstruct the energy of an air shower from its radio signal 

3 

4To reconstruct the air-shower energy, the radio signals should be bandpass filtered 

5between 80 to 300 MHz with a 10th order Butterworth filter, and both the cosmic ray 

6direction and the unfolded electric field should be present. The latter can be obtained 

7(after the signal direction has been reconstructed) by running the 

8`NuRadioReco.modules.voltageToAnalyticEfieldConverter` module. 

9 

10For more details on this algorithm, see https://dx.doi.org/10.1088/1475-7516/2019/10/075 

11 

12""" 

13 

14import numpy as np 

15from NuRadioReco.modules.base.module import register_run 

16from NuRadioReco.framework.parameters import stationParameters as stnp 

17from NuRadioReco.framework.parameters import electricFieldParameters as efp 

18from NuRadioReco.utilities import units 

19import NuRadioReco.utilities.trace_utilities 

20import logging 

21import radiotools.helper as hp 

22import radiotools.coordinatesystems 

23import radiotools.atmosphere.models 

24 

25 

26class cosmicRayEnergyReconstructor: 

27 """ 

28 Reconstructs the energy of an air shower from its radio signal 

29 

30 Requires the following modules to be run beforehand: 

31  

32 * a 10th order Butterworth bandpass filter with passband 80-300 MHz 

33 * a direction reconstruction 

34 * the voltageToAnalyticEfieldConverter 

35  

36 """ 

37 

38 def __init__(self): 

39 self.logger = logging.getLogger('NuRadioReco.cosmicRayEnergyReconstructor') 

40 self.__atmosphere = radiotools.atmosphere.models.Atmosphere() 

41 self.__parametrizations = { 

42 'mooresbay': { 

43 'scale': np.array([(442.46, -281.75, 324.96), (394.08, -308.36, 436.30)]), 

44 'falloff': np.array([(-.1584, -.07943), (.8070, -1.4098)]) 

45 }, 

46 'southpole': { 

47 'scale': np.array([(976.30, -1213.43, 626.98), (643.39, -667.08, 478.06)]), 

48 'falloff': np.array([(-.2273, .05627), (1.3372, -2.1653)]) 

49 }, 

50 'auger': { 

51 'scale': np.array([(229.96, -123.75, 110.51), (214.46, -111.01, 119.18)]), 

52 'falloff': np.array([(-.1445, -.09820), (.5936, -1.1763)]) 

53 }, 

54 'summit': { 

55 'scale': np.array([[ 404.5 , -131.56, 11.7 ], [ 428.97, -92.11, 5.94]]), 

56 'falloff': np.array([[-0.3391, 0.1738], [ 0.9543, -1.6967]]) 

57 } 

58 } 

59 self.__elevations = { # TODO: This should be changed once we have implemented a proper coordinate system 

60 'mooresbay': 30., 

61 'southpole': 2800., 

62 'auger': 1560., 

63 'summit': 3216. 

64 } 

65 self.__site = None 

66 

67 def begin(self, site=None): 

68 """ 

69 Initialize the cosmicRayEnergyReconstructor (optional) 

70 

71 Parameters 

72 ---------- 

73 site : string | None (default: None) 

74 Specifies the site of the station. The parameterization of 

75 the cosmic ray energy depends on the site of the detector. 

76 

77 If None, the site will be determined from the detector 

78 passed to the `run` function. 

79 

80 """ 

81 self.__site = site 

82 if site not in self.__parametrizations.keys(): 

83 self.logger.error('Unsupported site. Please select one of the following: {}'.format(self.__parametrizations.keys())) 

84 raise ValueError 

85 

86 @register_run() 

87 def run(self, event, station, detector, electric_field=None): 

88 """ 

89 Determine the cosmic ray energy from the electric field fluence. 

90 

91 The reconstructed cosmic ray energy will be stored in the 

92 station in the :obj:`cr_energy_em <NuRadioReco.framework.parameters.stationParameters.cr_energy_em>` parameter. 

93 

94 Parameters 

95 ---------- 

96 event : Event 

97 station : Station 

98 The station containing the reconstructed electric field. 

99 If it contains multiple electric fields, only the last electric field 

100 will be used, unless another electric field is passed as a keyword-argument 

101 detector : Detector 

102 electric_field : ElectricField | None (default: None) 

103 If not None, reconstruct the energy for this electric field. 

104 Otherwise, reconstruct the last electric field in the station. 

105 Useful if a station contains multiple reconstructed electric fields. 

106 

107 Returns 

108 ------- 

109 rec_energy : float 

110 The reconstructed cosmic ray energy 

111 

112 """ 

113 

114 if not station.is_cosmic_ray(): 

115 self.logger.warning('Event is not a cosmic ray!') 

116 if not station.has_parameter(stnp.zenith) or not station.has_parameter(stnp.azimuth): 

117 self.logger.error('No incoming direction available. Energy can not be reconstructed!') 

118 return 

119 zenith = station.get_parameter(stnp.zenith) 

120 azimuth = station.get_parameter(stnp.azimuth) 

121 site = self.__site 

122 if site is None: 

123 site = detector.get_site(station.get_id()) 

124 if site not in self.__parametrizations.keys(): 

125 self.logger.error('Unsupported site. Please select one of the following: {}'.format(self.__parametrizations.keys())) 

126 raise ValueError 

127 parametrization_for_site = self.__parametrizations[site] 

128 elevation = self.__elevations[site] 

129 

130 if zenith < 30. * units.deg: 

131 self.logger.warning('Zenith angle is smaller than 30deg. Energy reconstruction is likely to be inaccurate!') 

132 if electric_field is None: 

133 n_efields = len(station.get_electric_fields()) 

134 if n_efields == 0: 

135 self.logger.error('No E-field found. Please run the voltageToAnalyticEfieldConverter beforehand!') 

136 return 

137 if n_efields > 1: 

138 self.logger.warning('Multiple E-fields were found. Only the last E-field will be used.') 

139 electric_field = station.get_electric_fields()[-1] 

140 

141 spectrum_slope = electric_field.get_parameter(efp.cr_spectrum_slope) 

142 alpha = hp.get_angle_to_magnetic_field_vector(zenith, azimuth, site) 

143 cs = radiotools.coordinatesystems.cstrafo(zenith, azimuth, site=site) 

144 efield_trace_vxB_vxvxB = cs.transform_to_vxB_vxvxB(cs.transform_from_onsky_to_ground(electric_field.get_trace())) 

145 efield_trace_vxB_vxvxB[0] /= np.sin(alpha) # correct energy fluence for effect of angle to magnetic field 

146 energy_fluence = NuRadioReco.utilities.trace_utilities.get_electric_field_energy_fluence(efield_trace_vxB_vxvxB, electric_field.get_times()) 

147 energy_fluence = np.abs(energy_fluence[0]) + np.abs(energy_fluence[1]) 

148 xmax_distance = self.__atmosphere.get_distance_xmax_geometric(zenith, 750., elevation) # parametrization is for Xmax of 750g/cm^2 

149 if np.any(xmax_distance) < 0: 

150 self.logger.warning( 

151 f"Estimated distance to Xmax is negative for zenith {zenith/units.deg:.0f} and elevation {elevation}. " 

152 "This means Xmax may be below the detector elevation. The absolute value " 

153 "of the distance will be used instead, but note that the resulting energy estimates may be inaccurate" 

154 ) 

155 

156 xmax_distance = np.abs(xmax_distance) # for some zeniths and altitudes the parameterization can become negative 

157 

158 # find out if we are inside or outside of the Cherenkov ring 

159 second_order_spectrum_parameter = electric_field.get_parameter(efp.cr_spectrum_quadratic_term) 

160 if second_order_spectrum_parameter <= spectrum_slope * .1: 

161 scale_parameter = parametrization_for_site['scale'][0][0] * zenith ** 2 + parametrization_for_site['scale'][0][1] * zenith + parametrization_for_site['scale'][0][2] 

162 falloff_parameter = parametrization_for_site['falloff'][0][0] * zenith + parametrization_for_site['falloff'][0][1] 

163 else: 

164 scale_parameter = parametrization_for_site['scale'][1][0] * zenith ** 2 + parametrization_for_site['scale'][1][1] * zenith + parametrization_for_site['scale'][1][2] 

165 falloff_parameter = parametrization_for_site['falloff'][1][0] * zenith + parametrization_for_site['falloff'][1][1] 

166 rec_energy = 1.e18 * np.sqrt(energy_fluence) * (xmax_distance / units.km) / (scale_parameter * np.exp(falloff_parameter * np.abs(spectrum_slope) ** 0.8)) 

167 station.set_parameter(stnp.cr_energy_em, rec_energy) 

168 

169 return rec_energy
