24def get_random_core_positions(xmin, xmax, ymin, ymax, n_cores, seed=None): 

25 """ 

26 Generate random core positions within a rectangle 

27 

28 Parameters 

29 ---------- 

30 xmin: float 

31 minimum x value 

32 xmax: float 

33 maximum x value 

34 ymin: float 

35 minimum y value 

36 ymax: float 

37 maximum y value 

38 n_cores: int 

39 number of cores to generate 

40 seed: int 

41 seed for the random number generator 

42 

43 Returns 

44 ------- 

45 cores: np.ndarray 

46 array containing the core positions, shaped as (n_cores, 2) 

47 """ 

48 random_generator = np.random.RandomState(seed) 

49 

50 # generate core positions randomly within a rectangle 

51 cores = np.array([ 

52 random_generator.uniform(xmin, xmax, n_cores), 

53 random_generator.uniform(ymin, ymax, n_cores), 

54 ]).T 

55 

56 return cores

59def apply_hanning(efield): 

60 """ 

61 Apply a half-Hann window to the electric field in the time domain. This smoothens the edges 

62 to avoid ringing effects. 

63 

64 Parameters 

65 ---------- 

66 efield: np.ndarray 

67 The electric field in the time domain, shaped as (n_samples, n_polarizations) 

68 

69 Returns 

70 ------- 

71 smoothed_efield: np.ndarray 

72 The smoothed trace, shaped as (n_samples, n_polarizations) 

73 """ 

74 smoothed_trace = np.zeros_like(efield) 

75 filter = half_hann_window(efield.shape[0], half_percent=0.1) 

76 for pol in range(efield.shape[1]): 

77 smoothed_trace[:, pol] = efield[:, pol] * filter 

78 return smoothed_trace

81def select_channels_per_station(det, station_id, requested_channel_ids): 

82 """ 

83 Returns a defaultdict object containing the requested channel ids that are in the given station. 

84 This dict contains the channel group ids as keys with lists of channel ids as values. 

85 

86 Parameters 

87 ---------- 

88 det : DetectorBase 

89 The detector object that contains the station 

90 station_id : int 

91 The station id to select channels from 

92 requested_channel_ids : list of int 

93 List of requested channel ids 

94 

95 Returns 

96 ------- 

97 channel_ids : defaultdict 

98 Dictionary with channel group ids as keys and lists of channel ids as values 

99 """ 

100 channel_ids = defaultdict(list) 

101 for channel_id in requested_channel_ids: 

102 if channel_id in det.get_channel_ids(station_id): 

103 channel_group_id = det.get_channel_group_id(station_id, channel_id) 

104 channel_ids[channel_group_id].append(channel_id) 

105 return channel_ids

108class readCoREASDetector: 

109 """ 

110 Use this as default when reading CoREAS files and combining them with a detector. 

111 

112 This module reads the electric fields of a CoREAS file with a star shaped pattern of observers. 

113 The electric field is then interpolated at the positions of the antennas or stations of a detector. 

114 If the angle between magnetic field and shower direction are below about 15 deg, 

115 the interpolation is no longer reliable and the closest observer is used instead. 

116 """ 

117 

118 def __init__(self): 

119 self.__t = 0 

120 self.__t_event_structure = 0 

121 self.__t_per_event = 0 

122 self.__corsika_evt = None 

123 

124 self.coreas_interpolator = None 

125 

126 self.logger = logging.getLogger('NuRadioReco.readCoREASDetector')

128 def begin(self, input_file, interp_lowfreq=30 * units.MHz, interp_highfreq=1000 * units.MHz, 

129 site=None, declination=None, log_level=logging.NOTSET): 

130 """ 

131 Begin method to initialize readCoREASDetector module. 

132 

133 This function creates an Event using the provided CoREAS HDF5 file, using the `coreas.read_CORSIKA7()` function. 

134 The latter takes in the `declination` and `site` parameters in order to specify the declination of the magnetic 

135 field. Then, it creates a `coreasInterpolator.coreasInterpolator()` object with the event and initializes the 

136 electric field interpolator. 

137 

138 Parameters 

139 ---------- 

140 input_file: str 

141 coreas hdf5 file 

142 interp_lowfreq: float, default=30 * units.MHz 

143 lower frequency for the bandpass filter in interpolation, 

144 should be broader than the sensitivity band of the detector 

145 interp_highfreq: float, default=1000 * units.MHz 

146 higher frequency for the bandpass filter in interpolation, 

147 should be broader than the sensitivity band of the detector 

148 declination: float, default=None 

149 The declination to use for the magnetic field, in internal units. Takes precedence over site. 

150 This parameter is passed on to the `coreas.read_CORSIKA7()` function. 

151 site: str, default=None 

152 Instead of declination, a site name can be given to retrieve the declination. 

153 This parameter is passed on to the `coreas.read_CORSIKA7()` function. 

154 log_level: default=logging.NOTSET 

155 log level for the logger 

156 """ 

157 self.logger.setLevel(log_level) 

158 

159 filesize = os.path.getsize(input_file) 

160 if filesize < 18456 * 2: # based on the observation that a file with such a small filesize is corrupt 

161 self.logger.warning("file {} seems to be corrupt".format(input_file)) 

162 

163 self.__corsika_evt = coreas.read_CORSIKA7(input_file, site=site, declination=declination) 

164 self.logger.info( 

165 f"Using coreas simulation {input_file} with " 

166 f"E={self.__corsika_evt.get_first_sim_shower().get_parameter(shp.energy):.2g}eV, " 

167 f"zenith angle = {self.__corsika_evt.get_first_sim_shower().get_parameter(shp.zenith) / units.deg:.2f}deg and " 

168 f"azimuth angle = {self.__corsika_evt.get_first_sim_shower().get_parameter(shp.azimuth) / units.deg:.2f}deg" 

169 ) 

170 

171 self.coreas_interpolator = coreasInterpolator.coreasInterpolator(self.__corsika_evt) 

172 self.coreas_interpolator.initialize_efield_interpolator(interp_lowfreq, interp_highfreq)

174 @register_run() 

175 def run(self, detector, core_position_list, selected_station_channel_ids=None): 

176 """ 

177 run method, get interpolated electric fields for the given detector and core positions and set them in the event. 

178 The trace is smoothed with a half-Hann window to avoid ringing effects. When using short traces, this might have 

179 a significant effect on the result. 

180 

181 Parameters 

182 ---------- 

183 detector: `NuRadioReco.detector.detector_base.DetectorBase` 

184 Detector description of the detector that shall be simulated 

185 core_position_list: list of (list of float) 

186 List of 2D or 3D core positions in the format [[x1, y1, (z1)], [x2, y2, (z2)], ...] 

187 The z coordinate is optional. It is actually encouraged to **not** use it, as it can mess with 

188 the observation level of the event. If not provided, all oberser positions are put at the 

189 observation by the interpolator (this is the behaviour of `coreasInterpolator.get_interp_efield_value`). 

190 selected_station_channel_ids: dict, default=None 

191 A dictionary containing the list of channels IDs to simulate per station. 

192 If None, all channels of all stations in the detector are simulated. 

193 To select a station and simulate all its channels, set its value to None. 

194 

195 Yields 

196 ------ 

197 evt : `NuRadioReco.framework.event.Event` 

198 An Event containing a Station object for every selected station, which holds a SimStation containing 

199 the interpolated ElectricField traces for the selected channels. 

200 

201 Examples 

202 -------- 

203 When running the module, you will probably want to run it with 2-dimensional core positions. This 

204 ensures that the observation level is not altered. It is possible to run with 3-dimensional core positions, 

205 but be careful to have the correct altitude, otherwise unexpected results might occur. 

206 

207 >>> reader = readCoREASDetector() 

208 >>> reader.begin('coreas.hdf5', interp_lowfreq=30 * units.MHz, interp_highfreq=80 * units.MHz) 

209 >>> for evt in reader.run(detector, [[0, 0], [10 * units.m, 10 * units.m]]): 

210 >>> print(evt.get_id()) 

211 

212 If we only want to simulate a subset of the stations of our detector, we can select them by passing a dictionary 

213 for the selected_station_channel_ids parameter. The keys should be the station IDs and the value should be 

214 `None` to indicate we want to simulate all channels. So to simulate all channels of stations 2 and 7, 

215 we would do the following. 

216 

217 >>> my_selection = {2: None, 7: None} 

218 >>> reader = readCoREASDetector() 

219 >>> reader.begin('coreas.hdf5', interp_lowfreq=30 * units.MHz, interp_highfreq=80 * units.MHz) 

220 >>> for evt in reader.run(detector, [[0, 0], [10 * units.m, 10 * units.m]], selected_station_channel_ids): 

221 >>> print(evt.get_id()) 

222 

223 Setting the value to a list of channel IDs will only simulate these selected channels of the station 

224 they are associated to. Here we simulate the first 4 channels of station 2 and the first 2 channels 

225 of station 7, for the case of the LOFAR detector description. 

226 

227 >>> my_selection = {2: [2000000, 2000001, 2000002, 2000003], 7: [7000000, 7000001]} 

228 >>> reader = readCoREASDetector() 

229 >>> reader.begin('coreas.hdf5', interp_lowfreq=30 * units.MHz, interp_highfreq=80 * units.MHz) 

230 >>> for evt in reader.run(detector, [[0, 0], [10 * units.m, 10 * units.m]], selected_station_channel_ids): 

231 >>> print(evt.get_id()) 

232 """ 

233 

234 if selected_station_channel_ids is None: 

235 selected_station_ids = detector.get_station_ids() 

236 selected_station_channel_ids = {station_id: None for station_id in selected_station_ids} 

237 self.logger.info(f"Using all station ids in detector description: {selected_station_ids}") 

238 else: 

239 selected_station_ids = list(selected_station_channel_ids.keys()) 

240 self.logger.info(f"Using selected station ids: {selected_station_ids}") 

241 

242 t = time.time() 

243 t_per_event = time.time() 

244 self.__t_per_event += time.time() - t_per_event 

245 self.__t += time.time() - t 

246 

247 # Loop over all cores 

248 for iCore, core in enumerate(core_position_list): 

249 t = time.time() 

250 

251 # Create the Event and add the SimShower 

252 evt = NuRadioReco.framework.event.Event(self.__corsika_evt.get_run_number(), iCore) 

253 corsika_sim_stn = self.__corsika_evt.get_station(0).get_sim_station() 

254 sim_shower = copy.deepcopy(self.__corsika_evt.get_first_sim_shower()) # Don't modify the original shower 

255 new_core = sim_shower.get_parameter(shp.core) 

256 new_core[:len(core)] = core # If no z coordinate is given, keep the original one 

257 sim_shower.set_parameter(shp.core, new_core) 

258 evt.add_sim_shower(sim_shower) 

259 

260 # Loop over all selected stations 

261 for station_id in selected_station_ids: 

262 # Make the (Sim)Station objects to add to the Event 

263 station = NuRadioReco.framework.station.Station(station_id) 

264 sim_station = NuRadioReco.framework.sim_station.SimStation(station_id) 

265 

266 # Copy relevant SimStation parameters over 

267 for key, value in corsika_sim_stn.get_parameters().items(): 

268 sim_station.set_parameter(key, value) 

269 sim_station.set_magnetic_field_vector(corsika_sim_stn.get_magnetic_field_vector()) 

270 sim_station.set_is_cosmic_ray() 

271 

272 # Find all the selected channels for this station 

273 det_station_position = detector.get_absolute_position(station_id) 

274 if selected_station_channel_ids[station_id] is None: 

275 selected_channel_ids = detector.get_channel_ids(station_id) 

276 else: 

277 selected_channel_ids = selected_station_channel_ids[station_id] 

278 

279 # Get the channels in a dictionary with channel group as key and a list of channel ids as value 

280 channel_ids_dict = select_channels_per_station(detector, station_id, selected_channel_ids) 

281 for ch_g_ids, channel_ids_for_group_id in channel_ids_dict.items(): 

282 # Get the absolute antenna position 

283 antenna_position_rel = detector.get_relative_position(station_id, channel_ids_for_group_id[0]) 

284 antenna_position = det_station_position + antenna_position_rel 

285 

286 # Get the interpolated electric field and smooth it 

287 res_efield, res_trace_start_time = self.coreas_interpolator.get_interp_efield_value( 

288 antenna_position[:len(core)] - core # get the trace at the relative distance from the core 

289 ) 

290 smooth_res_efield = apply_hanning(res_efield) 

291 

292 # Store the trace in an ElecticField object 

293 coreas.add_electric_field_to_sim_station( 

294 sim_station, channel_ids_for_group_id, smooth_res_efield.T, res_trace_start_time, 

295 sim_shower[shp.zenith], sim_shower[shp.azimuth], self.coreas_interpolator.sampling_rate, 

296 efield_position=antenna_position_rel) 

297 

298 sim_station.set_parameter(stnp.zenith, sim_shower[shp.zenith]) 

299 sim_station.set_parameter(stnp.azimuth, sim_shower[shp.azimuth]) 

300 station.set_sim_station(sim_station) 

301 

302 evt.set_station(station) 

303 

304 self.__t += time.time() - t 

305 yield evt
