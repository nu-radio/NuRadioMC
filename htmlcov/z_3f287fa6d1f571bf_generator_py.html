71def get_tau_95_length(energies): 

72 """ 

73 Returns a fit to the 95% percentile of the tau track length calculated 

74 with PROPOSAL. We calculate the 95% percentile for the largest energy. 

75 """ 

76 

77 coeffs = [6.80016451e+02, -1.61902120e+02, 1.42383021e+01, -5.47388025e-01, 7.79239697e-03] 

78 log_length = 0 

79 

80 log_energy_eV = np.log10(np.max(energies) / units.eV) 

81 

82 for ipow, coeff in enumerate(coeffs): 

83 log_length += coeff * (log_energy_eV) ** ipow 

84 

85 return 10 ** log_length * units.m

242def get_energy_from_flux(Emin, Emax, n_events, flux, rnd=None): 

243 """ 

244 returns randomly distribution of energy according to a flux 

245 

246 Parameters 

247 ---------- 

248 Emin: float 

249 minumum energy 

250 Emax: float 

251 maximum energy 

252 n_event: int 

253 number of events to generate 

254 flux: function 

255 must return flux as function of energy in units of events per energy, time, solid angle and area 

256 rnd: random generator object 

257 if None is provided, a new default random generator object is initialized 

258 

259 Returns 

260 ------- 

261 energies: array of floats 

262 array of energies 

263 """ 

264 if(rnd is None): 

265 rnd = np.random.default_rng() 

266 xx_edges = np.linspace(Emin, Emax, 10000000) 

267 xx = 0.5 * (xx_edges[1:] + xx_edges[:-1]) 

268 yy = flux(xx) 

269 cum_values = np.zeros(xx_edges.shape) 

270 cum_values[1:] = np.cumsum(yy * np.diff(xx_edges)) 

271 inv_cdf = interpolate.interp1d(cum_values, xx_edges) 

272 r = rnd.uniform(0, cum_values.max(), n_events) 

273 return inv_cdf(r)

276def get_product_position_time(data_sets, product, iE): 

277 

278 """ 

279 Calculates the position of a product particle given by the NuRadioProposal 

280 module that has been created by a PROPOSAL lepton propagation. 

281 

282 Parameters 

283 ---------- 

284 data_sets: dictionary 

285 Dictionary with the data sets from the generating functions 

286 (generate_eventlist_cylinder and generate_surface_muons) 

287 product: secondary_properties class from NuRadioPROPOSAL 

288 Contains the properties of the shower-inducing particle given by PROPOSAL 

289 iE: int 

290 Number of the event in data_sets corresponding to the product particle 

291 

292 Returns 

293 ------- 

294 x, y, z, time: tuple 

295 The 3-D position of the shower-inducing product particle and the time 

296 elapsed since the first interaction to the present interaction 

297 """ 

298 

299 dist = product.distance 

300 prop_time = dist / cspeed 

301 x = data_sets["xx"][iE] - dist * np.sin(data_sets["zeniths"][iE]) * np.cos(data_sets["azimuths"][iE]) 

302 y = data_sets["yy"][iE] - dist * np.sin(data_sets["zeniths"][iE]) * np.sin(data_sets["azimuths"][iE]) 

303 z = data_sets["zz"][iE] - dist * np.cos(data_sets["zeniths"][iE]) 

304 

305 return x, y, z, prop_time

308def get_energies(n_events, Emin, Emax, spectrum_type, rnd=None): 

309 """ 

310 generates a random distribution of enrgies following a certain spectrum 

311 

312 Parameters 

313 ---------- 

314 n_events: int 

315 the total number of events 

316 Emin: float 

317 the minimal energy 

318 Emax: float 

319 the maximum energy 

320 spectrum_type: string 

321 defines the probability distribution for which the neutrino energies are generated 

322 

323 * 'log_uniform': uniformly distributed in the logarithm of energy 

324 * 'E-?': E to the -? spectrum where ? can be any float 

325 * 'IceCube-nu-2017': astrophysical neutrino flux measured with IceCube muon sample (https://doi.org/10.22323/1.301.1005) 

326 * 'GZK-1': GZK neutrino flux model from van Vliet et al., 2019, https://arxiv.org/abs/1901.01899v1 for 

327 10% proton fraction (see get_proton_10 in examples/Sensitivities/E2_fluxes3.py for details) 

328 * 'GZK-2': GZK neutrino flux model from fit to TA data (https://pos.sissa.it/395/338/) 

329 * 'GZK-1+IceCube-nu-2017': a combination of the cosmogenic (GZK-1) and astrophysical (IceCube nu 2017) flux 

330 * 'GZK-1+IceCube-nu-2022': a combination of the cosmogenic (GZK-1) and astrophysical (IceCube nu 2022) flux 

331 * 'GZK-2+IceCube-nu-2022': a combination of the cosmogenic (GZK-2) and astrophysical (IceCube nu 2022) flux 

332 

333 rnd: random generator object 

334 if None is provided, a new default random generator object is initialized 

335 """ 

336 

337 if(rnd is None): 

338 rnd = np.random.default_rng() 

339 logger.debug("generating energies") 

340 if(spectrum_type == 'log_uniform'): 

341 energies = 10 ** rnd.uniform(np.log10(Emin), np.log10(Emax), n_events) 

342 elif(spectrum_type.startswith("E-")): # enerate an E^gamma spectrum. 

343 gamma = float(spectrum_type[1:]) 

344 gamma += 1 

345 Nmin = (Emin) ** gamma 

346 Nmax = (Emax) ** gamma 

347 

348 def get_inverse_spectrum(N, gamma): 

349 return np.exp(np.log(N) / gamma) 

350 

351 energies = get_inverse_spectrum(rnd.uniform(Nmax, Nmin, size=n_events), gamma) 

352 elif(spectrum_type == "GZK-1"): 

353 """ 

354 model of (van Vliet et al., 2019, https://arxiv.org/abs/1901.01899v1) of the cosmogenic neutrino ﬂux 

355 for a source evolution parameter of m = 3.4, 

356 a spectral index of the injection spectrum of α = 2.5, a cut-oﬀ rigidity of R = 100 EeV, 

357 and a proton fraction of 10% at E = 10^19.6 eV 

358 """ 

359 from NuRadioMC.examples.Sensitivities.E2_fluxes3 import get_proton_10 

360 energies = get_energy_from_flux(Emin, Emax, n_events, get_proton_10, rnd) 

361 elif(spectrum_type == "IceCube-nu-2017"): 

362 energies = get_energy_from_flux(Emin, Emax, n_events, ice_cube_nu_fit, rnd) 

363 elif(spectrum_type == "IceCube-nu-2022"): 

364 energies = get_energy_from_flux(Emin, Emax, n_events, ice_cube_nu_fit_2022, rnd) 

365 elif(spectrum_type == "GZK-1+IceCube-nu-2017"): 

366 from NuRadioMC.examples.Sensitivities.E2_fluxes3 import get_proton_10 

367 

368 def J(E): 

369 return ice_cube_nu_fit(E) + get_proton_10(E) 

370 

371 energies = get_energy_from_flux(Emin, Emax, n_events, J, rnd) 

372 elif(spectrum_type == "GZK-1+IceCube-nu-2022"): 

373 from NuRadioMC.examples.Sensitivities.E2_fluxes3 import get_proton_10 

374 

375 def J(E): 

376 return ice_cube_nu_fit_2022(E) + get_proton_10(E) 

377 

378 energies = get_energy_from_flux(Emin, Emax, n_events, J, rnd) 

379 elif(spectrum_type == "GZK-2+IceCube-nu-2022"): 

380 from NuRadioMC.examples.Sensitivities.E2_fluxes3 import get_TAGZK_flux_ICRC2021 

381 

382 def J(E): 

383 return ice_cube_nu_fit_2022(E) + get_TAGZK_flux_ICRC2021(E) 

384 

385 energies = get_energy_from_flux(Emin, Emax, n_events, J, rnd) 

386 else: 

387 logger.error("spectrum {} not implemented".format(spectrum_type)) 

388 raise NotImplementedError("spectrum {} not implemented".format(spectrum_type)) 

389 return energies

392def set_volume_attributes(volume, proposal, attributes): 

393 """ 

394 Helper function that interprets the volume settings and sets the relevant quantities as attributes dictinary. 

395 The relevant quantities to define the volume in which neutrinos are generated are (rmin, rmax, zmin, zmax) or 

396 (xmin, xmax, ymin, ymax, zmin, zmax) based on the user settings (see explanation blow). 

397 

398 The user can specify a "fiducial" (mandatory) and "full" (optional) volume. In the end only events are 

399 considered (= passed to radio simulation) which produce a shower (over a configurable energy threshold) 

400 within the fiducial volume. However, neutrino vertices might be generated outside this fiducial volume 

401 but the generated charge lepton (mu or tau) reaches the fiducial volume and produces a trigger 

402 (this obviously only makes sense when running proposal). Hence, a "full" volume can be defined in which 

403 neutrinos interactions are generated. If the "full" volume is not specified in the case you run proposal, 

404 the volume is determind using the energy dependent mean-free-path length of the tau. 

405 

406 TL;DR 

407 - The fiducial volume needs to be chosen large enough such that no events outside of it will trigger. 

408 - The full volume is optional and most of the times it is not necessary to specify it (because it is 

409 automatically determined). 

410 

411 Parameters 

412 ---------- 

413 volume: dict 

414 A dictionary specifying the simulation volume. Can be either a cylinder (specified with min/max radius and z-coordinate) 

415 or a cube (specified with min/max x-, y-, z-coordinates). 

416 

417 Cylinder: 

418 

419 * fiducial_{rmin,rmax,zmin,zmax}: float 

420 lower r / upper r / lower z / upper z coordinate of fiducial volume 

421 * full_{rmin,rmax,zmin,zmax}: float (optional) 

422 lower r / upper r / lower z / upper z coordinate of simulated volume 

423 

424 Cube: 

425 

426 * fiducial_{xmin,xmax,ymin,ymax,zmin,zmax}: float 

427 lower / upper x-, y-, z-coordinate of fiducial volume 

428 * full_{xmin,xmax,ymin,ymax,zmin,zmax}: float (optional) 

429 lower / upper x-, y-, z-coordinate of simulated volume 

430 

431 Optinal you can also define the horizontal center of the volume (if you want to displace it from the origin) 

432 

433 * x0, y0: float 

434 x-, y-coordinate this shift the center of the simulation volume 

435 proposal: bool 

436 specifies if secondary interaction via proposal are calculated 

437 attributes: dictionary 

438 dict storing hdf5 attributes 

439 """ 

440 

441 n_events = attributes['n_events'] 

442 

443 attributes["x0"] = volume.get('x0', 0) 

444 attributes["y0"] = volume.get('y0', 0) 

445 

446 if "fiducial_rmax" in volume: # user specifies a cylinder 

447 attributes['fiducial_rmin'] = volume.get('fiducial_rmin', 0) 

448 

449 # copy keys 

450 for key in ['fiducial_rmax', 'fiducial_zmin', 'fiducial_zmax']: 

451 attributes[key] = volume[key] 

452 

453 rmin = attributes['fiducial_rmin'] 

454 rmax = attributes['fiducial_rmax'] 

455 zmin = attributes['fiducial_zmin'] 

456 zmax = attributes['fiducial_zmax'] 

457 volume_fiducial = np.pi * (rmax ** 2 - rmin ** 2) * (zmax - zmin) 

458 

459 if 'full_rmax' in volume: 

460 rmax = volume['full_rmax'] 

461 if 'full_rmin' in volume: 

462 rmin = volume['full_rmin'] 

463 if 'full_zmax' in volume: 

464 zmax = volume['full_zmax'] 

465 if 'full_zmin' in volume: 

466 zmin = volume['full_zmin'] 

467 

468 # We increase the radius of the cylinder according to the tau track length 

469 if proposal: 

470 logger.info("simulation of second interactions via PROPOSAL activated") 

471 rmin = volume.get('full_rmin', attributes['fiducial_rmin'] / 3.) 

472 

473 if 'full_rmax' in volume: 

474 rmax = volume['full_rmax'] 

475 else: 

476 tau_95_length = get_tau_95_length(attributes['Emax']) 

477 # check if thetamax/thetamin are in same quadrant 

478 is_same_quadrant = np.sign(np.cos(attributes['thetamin'])) == np.sign(np.cos(attributes['thetamax'])) 

479 if is_same_quadrant: 

480 max_horizontal_dist = np.abs(max(np.abs(np.tan(attributes['thetamin'])), 

481 np.abs(np.tan(attributes['thetamax']))) * volume['fiducial_zmin']) # calculates the maximum horizontal distance through the ice 

482 else: 

483 # not same quadrant: theta range surpasses pi/2 -> horizontal trajectories 

484 # geometric trajectory is infinite, but not acutally used. distance is limited by the min() to allowed length below 

485 max_horizontal_dist = np.inf 

486 

487 d_extent = min(tau_95_length, max_horizontal_dist) 

488 logger.info(f"95% quantile of tau decay length is {tau_95_length/units.km:.1f}km. Maximum horizontal " 

489 f"distance for zenith range of {attributes['thetamin']/units.deg:.0f} - " 

490 f"{attributes['thetamax']/units.deg:.0f}deg and depth of {volume['fiducial_zmin']/units.km:.1f}km " 

491 f"is {max_horizontal_dist/units.km:.1f}km -> extending horizontal volume by {d_extent/units.km:.1f}km") 

492 

493 rmax = d_extent + attributes['fiducial_rmax'] 

494 

495 zmax = volume.get('full_zmax', attributes['fiducial_zmax'] / 3.) 

496 zmin = volume.get('full_zmin', attributes['fiducial_zmin']) 

497 

498 volume_full = np.pi * (rmax ** 2 - rmin ** 2) * (zmax - zmin) 

499 # increase the total number of events such that we end up with the same number of events in the fiducial volume 

500 n_events = int(n_events * volume_full / volume_fiducial) 

501 logger.info(f"increasing rmax from {attributes['fiducial_rmax'] / units.km:.01f}km to {rmax /units.km:.01f}km, " 

502 f"zmax from {attributes['fiducial_zmax'] / units.km:.01f}km to {zmax / units.km:.01f}km") 

503 

504 if attributes['fiducial_rmin'] != rmin: 

505 logger.info(f"decreasing rmin from {attributes['fiducial_rmin'] / units.km:.01f}km to {rmin / units.km:.01f}km") 

506 

507 if attributes['fiducial_zmin'] != zmin: 

508 logger.info(f"decreasing zmin from {attributes['fiducial_zmin'] / units.km:.01f}km to {zmin / units.km:.01f}km") 

509 

510 n_events = int(n_events * volume_full / volume_fiducial) 

511 logger.info(f"increasing number of events from {attributes['n_events']:.6g} to {n_events:.6g}") 

512 attributes['n_events'] = n_events 

513 

514 attributes['rmin'] = rmin 

515 attributes['rmax'] = rmax 

516 attributes['zmin'] = zmin 

517 attributes['zmax'] = zmax

630def intersection_box_ray(bounds, ray): 

631 """ 

632 this function calculates the intersection between a ray and an axis-aligned box 

633 code adapted from https://www.scratchapixel.com/lessons/3d-basic-rendering/minimal-ray-tracer-rendering-simple-shapes/ray-box-intersection 

634 

635 Parameters 

636 ---------- 

637 box: array with shape (2,3) 

638 definition of box with two points 

639 ray: array with shape (2,3) 

640 definiton of ray using origin and direction 3-dim vectors 

641 """ 

642 orig = np.array(ray[0]) 

643 direction = np.array(ray[1]) 

644 invdir = 1 / direction 

645 sign = np.zeros(3, dtype=int) 

646 sign[0] = (invdir[0] < 0) 

647 sign[1] = (invdir[1] < 0) 

648 sign[2] = (invdir[2] < 0) 

649 

650 tmin = (bounds[sign[0]][0] - orig[0]) * invdir[0] 

651 tmax = (bounds[1 - sign[0]][0] - orig[0]) * invdir[0] 

652 tymin = (bounds[sign[1]][1] - orig[1]) * invdir[1] 

653 tymax = (bounds[1 - sign[1]][1] - orig[1]) * invdir[1] 

654 

655 if ((tmin > tymax) or (tymin > tmax)): 

656 return False 

657 if (tymin > tmin): 

658 tmin = tymin 

659 if (tymax < tmax): 

660 tmax = tymax 

661 

662 tzmin = (bounds[sign[2]][2] - orig[2]) * invdir[2] 

663 tzmax = (bounds[1 - sign[2]][2] - orig[2]) * invdir[2] 

664 

665 if ((tmin > tzmax) or (tzmin > tmax)): 

666 return False 

667 if (tzmin > tmin): 

668 tmin = tzmin 

669 if (tzmax < tmax): 

670 tmax = tzmax 

671 t = tmin 

672 if (t < 0): 

673 t = tmax 

674 if (t < 0): 

675 return False # I think this removes events where the box is behind the the neutrino interaction which is what we want 

676 return True

708def mask_arrival_azimuth(data_sets, fiducial_rmax): 

709 

710 # Now we filter the events as a function of their arrival direction to 

711 # save computing time. Those events that won't make it to the fiducial 

712 # cylinder are discarded. 

713 

714 rhos = np.sqrt(data_sets['xx'] ** 2 + data_sets['yy'] ** 2) 

715 

716 # Let us considered our problem seen from above, projected on the z = vertex_z plane. 

717 # tangent_angle is the angle a tangent to the cylinder that reaches the vertex makes 

718 # with the antenna-vertex line on that plane. 

719 sine_tangent_angles = fiducial_rmax / rhos 

720 sine_tangent_angles[sine_tangent_angles > 1] = 1 

721 tangent_angles = np.arcsin(sine_tangent_angles) 

722 phis_low = 2 * np.pi - tangent_angles 

723 phis_high = tangent_angles 

724 phis_0 = np.arctan2(data_sets['yy'], data_sets['xx']) 

725 # NuRadioMC azimuth angles span [0, 2pi), unlike the result of arctan2: [-pi, pi) 

726 phis_0[phis_0 < 0] += 2 * np.pi 

727 phis = data_sets["azimuths"] - phis_0 # phi is the azimuth angle of the incoming neutrino if 

728 # we take phi = 0 as the vertex position 

729 phis[phis < 0] += 2 * np.pi 

730 

731 mask_phi = [ (phi > phi_low and phi < 2 * np.pi) or (phi < phi_high and phi > 0) or rho < fiducial_rmax 

732 for phi, phi_low, phi_high, rho in zip(phis, phis_low, phis_high, rhos) ] 

733 

734 mask_phi = np.array(mask_phi) 

735 

736 return mask_phi

998 logger.info(f"Time per event: {time_per_evt*1e3:.01f}ms") 

999 logger.info(f"Total time {pretty_time_delta(proposal_time)}") 

1000 

1001 logger.status(f"number of fiducial showers {len(data_sets_fiducial['flavors'])}") 

1002

1004 # write_events_to_hdf5 will cause the program to crash. However, we need 

1005 # the output file to have empty data sets but also to have the total 

1006 # number of input muons even though none of them triggers, so as not to 

1007 # bias an effective volume calculation done with several files. 

1008 # As a solution, we take a muon neutrino event (not an atmospheric muon) 

1009 # at the top of the ice, and since its inelasticity is zero, it won't create 

1010 # an electric field or trigger. 

1011 if len(data_sets_fiducial["event_group_ids"]) == 0: 

1012 for key, value in data_sets.items(): 

1013 data_sets_fiducial[key] = np.array([data_sets[key][0]]) 

1014 data_sets_fiducial['flavors'] = np.array([14]) 

1015 data_sets_fiducial['shower_energies'] = np.array([0]) 

1016 

1017 data_sets_fiducial["shower_ids"] = np.arange(0, len(data_sets_fiducial['shower_energies']), dtype=int) 

1018 write_events_to_hdf5(filename, data_sets_fiducial, attributes, n_events_per_file=n_events_per_file, start_file_id=start_file_id) 

1019 logger.status(f"finished in {pretty_time_delta(time.time() - t_start)}") 

1020 return None

1039 write_events=True, 

1040 seed=None, 

1041 interaction_type="ccnc", 

1042 cross_sections_model="hedis_bgr18"): 

1043 """ 

1044 Event generator 

1045 

1046 Generates neutrino interactions, i.e., vertex positions, neutrino directions, 

1047 neutrino flavor, charged currend/neutral current and inelastiviy distributions. 

1048 All events are saved in an hdf5 file.

1051 ----------

1058 Emax: float

1067 lower r / upper r / lower z / upper z coordinate of fiducial volume

1071 Cube: 

1072 

1073 * fiducial_{xmin,xmax,ymin,ymax,zmin,zmax}: float 

1074 lower / upper x-, y-, z-coordinate of fiducial volume

1099 particles have positive sign, anti-particles have negative sign, 

1100 relevant for us are:

1122 * 'GZK-1+IceCube-nu-2022': a combination of the cosmogenic (GZK-1) and astrophysical (IceCube nu 2022) flux

1124 

1125 deposited: bool 

1126 if True, generate deposited energies instead of primary neutrino energies

1128 if True, the tau and muon secondaries are calculated using PROPOSAL 

1129 proposal_config: string or path 

1130 The user can specify the path to their own config file or choose among 

1131 the available options: 

1132 

1133 * 'SouthPole', a config file for the South Pole (spherical Earth). It 

1134 consists of a 2.7 km deep layer of ice, bedrock below and air above. 

1135 * 'MooresBay', a config file for Moore's Bay (spherical Earth). It 

1136 consists of a 576 m deep ice layer with a 2234 m deep water layer below, 

1137 and bedrock below that. 

1138 * 'InfIce', a config file with a medium of infinite ice 

1139 * 'Greenland', a config file for Summit Station, Greenland (spherical Earth), 

1140 same as SouthPole but with a 3 km deep ice layer. 

1141 proposal_tables_path: path 

1142 path where the proposal cross section tables are stored or should be generated 

1143 start_file_id: int (default 0) 

1144 in case the data set is distributed over several files, this number specifies the id of the first file 

1145 (useful if an existing data set is extended) 

1146 log_level: logging log level or None 

1147 sets the log level in the event generation. None means system default. 

1148 proposal_kwargs: dict 

1149 additional kwargs that are passed to the get_secondaries_array function of the NuRadioProposal class 

1150 max_n_events_batch: int (default 1e6) 

1151 the maximum numbe of events that get generated per batch. Relevant if a fiducial volume cut is applied) 

1152 write_events: bool 

1153 if True (default) the event list is written to an output file 

1154 if False the event datasets + atrributes are returned 

1155 seed: None of int 

1156 seed of the random state 

1157 interaction_type: string 

1158 the interaction type. default is "ccnc" which randomly choses neutral current (NC) or charged-current (CC) interactions. 

1159 The use can also specify "nc" or "cc" to exclusively simulate NC or CC interactions 

1160 """ 

1161 rnd = Generator(Philox(seed)) 

1162 t_start = time.time() 

1163 

1164 if log_level is not None: 

1165 logger.setLevel(log_level) 

1166 

1167 if proposal: 

1168 from NuRadioMC.EvtGen.NuRadioProposal import ProposalFunctions 

1169 proposal_functions = ProposalFunctions(config_file=proposal_config, tables_path=proposal_tables_path) 

1170 max_n_events_batch = int(max_n_events_batch) 

1171 attributes = {} 

1172 n_events = int(n_events) 

1173 

1174 # sanity check 

1175 for f in flavor: 

1176 if f not in [12, -12, 14, -14, 16, -16]: 

1177 logger.error(f"Input illegal flavor: {flavor}") 

1178 raise ValueError(f"Input illegal flavor: {flavor}") 

1179 

1180 # save current NuRadioMC version as attribute 

1181 # save NuRadioMC and NuRadioReco versions 

1182 attributes['NuRadioMC_EvtGen_version'] = NuRadioMC.__version__ 

1183 attributes['NuRadioMC_EvtGen_version_hash'] = version.get_NuRadioMC_commit_hash() 

1184 attributes['start_event_id'] = start_event_id 

1185 attributes['n_events'] = n_events 

1186 attributes['flavors'] = flavor 

1187 attributes['Emin'] = Emin 

1188 attributes['Emax'] = Emax 

1189 attributes['thetamin'] = thetamin 

1190 attributes['thetamax'] = thetamax 

1191 attributes['phimin'] = phimin 

1192 attributes['phimax'] = phimax 

1193 attributes['deposited'] = deposited

1195 data_sets = {} 

1196 data_sets_fiducial = {} 

1197 

1198 time_proposal = 0 

1199 

1200 set_volume_attributes(volume, proposal=proposal, attributes=attributes) 

1201 n_events = attributes['n_events'] # important! the number of events might have been increased by the generate vertex function 

1202 n_batches = int(np.ceil(n_events / max_n_events_batch)) 

1203 for i_batch in range(n_batches): # do generation of events in batches 

1204 n_events_batch = max_n_events_batch 

1205 

1206 if i_batch + 1 == n_batches: # last batch? 

1207 n_events_batch = n_events - (i_batch * max_n_events_batch) 

1208

1214 

1215 # fmask = (rr_full >= fiducial_rmin) & (rr_full <= fiducial_rmax) & (data_sets["zz"] >= fiducial_zmin) & (data_sets["zz"] <= fiducial_zmax) # fiducial volume mask

1231 if interaction_type == "ccnc": 

1232 data_sets["interaction_type"] = inelasticities.get_ccnc(

1242 logger.debug("generating inelasticities")

1269 idx_to_copy = orig_idx + n_inserted # orig idx in array with already inserted entries

1273 data_sets['shower_energies'][idx_to_insert] = \ 

1274 (1 - data_sets['inelasticity'][idx_to_copy]) * data_sets['energies'][idx_to_copy] 

1275 data_sets['shower_type'][idx_to_insert] = 'em' 

1276 

1277 logger.debug("converting to numpy arrays") 

1278 # make all arrays numpy arrays 

1279 for key in data_sets: 

1280 data_sets[key] = np.array(data_sets[key])

1283 logger.debug("starting proposal simulation") 

1284 init_time = time.time()

1286 for key, value in iteritems(data_sets): 

1287 if(key not in data_sets_fiducial): 

1288 data_sets_fiducial[key] = []

1298 lepton_codes = copy.copy(data_sets["flavors"])

1310 [-np.sin(theta) * np.cos(phi), -np.sin(theta) * np.sin(phi), -np.cos(theta)]

1412 

1413 logger.status(f"Generated {len(data_sets_fiducial['event_group_ids'])} event groups in {pretty_time_delta(time.time() - t_start)}")
