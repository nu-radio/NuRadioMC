20class triggerSimulator(phasedTrigger): 

21 """ 

22 Calculates the trigger for a envelope phased array. 

23 The channels that participate in the beam forming and the pointing angle for each 

24 beam can be specified. The envelope filter is implemented using the model 

25 for the ARA tunnel diode specified in utilities.diodeSimulator. 

26 

27 See https://arxiv.org/pdf/1903.11043.pdf 

28 and https://elog.phys.hawaii.edu/elog/anita_notes/080827_041639/powertrigger.pdf 

29 """

31 def envelope_trigger(self, 

32 station, 

33 det, 

34 beam_rolls, 

35 triggered_channels, 

36 threshold_factor, 

37 power_mean, 

38 power_std, 

39 output_passband=(None, 200 * units.MHz), 

40 cut_times=(None, None), 

41 trigger_adc=False): 

42 """ 

43 Calculates the envelope trigger for a certain phasing configuration. 

44 Beams are formed. Then, each channel to be phased is filtered with a 

45 tunnel diode, the outputs are phased and, if the minimum is lower than 

46 the number of antennas times (power_mean - power_std * np.abs(threshold_factor)), 

47 a trigger is created. 

48 

49 Parameters 

50 ---------- 

51 station: Station object 

52 Description of the current station 

53 det: Detector object 

54 Description of the current detector 

55 beam_rolls: array of ints 

56 Contains the integers for rolling the voltage traces (delays) 

57 triggered_channels: array of ints 

58 Ids of the triggering channels 

59 threshold_factor: float 

60 the threshold factor 

61 power_mean: float 

62 mean of the noise trace after being filtered with the diode 

63 power_std: float 

64 standard deviation of the noise trace after being filtered with the 

65 diode. power_mean and power_std can be calculated with the function 

66 calculate_noise_parameters from utilities.diodeSimulator 

67 output_passband: (float, float) tuple 

68 Frequencies for a 6th-order Butterworth filter to be applied after 

69 the diode filtering. 

70 cut_times: (float, float) tuple 

71 Times for cutting the trace after diode filtering. This helps reducing 

72 the number of noise-induced triggers. Doing it the other way, that is, 

73 cutting and then filtering, will create two artificial pulses on the 

74 edges of the trace. 

75 trigger_adc: bool 

76 If True, analog to digital conversion is performed. It must be specified in the 

77 detector file. See analogToDigitalConverter module for information 

78 

79 Returns 

80 ------- 

81 is_triggered: bool 

82 True if the triggering condition is met 

83 trigger_delays: dictionary 

84 the delays for the primary channels that have caused a trigger. 

85 If there is no trigger, it's an empty dictionary 

86 """ 

87 station_id = station.get_id() 

88 

89 diode = diodeSimulator(output_passband) 

90 

91 traces = {} 

92 for channel in station.iter_channels(use_channels=triggered_channels): 

93 channel_id = channel.get_id() 

94 time_step = 1 / channel.get_sampling_rate() 

95 

96 if trigger_adc: 

97 

98 ADC = analogToDigitalConverter() 

99 trace = ADC.get_digital_trace(station, det, channel, 

100 trigger_adc=trigger_adc, 

101 random_clock_offset=True, 

102 adc_type='perfect_floor_comparator', 

103 diode=diode) 

104 time_step = 1 / det.get_channel(station_id, channel_id)['trigger_adc_sampling_frequency'] 

105 times = np.arange(len(trace), dtype=float) * time_step 

106 times += channel.get_trace_start_time() 

107 

108 else: 

109 

110 trace = diode.tunnel_diode(channel) # get the enveloped trace 

111 times = np.copy(channel.get_times()) # get the corresponding time bins 

112 

113 if cut_times != (None, None): 

114 left_bin = np.argmin(np.abs(times - cut_times[0])) 

115 right_bin = np.argmin(np.abs(times - cut_times[1])) 

116 trace[0:left_bin] = 0 

117 trace[right_bin:None] = 0 

118 

119 traces[channel_id] = trace[:] 

120 

121 for subbeam_rolls in beam_rolls: 

122 

123 phased_trace = None 

124 # Number of antennas: primary beam antennas 

125 Nant = len(beam_rolls[0]) 

126 

127 for channel_id in traces: 

128 

129 trace = traces[channel_id] 

130 

131 if(phased_trace is None): 

132 phased_trace = np.roll(trace, subbeam_rolls[channel_id]) 

133 else: 

134 phased_trace += np.roll(trace, subbeam_rolls[channel_id]) 

135 

136 low_trigger = power_mean - power_std * np.abs(threshold_factor) 

137 low_trigger *= Nant 

138 

139 threshold_passed = np.min(phased_trace) < low_trigger 

140 

141 if threshold_passed: 

142 trigger_delays = {} 

143 for channel_id in subbeam_rolls: 

144 trigger_delays[channel_id] = subbeam_rolls[channel_id] * time_step 

145 logger.debug("Station has triggered") 

146 return True, trigger_delays 

147 

148 return False, {}

151 def run(self, evt, station, det, 

152 threshold_factor=6.5, 

153 power_mean=None, 

154 power_std=None, 

155 triggered_channels=None, 

156 trigger_name='envelope_phased_threshold', 

157 phasing_angles=default_angles, 

158 set_not_triggered=False, 

159 ref_index=1.75, 

160 output_passband=(None, 200 * units.MHz), 

161 cut_times=(None, None), 

162 trigger_adc=False): 

163 """ 

164 simulates phased array trigger for each event 

165 

166 Several channels are phased by delaying their signals by an amount given 

167 by a pointing angle. Several pointing angles are possible in order to cover 

168 the sky. The array triggered_channels controls the channels that are phased, 

169 according to the angles phasing_angles. A secondary phasing that is added 

170 to this primary phasing is possible, and it is controlled by the parametres 

171 secondary_channels and secondary_phasing_angles. 

172 

173 Parameters 

174 ---------- 

175 evt: Event object 

176 Description of the current event 

177 station: Station object 

178 Description of the current station 

179 det: Detector object 

180 Description of the current detector 

181 threshold_factor: float 

182 threshold factor to be used for the envelope trigger 

183 power_mean: float 

184 mean of the noise trace after being filtered with the diode 

185 power_std: float 

186 standard deviation of the noise trace after being filtered with the 

187 diode. power_mean and power_std can be calculated with the function 

188 calculate_noise_parameters from utilities.diodeSimulator 

189 triggered_channels: array of ints 

190 channels ids of the channels that form the primary phasing array 

191 if None, all channels are taken 

192 trigger_name: string 

193 name for the trigger 

194 phasing_angles: array of float 

195 pointing angles for the primary beam 

196 set_not_triggered: bool (default: False) 

197 if True not trigger simulation will be performed and this trigger will be set to not_triggered 

198 ref_index: float 

199 refractive index for beam forming 

200 output_passband: (float, float) tuple 

201 Frequencies for a 6th-order Butterworth filter to be applied after 

202 the diode filtering. 

203 cut_times: (float, float) tuple 

204 Times for cutting the trace after diode filtering. This helps reducing 

205 the number of noise-induced triggers. Doing it the other way, that is, 

206 cutting and then filtering, will create two artificial pulses on the 

207 edges of the trace. 

208 trigger_adc: bool 

209 If True, analog to digital conversion is performed. It must be specified in the 

210 detector file. See analogToDigitalConverter module for information 

211 

212 Returns 

213 ------- 

214 is_triggered: bool 

215 True if the triggering condition is met 

216 """ 

217 

218 if (triggered_channels is None): 

219 triggered_channels = [channel.get_id() for channel in station.iter_channels()] 

220 

221 if (power_mean is None) or (power_std is None): 

222 error_msg = 'The power_mean or power_std parameters are not defined. ' 

223 error_msg += 'Please define them. You can use the calculate_noise_parameters ' 

224 error_msg += 'function in utilities.diodeSimulator to do so.' 

225 raise ValueError(error_msg) 

226 

227 if set_not_triggered: 

228 

229 is_triggered = False 

230 trigger_delays = {} 

231 

232 else: 

233 

234 channel_trace_start_time = get_channel_trace_start_time(station, triggered_channels) 

235 

236 logger.debug("primary channels:", triggered_channels) 

237 beam_rolls = get_beam_rolls(station, det, triggered_channels, 

238 phasing_angles, ref_index=ref_index) 

239 

240 is_triggered, trigger_delays = self.envelope_trigger(station, det, beam_rolls, 

241 triggered_channels, threshold_factor, 

242 power_mean, power_std, output_passband, cut_times, 

243 trigger_adc) 

244 

245 trigger = EnvelopePhasedTrigger(trigger_name, threshold_factor, power_mean, power_std, 

246 triggered_channels, phasing_angles, trigger_delays, 

247 output_passband) 

248 trigger.set_triggered(is_triggered) 

249 if is_triggered: 

250 trigger.set_trigger_time(channel_trace_start_time) 

251 else: 

252 trigger.set_trigger_time(None) 

253 station.set_trigger(trigger) 

254 

255 return is_triggered
