11def generate_my_events(filename, n_events): 

12 """ 

13 Event generator skeleton 

14 

15 Parameters 

16 ---------- 

17 filename: string 

18 the output filename of the hdf5 file 

19 n_events: int 

20 number of events to generate 

21 """ 

22 

23 # first set the meta attributes 

24 attributes = {} 

25 n_events = int(n_events) 

26 attributes['n_events'] = n_events # the number of events contained in this file 

27 attributes['start_event_id'] = 0 

28 # define the fiducial simulation volume. Instead of specifying fiducial_rmin and fiducial_rmin one can also specify 

29 # fiducial_xmin, fiducial_xmax, fiducial_ymin and fiducial_ymax 

30 # the concept of the diducial volume is described in the NuRadioMC paper. In short: only interactions in this smaller 

31 # fiducial volume are saved. This is useful for the simulation of secondary interactions. For this dummy example 

32 # the fiduial volume is the same as the full volume. 

33 attributes['fiducial_rmin'] = 0 

34 attributes['fiducial_rmax'] = 1 * units.km 

35 attributes['fiducial_zmin'] = 0 * units.m 

36 attributes['fiducial_zmax'] = -2 * units.km 

37 # define the full simulation volume. Instead of specifying rmin and rmin one can also specify 

38 # xmin, xmax, ymin and ymax 

39 attributes['rmin'] = 0 

40 attributes['rmax'] = 1 * units.km 

41 attributes['zmin'] = 0 * units.m 

42 attributes['zmax'] = -2 * units.km 

43 

44 attributes['volume'] = attributes['rmax'] ** 2 * np.pi * np.abs(attributes['zmax']) 

45 

46 # if only interactions on a surface (e.g. for muons from air showers) are generated, the surface area needs to be 

47 # specified attributes['area'] 

48 

49 # define the minumum and maximum energy 

50 attributes['Emin'] = 1 * units.EeV 

51 attributes['Emax'] = 1 * units.EeV 

52 

53 # the interval of zenith directions 

54 attributes['thetamin'] = 0 

55 attributes['thetamax'] = np.pi 

56 # the interval of azimuths directions 

57 attributes['phimin'] = 0 

58 attributes['phimax'] = 2 * np.pi 

59 

60 # now generate the events and fill all required data sets 

61 # here we fill all data sets with dummy values 

62 

63 # each line/entry specified a particle shower of certain energy. 

64 # In principle only the shower direction (zeniths, azimuths fields), the shower position (xx, yy, zz fields) 

65 # the shower energy, the shower type and the event group id are required. We set them first 

66 data_sets = {} 

67 # the direction of the shower 

68 data_sets["azimuths"] = np.ones(n_events) 

69 data_sets["zeniths"] = np.ones(n_events) 

70 # the position of the shower 

71 data_sets["xx"] = np.ones(n_events) 

72 data_sets["yy"] = np.ones(n_events) 

73 data_sets["zz"] = np.ones(n_events) 

74 # the shower energy 

75 data_sets["shower_energies"] = np.ones(n_events) * 1 * units.EeV 

76 # the shower type (here we only generate hadronic showers). This infomration is needed for the Askaryan emission model 

77 data_sets["shower_type"] = ['had'] * n_events 

78 # give each shower a unique id (we can also have multiple showers for a single event by just giving several showers 

79 # the same event_group_id) 

80 data_sets["event_group_ids"] = np.arange(n_events) 

81 data_sets["shower_ids"] = np.arange(n_events) 

82 

83 # there are a couple of additional parameters that are required to run a NuRadioMC simulations. These parameters 

84 # don't influence the simulated radio signals but are required for other post analysis tasks. If these parameters 

85 # are not relevant for the type of data you're generating, just set them to any value. 

86 

87 # specify which interaction it is (only relevant if multiple showers from the same initial neutrino are simulated) 

88 # here it is just 1 for all events. 

89 data_sets["n_interaction"] = np.ones(n_events, dtype=int) 

90 

91 # the neutrino flavor. Here we only generate electron neutinos which have the integer code 12. 

92 # the neutrino flavor is only used in the calculation of the "weight", i.e. the probability of the neutrino reaching 

93 # the detector. If other particles than a neutrino are simulated, just set the flavor to the corresponding particle code 

94 # following https://pdg.lbl.gov/2019/reviews/rpp2019-rev-monte-carlo-numbering.pdf or just set it to zero. 

95 data_sets["flavors"] = 12 * np.ones(n_events, dtype=int) 

96 # the neutrino energy. This field is also only used for the weight calculation. 

97 data_sets["energies"] = np.ones(n_events) * 1 * units.EeV 

98 

99 # optionally one can also directly set the event weight here (useful if particles other than neutrinos, or calibration 

100 # setups are simulated 

101 # data_sets["weights"] = np.ones(n_events) 

102 

103 # the interaction type. For neutrino interactions is can be either CC or NC. This parameter is not used but passed 

104 # to the output file for information purposes. 

105 data_sets["interaction_type"] = np.full(n_events, "nc", dtype='U2') 

106 # The inelasiticiy, i.e. the fraction of the neutrino energy that is transferred into the hadronic shower. 

107 # This parameter is not used but saved into the output file for information purposes. 

108 data_sets["inelasticity"] = np.ones(n_events) 

109 

110 # write events to file 

111 write_events_to_hdf5(filename, data_sets, attributes)
