35def read_grip_temperature(): 

36 """ 

37 Read temperature file from grip. 

38 

39 Data file from footnote 5 from https://arxiv.org/pdf/2201.07846 

40 

41 Returns 

42 ------- 

43 depths: array of floats 

44 Depths in meters (positive!). 

45 temps: array of floats 

46 Temperatures in Kelvin. 

47 """ 

48 path = os.path.join(os.path.dirname(__file__), 'data/griptemp.txt') 

49 

50 depths = [] 

51 temps = [] 

52 

53 with open(path) as f: 

54 # search for "DATA:" section in file 

55 search = re.search("DATA:", f.read()) 

56 assert search is not None, "No DATA section found in file" 

57 

58 # Jumpe in file to "DATA:" section 

59 f.seek(search.span()[0]) 

60 

61 for line in f.read().split("\n"): 

62 if re.match(r'^\d', line) is not None: # match lines beginning with a digit 

63 line = line.strip("\n") # remove newline character 

64 depth, temp = [float(d) for d in line.split("\t")] 

65 depths.append(depth) 

66 temps.append(temp) 

67 

68 return np.asarray(depths, dtype=float), np.asarray(temps, dtype=float) + 273.15 # convert to Kelvin

99def fit_GL1(z): 

100 """ 

101 Returns the attenuation length at 75 MHz as a function of depth for Greenland 

102 # Model for Greenland. Taken from DOI: https://doi.org/10.3189/2015JoG15J057 

103 

104 Parameters 

105 ---------- 

106 z: float 

107 depth in default units 

108 """ 

109 

110 fit_values = [1.16052586e+03, 6.87257150e-02, -9.82378264e-05, 

111 -3.50628312e-07, -2.21040482e-10, -3.63912864e-14] 

112 min_length = 100 * units.m 

113 

114 if not hasattr(z, '__len__'): 

115 att_length = 0 

116 else: 

117 att_length = np.zeros_like(z) 

118 

119 for power, coeff in enumerate(fit_values): 

120 att_length += coeff * z ** power 

121 

122 if not hasattr(att_length, '__len__'): 

123 if (att_length < min_length): 

124 att_length = min_length 

125 else: 

126 att_length[att_length < min_length] = min_length 

127 

128 return att_length

145def get_attenuation_length(z, frequency, model): 

146 """ 

147 Get attenuation length in ice for different ice models 

148 

149 Parameters 

150 ---------- 

151 z: float 

152 depth in default units 

153 frequency: float 

154 frequency of signal in default units 

155 model: string 

156 Ice model for attenuation length. Options: 

157 

158 * SP1: South Pole model, see various compilation 

159 * GL1: Greenland model, see https://arxiv.org/abs/1409.5413 

160 * GL2: 2021 Greenland model, using the Bogorodsky model for depth dependence 

161 see: https://arxiv.org/abs/2201.07846, specifically Fig. 7 

162 * GL3: 2021 Greenland model, using the MacGregor model for depth dependence 

163 see: https://arxiv.org/abs/2201.07846, specifically Fig. 7 

164 * MB1: Moore's Bay Model, from 10.3189/2015JoG14J214 and 

165 Phd Thesis C. Persichilli (depth dependence) 

166 

167 """ 

168 if model == "SP1": 

169 t = get_temperature(z) 

170 f0 = 0.0001 

171 f2 = 3.16 

172 w0 = np.log(f0) 

173 w1 = 0.0 

174 w2 = np.log(f2) 

175 w = np.log(frequency / units.GHz) 

176 b0 = -6.74890 + t * (0.026709 - t * 0.000884) 

177 b1 = -6.22121 - t * (0.070927 + t * 0.001773) 

178 b2 = -4.09468 - t * (0.002213 + t * 0.000332) 

179 if not hasattr(frequency, '__len__'): 

180 if (frequency < 1. * units.GHz): 

181 a = (b1 * w0 - b0 * w1) / (w0 - w1) 

182 bb = (b1 - b0) / (w1 - w0) 

183 else: 

184 a = (b2 * w1 - b1 * w2) / (w1 - w2) 

185 bb = (b2 - b1) / (w2 - w1) 

186 else: 

187 a = np.ones_like(frequency) * (b2 * w1 - b1 * w2) / (w1 - w2) 

188 bb = np.ones_like(frequency) * (b2 - b1) / (w2 - w1) 

189 a[frequency < 1. * units.GHz] = (b1 * w0 - b0 * w1) / (w0 - w1) 

190 bb[frequency < 1. * units.GHz] = (b1 - b0) / (w1 - w0) 

191 

192 att_length_f = 1. / np.exp(a + bb * w) 

193 

194 elif model == "GL1": 

195 att_length_75 = fit_GL1(z / units.m) 

196 att_length_f = att_length_75 - 0.55 * units.m * (frequency / units.MHz - 75) 

197 

198 elif model == 'GL2': 

199 fit_values_GL2 = [1.20547286e+00, 1.58815679e-05, -2.58901767e-07, -5.16435542e-10, -2.89124473e-13, -4.58987344e-17] 

200 freq_slope = -0.54 * units.m / units.MHz 

201 freq_inter = 852.0 * units.m 

202 

203 bulk_att_length_f = freq_inter + freq_slope * frequency 

204 att_length_f = bulk_att_length_f * np.poly1d(np.flip(fit_values_GL2))(z) 

205 

206 elif model == 'GL3': 

207 if hasattr(z, '__len__'): 

208 # If z is an array, make sure not not use the numba-compiled functions 

209 slopes = _gl3_slope_interpolation(-z) 

210 offsets = _gl3_offset_interpolation(-z) 

211 else: 

212 slopes = gl3_slope_interpolation(-z) 

213 offsets = gl3_offset_interpolation(-z) 

214 

215 # # restric frequency to prevent negative attenuation lengths 

216 # if hasattr(frequency, '__len__'): 

217 # frequency[frequency > 0.6 * units.GHz] = 0.6 * units.GHz 

218 # else: 

219 # if frequency > 0.6 * units.GHz: 

220 # frequency = 0.6 * units.GHz 

221 

222 att_length_f = slopes * frequency + offsets 

223 

224 elif model == "MB1": 

225 # 10.3189/2015JoG14J214 measured the depth-averaged attenuation length as a function of frequency 

226 # the derived parameterization assumed a reflection coefficient of 1 

227 # however a reflection coefficient of 0.82 was measured which results in a slight increase of the derived 

228 # attenution length (correction factor below) 

229 R = 0.82 

230 d_ice = 576 * units.m 

231 att_length_f = 460 * units.m - 180 * units.m / units.GHz * frequency 

232 att_length_f *= (1 + att_length_f / (2 * d_ice) * np.log(R)) ** -1 # additional correction for reflection coefficient being less than 1. 

233 

234 # The temperature dependence of the attenuation length is independent of the frequency dependence 

235 # the relationship between temparature and L is from 10.1063/1.363582 

236 # the temparature profile of the Ross Ice shelf is from 10.1126/science.203.4379.433 and rescaled to the shelf 

237 # thickness of the ARIANNA site (almost linear from -28 - -2deg C. 

238 # this theoretical depth dependence is scaled to match the depth-averaged measurement of the attenuation length. 

239 d = -z * 420. * units.m / d_ice 

240 L = (1250.*0.08886 * np.exp(-0.048827 * (225.6746 - 86.517596 * np.log10(848.870 - (d))))) 

241 # this differs from the equation published in F. Wu PhD thesis UCI. 

242 # 262m is supposed to be the depth averaged attenuation length but the 

243 # integral (int(1/L, 420, 0)/420) ^ -1 = 231.21m and NOT 262m. 

244 att_length_f *= L / 231.21 * units.m 

245 

246 else: 

247 raise NotImplementedError("attenuation model {} is not implemented.".format(model)) 

248 

249 if hasattr(z, '__len__') and not np.any(z <= 0): 

250 logger.warning("You requested the attenuation length for exlusively positive depths, i.e., for air. Return inf for all frequencies.") 

251 

252 min_length = 1 * units.m 

253 if not hasattr(frequency, '__len__') and not hasattr(z, '__len__'): 

254 if att_length_f < min_length: 

255 att_length_f = min_length 

256 if z > 0: 

257 att_length_f = np.inf 

258 else: 

259 att_length_f[att_length_f < min_length] = min_length 

260 att_length_f[z > 0] = np.inf 

261 

262 return att_length_f

268 def gl3_slope_interpolation_f(x): 

269 """ 

270 Interpolation function for the slope of the GL3 model.

279 array-like
