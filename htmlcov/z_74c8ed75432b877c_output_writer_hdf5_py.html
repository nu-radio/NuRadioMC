453 def write_output_file(self): 

454 """ 

455 Write the output file in HDF5 format. 

456 

457 Returns: 

458 bool: False if there are no events to save, True otherwise. 

459 """ 

460 if 'shower_ids' not in self._mout or len(self._mout['shower_ids']) == 0: 

461 logger.warning("no events to save, not writing output file") 

462 return False 

463 

464 folder = os.path.dirname(self._output_filename) 

465 if not os.path.exists(folder) and folder != '': 

466 logger.warning(f"output folder {folder} does not exist, creating folder...") 

467 os.makedirs(folder) 

468 

469 logger.status("Start saving events to hdf5 ...") 

470 fout = h5py.File(self._output_filename, 'w') 

471 

472 # save data sets 

473 # all arrays need to be sorted by shower id 

474 sort = np.argsort(np.array(self._mout['shower_ids'])) 

475 for (key, value) in self._mout.items(): 

476 logger.debug(f"saving {key} {value} {type(value)}") 

477 if np.array(value).dtype.char == 'U': 

478 fout[key] = np.array(value, dtype=h5py.string_dtype(encoding='utf-8'))[sort] 

479 else: 

480 fout[key] = np.array(value)[sort] 

481 

482 # save all data sets of the station groups 

483 keys_per_event = ['event_group_ids', 'event_ids', 'multiple_triggers_per_event', 'trigger_times_per_event', 

484 'maximum_amplitudes', 'maximum_amplitudes_envelope', 'triggered_per_event'] 

485 for (key, value) in self._mout_groups.items(): 

486 sg = fout.create_group("station_{:d}".format(key)) 

487 if 'shower_id' not in value: 

488 continue 

489 sort = np.argsort(np.array(value['shower_id'])) 

490 for (key2, value2) in value.items(): 

491 # a few arrays are counting values for different events, so we need to sort them 

492 if key2 not in keys_per_event: 

493 sg[key2] = np.array(value2)[sort] 

494 else: 

495 sg[key2] = np.array(value2) 

496 

497 # TODO save detector 

498 # if isinstance(self._det, detector.rnog_detector.Detector): 

499 # fout.attrs['detector'] = self._det.export_as_string() 

500 # else: 

501 # with open(self._detectorfile, 'r') as fdet: 

502 # fout.attrs['detector'] = fdet.read() 

503 

504 

505 # Save station level attributes 

506 for station_id in self._mout_groups: 

507 n_channels = self._det.get_number_of_channels(station_id) 

508 positions = np.zeros((n_channels, 3)) 

509 for iCh, channel_id in enumerate(self._det.get_channel_ids(station_id)): 

510 positions[iCh] = self._det.get_relative_position(station_id, channel_id) + self._det.get_absolute_position(station_id) 

511 fout[f"station_{station_id:d}"].attrs['antenna_positions'] = positions 

512 for key in self._mout_groups_attributes[station_id]: 

513 if key not in fout[f"station_{station_id:d}"].attrs: 

514 fout[f"station_{station_id:d}"].attrs[key] = self._mout_groups_attributes[station_id][key] 

515 

516 # Store top level attributes 

517 fout.attrs['config'] = yaml.dump(self._mout_attributes['config']) 

518 

519 # save NuRadioMC and NuRadioReco versions 

520 fout.attrs['NuRadioMC_version'] = NuRadioMC.__version__ 

521 fout.attrs['NuRadioMC_version_hash'] = version.get_NuRadioMC_commit_hash() 

522 

523 for key in self._mout_attributes: 

524 if key not in fout.attrs: 

525 if self._mout_attributes[key] is not None: 

526 fout.attrs[key] = self._mout_attributes[key] 

527 else: 

528 logger.warning(f"attribute {key} is None, not saving it") 

529 

530 fout.close() 

531 return True

533 def calculate_Veff(self): 

534 """ 

535 Calculate the effective volume (Veff) 

536 

537 Returns: 

538 float: The calculated effective volume (Veff) 

539 """ 

540 if not self._mout: # if no events triggered, this is an empty dict 

541 return None 

542 

543 triggered = remove_duplicate_triggers(self._mout['triggered'], self._mout['event_group_ids']) 

544 n_triggered = np.sum(triggered) 

545 try: 

546 n_triggered_weighted = np.sum(np.array(self._mout['weights'])[triggered]) 

547 except KeyError: 

548 logger.debug("No 'weights' present in HDF5 output, assuming all weights are unity.") 

549 n_triggered_weighted = n_triggered 

550 n_events = self._mout_attributes['n_events'] 

551 logger.status(f'Fraction of triggered events = {n_triggered:.0f}/{n_events:.0f}. Sum of triggered event weights = {n_triggered_weighted:.2f}.') 

552 

553 if 'volume' in self._mout_attributes: # this key is not present for e.g. emitter simulations 

554 V = self._mout_attributes['volume'] 

555 Veff = V * n_triggered_weighted / n_events 

556 logger.status(f"Veff = {Veff / units.km ** 3:.4g} km^3, Veffsr = {Veff * 4 * np.pi/units.km**3:.4g} km^3 sr") 

557 return Veff 

558 else: 

559 return None
