118 def __init__(self, n_samples, sampling_rate, Vrms, threshold, time_coincidence, n_majority, time_coincidence_majority, 

119 n_channels, trigger_time, filt, noise_type="rayleigh", keep_full_band=False): 

120 """ 

121 Efficient algorithms to generate thermal noise fluctuations that fulfill a high/low trigger + a majority 

122 coincidence logic (as used by ARIANNA) 

123 

124 Parameters 

125 ---------- 

126 n_samples: int 

127 the number of samples of the trace 

128 sampling_rate: float 

129 the sampling rate 

130 Vrms: float or list of floats 

131 the RMS noise 

132 if only one value is supplied, it is assumed to be the same for all channels 

133 threshold: float or list of floats 

134 the trigger threshold (assuming a symmetric high and low threshold) 

135 if only one value is supplied, it is assumed to be the same for all channels 

136 time_coincidence: float 

137 the high/low coincidence time 

138 n_majority: int 

139 specifies how many channels need to have a trigger 

140 time_coincidence_majority: float 

141 multi channel coincidence window 

142 n_channels: int 

143 number of channels to generate 

144 trigger_time: float 

145 the trigger time (time when the trigger completes) 

146 filt: array of floats or list of array of floats 

147 the filter that should be applied after noise generation (needs to match frequency binning) 

148 if only one array is supplied, it is assumed to be the same for all channels 

149 noise_type: string 

150 the type of the noise, can be 

151 * "rayleigh" (default) 

152 * "noise" 

153 keep_full_band: bool 

154 Flag to keep the full-band waveform instead of the one with the filter applied. Allows for 

155 triggering on a different (i.e. `filt`) band than what is returned, (default is False) 

156 """ 

157 self.n_samples = n_samples 

158 self.sampling_rate = sampling_rate 

159 

160 self.time_coincidence = time_coincidence 

161 self.n_majority = n_majority 

162 self.time_coincidence_majority = time_coincidence_majority 

163 self.trigger_time = trigger_time 

164 self.n_channels = n_channels 

165 self.noise_type = noise_type 

166 

167 self.min_freq = 0 * units.MHz 

168 self.max_freq = 0.5 * self.sampling_rate 

169 self.dt = 1. / self.sampling_rate 

170 self.ff = np.fft.rfftfreq(self.n_samples, 1. / self.sampling_rate) 

171 self.Vrms = {} 

172 if isinstance(Vrms, float): 

173 for i in range(n_channels): 

174 self.Vrms[i] = Vrms 

175 else: 

176 if len(Vrms) != n_channels: 

177 raise ValueError(f"Vrms has {len(Vrms)} values, but {n_channels} channels are requested") 

178 for i in range(n_channels): 

179 self.Vrms[i] = Vrms[i] 

180 self.threshold = {} 

181 if isinstance(threshold, float): 

182 for i in range(n_channels): 

183 self.threshold[i] = threshold 

184 else: 

185 if len(threshold) != n_channels: 

186 raise ValueError(f"threshold has {len(threshold)} values, but {n_channels} channels are requested") 

187 for i in range(n_channels): 

188 self.threshold[i] = threshold[i] 

189 self.filt = {} 

190 if isinstance(filt, np.ndarray): 

191 if filt.ndim == 1: 

192 for i in range(n_channels): 

193 self.filt[i] = filt 

194 elif filt.ndim == 2: 

195 if filt.shape[0] != n_channels: 

196 raise ValueError(f"filt has {filt.shape[0]} values, but {n_channels} channels are requested") 

197 for i in range(n_channels): 

198 self.filt[i] = filt[i] 

199 else: 

200 raise ValueError("filt must be a 1D or 2D numpy array") 

201 elif isinstance(filt, list): 

202 if len(filt) != n_channels: 

203 raise ValueError(f"filt has {len(filt)} values, but {n_channels} channels are requested") 

204 for i in range(n_channels): 

205 self.filt[i] = np.array(filt[i]) 

206 else: 

207 raise ValueError("filt must be a numpy array or a list of numpy arrays") 

208 

209 

210 self.trigger_bin = int(self.trigger_time / self.dt) 

211 self.trigger_bin_low = int((self.trigger_time - self.time_coincidence_majority) / self.dt) 

212 

213 self.norm = {} 

214 self.amplitude = {} 

215 for i in range(n_channels): 

216 self.norm[i] = integrate.trapezoid(np.abs(self.filt[i]) ** 2, self.ff) 

217 self.amplitude[i] = (self.max_freq - self.min_freq) ** 0.5 / self.norm[i] ** 0.5 * self.Vrms[i] 

218 

219 self.noise = channelGenericNoiseAdder.channelGenericNoiseAdder() 

220 

221 self.keep_full_band = keep_full_band

223 def generate_noise(self): 

224 """ 

225 generates noise traces for all channels that will cause a high/low majority logic trigger 

226 

227 Returns np.array of shape (n_channels, n_samples) 

228 """ 

229 n_traces = [None] * self.n_channels 

230 t_bins = [None] * self.n_channels 

231 number_of_triggers = 0 

232 while number_of_triggers < self.n_majority: 

233 for iCh in range(self.n_channels): 

234 if n_traces[iCh] is None: 

235 spec = self.noise.bandlimited_noise(self.min_freq, self.max_freq, self.n_samples, self.sampling_rate, 

236 self.amplitude[iCh], self.noise_type, time_domain=False) 

237 

238 if self.keep_full_band: 

239 trace_copy = fft.freq2time(spec, self.sampling_rate) 

240 

241 spec *= self.filt[iCh] 

242 trace = fft.freq2time(spec, self.sampling_rate) 

243 

244 if np.any(trace > self.threshold[iCh]) and np.any(trace < -self.threshold[iCh]): 

245 triggered_bins = get_high_low_triggers(trace, self.threshold[iCh], -self.threshold[iCh], self.time_coincidence, self.dt) 

246 if np.any(triggered_bins): 

247 number_of_triggers += 1 

248 t_bins[iCh] = triggered_bins 

249 trace_to_keep = trace if not self.keep_full_band else trace_copy 

250 if number_of_triggers == 1: 

251 n_traces[iCh] = np.roll(trace_to_keep, self.trigger_bin - np.argwhere(triggered_bins == True)[0]) 

252 else: 

253 tmp = np.random.randint(self.trigger_bin_low, self.trigger_bin) 

254 n_traces[iCh] = np.roll(trace_to_keep, tmp - np.argwhere(triggered_bins == True)[0]) 

255 if number_of_triggers == self.n_majority: 

256 # this additional break is needed because another channel might trigger 

257 # within the channel loop before the while statement is checked 

258 break 

259 

260 traces = np.zeros((self.n_channels, self.n_samples)) 

261 for iCh in range(self.n_channels): 

262 if(n_traces[iCh] is not None): 

263 traces[iCh] = n_traces[iCh] 

264 else: 

265 spec = self.noise.bandlimited_noise(self.min_freq, self.max_freq, self.n_samples, self.sampling_rate, 

266 self.amplitude[iCh], type=self.noise_type, time_domain=False) 

267 

268 if self.keep_full_band: 

269 traces[iCh] = fft.freq2time(spec, self.sampling_rate) 

270 else: 

271 traces[iCh] = fft.freq2time(spec * self.filt[iCh], self.sampling_rate) 

272 return traces

464 def __triggering(self): 

465 """ separated trigger part for PA noise trigger """ 

466 

467 self.max_amp = 0 

468 # take square over entire array 

469 coh_sum_squared = self._phased_traces ** 2 

470 # bin the data into windows of length self.step and normalise to step length 

471 reduced_array = np.add.reduceat(coh_sum_squared.T, np.arange(0, np.shape(coh_sum_squared)[1], self.step)).T / self.step 

472 

473 sliding_windows = [] 

474 # self.window can extend over multiple steps, 

475 # assuming self.window being an integer multiple of self.step the reduction sums over subsequent steps 

476 steps_per_window = self.window // self.step 

477 # better extend the array in order to also trigger on sum of last/first (matching a previous implementation) 

478 extended_reduced_array = np.column_stack([reduced_array, reduced_array[:, 0:steps_per_window]]) 

479 for offset in range(steps_per_window): 

480 # sum over steps_per_window adjacent steps 

481 window_sum = np.add.reduceat(extended_reduced_array.T, np.arange(offset, np.shape(extended_reduced_array)[1], steps_per_window)).T / steps_per_window 

482 sliding_windows.append(window_sum) 

483 # self.max_amp = max(np.array(sliding_windows).max(), self.max_amp) 

484 self.max_amp = np.array(sliding_windows).max() 

485 

486 # check if trigger condition is fulfilled anywhere 

487 if self.max_amp > self.threshold: 

488 sliding_windows = np.array(sliding_windows) 

489 tmp = np.argwhere(sliding_windows > self.threshold) 

490 triggered_step, triggered_beam, triggered_bin = tmp[0] 

491 triggered_bin *= (self.window + triggered_step * steps_per_window) 

492 if(self.debug): 

493 # check in which beam the trigger condition was fulfilled 

494 sliding_windows = np.concatenate(sliding_windows, axis=1) 

495 triggered_beams = np.amax(sliding_windows, axis=1) > self.threshold 

496 for iBeam, is_triggered in enumerate(triggered_beams): 

497 # print out each beam that has triggered 

498 if is_triggered: 

499 logger.info(f"triggered at beam {iBeam}") 

500 import matplotlib.pyplot as plt 

501 fig, ax = plt.subplots(self.n_channels + 1, 1, sharex=True) 

502 for iCh in range(self.n_channels): 

503 ax[iCh].plot(self._traces[iCh]) 

504 logger.info(f"{self._traces[iCh].std():.2f}") 

505 ax[self.n_channels].plot(self._phased_traces[iBeam]) 

506 fig.tight_layout() 

507 plt.show() 

508 return True, triggered_bin, triggered_beam 

509 return False, None, None

511 def __triggering_strided(self): 

512 """ separated trigger part for PA noise trigger using np.lib.stride_tricks.as_strided """ 

513 

514 self.max_amp = 0 

515 for iBeam, phased_trace in enumerate(self._phased_traces): 

516 # Create a sliding window 

517 coh_sum_squared = phased_trace ** 2 

518 num_frames = int(np.floor((len(phased_trace) - self.window) / self.step)) 

519 coh_sum_windowed = np.lib.stride_tricks.as_strided(coh_sum_squared, (num_frames, self.window), 

520 (coh_sum_squared.strides[0] * self.step, coh_sum_squared.strides[0])) 

521 squared_mean = np.sum(coh_sum_windowed, axis=1) / self.window 

522 # self.max_amp = max(squared_mean.max(), self.max_amp) 

523 self.max_amp = max(squared_mean.max(), self.max_amp) 

524 if True in (squared_mean > self.threshold): 

525 triggered_bin = np.where(squared_mean > self.threshold)[0,0] 

526 logger.debug(f"triggered at beam {iBeam}") 

527 if(self.debug): 

528 import matplotlib.pyplot as plt 

529 fig, ax = plt.subplots(self.n_channels+1, 1, sharex=True) 

530 for iCh in range(self.n_channels): 

531 ax[iCh].plot(self._traces[iCh]) 

532 logger.info(f"{self._traces[iCh].std():.2f}") 

533 ax[self.n_channels].plot(self._phased_traces[iBeam]) 

534 fig.tight_layout() 

535 plt.show() 

536 return True, triggered_bin, iBeam 

537 return False, None, None

539 def generate_noise(self, phasing_mode="slice", trigger_mode="binned_sum", debug=False): 

540 """ 

541 generates noise traces for all channels that will cause a high/low majority logic trigger 

542 

543 Parameters 

544 ---------- 

545 phasing_mode: string (default: "slice") 

546 "slice" or "roll", two implementations for phasing by either slicing the array or using 

547 np.roll. The default shows better performance, but "roll" is kept as an alternative 

548 trigger_mode: string (default: "binned_sum") 

549 "binned_sum" or "stride", two implementations for triggering 

550 The default shows better performance, but "stride" is kept as an alternative 

551 debug: 

552 generate debug plot 

553 Returns 

554 ------- 

555 np.array of shape (n_channels, n_samples), index of triggered bin, index of triggered beam 

556 """ 

557 self.debug = debug 

558 

559 # some variables for profiling code 

560 dt_generation = 0 

561 dt_phasing = 0 

562 dt_triggering = 0 

563 

564 # generate empty trace array 

565 self._traces = np.zeros((self.n_channels, self.n_samples * self.upsampling)) 

566 counter = 0 

567 

568 while True: 

569 counter += 1 

570 if(counter % 1000 == 0): 

571 logger.info(f"{counter:d}, {self.max_amp:.2g}, threshold = {self.threshold:.2g}") 

572 # some printout for profiling 

573 logger.info(f"Time consumption: GENERATION: {dt_generation:.4f}, PHASING: {dt_phasing:.4f}, TRIGGER: {dt_triggering:.4f}") 

574 tstart = time.process_time() 

575 

576 self.__generation() 

577 

578 # time profiling generation 

579 dt_generation += time.process_time() - tstart 

580 tstart = time.process_time() 

581 

582 if phasing_mode == "slice": 

583 self.__phasing() 

584 elif phasing_mode == "roll": 

585 # more time consuming attempt to do phasing compared to slicing the array 

586 self.__phasing_roll() 

587 else: 

588 logger.error(f"Requested phasing_mode {phasing_mode}. Only 'slice' and 'roll' are allowed") 

589 raise NotImplementedError(f"Requested phasing_mode {phasing_mode}. Only 'slice' and 'roll' are allowed") 

590 

591 # time profiling phasing 

592 dt_phasing += time.process_time() - tstart 

593 tstart = time.process_time() 

594 

595 if trigger_mode == "binned_sum": 

596 is_triggered, triggered_bin, triggered_beam = self.__triggering() 

597 elif trigger_mode == "stride": 

598 # more time consuming attempt to do triggering compared to taking binned sums 

599 is_triggered, triggered_bin, triggered_beam = self.__triggering_strided() 

600 else: 

601 logger.error(f"Requested trigger_mode {trigger_mode}. Only 'binned_sum' and 'stride' are allowed") 

602 raise NotImplementedError(f"Requested trigger_mode {trigger_mode}. Only 'binned_sum' and 'stride' are supported") 

603 

604 # time profiling trigger 

605 dt_triggering += time.process_time() - tstart 

606 

607 if is_triggered: 

608 triggered_bin = triggered_bin // self.upsampling # the trace is cut in the downsampled version. Therefore, triggered bin is factor of two smaller. 

609 i_low = triggered_bin - self.pre_trigger_bins 

610 i_high = i_low + self.n_samples_trigger 

611 

612 # traces need to be downsampled 

613 # resample and use axis -1 since trace might be either shape (N) for analytic trace or shape (3,N) for E-field 

614 self._traces = scipy.signal.resample(self._traces, np.shape(self._traces)[-1] // self.upsampling, axis=-1) 

615 

616 if (i_low >= 0) and (i_high < self.n_samples): # If range is directly a subset of the waveform 

617 return self._traces[:, i_low:i_high], self._phased_traces, triggered_beam 

618 

619 # Otherwise, roll the waveforms. Safe as long as noise is generated in the freq domain 

620 self._phased_traces = np.roll(self._phased_traces, -i_low * self.upsampling, axis=-1) 

621 self._traces = np.roll(self._traces, -i_low, axis=-1) 

622 return self._traces[:, :self.n_samples_trigger], self._phased_traces, triggered_beam

624 def generate_noise2(self, debug=False): 

625 """ 

626 generates noise traces for all channels that will cause a high/low majority logic trigger 

627 This implementation is slow due to nested python loops each rolling the trace over and over again 

628 

629 Returns np.array of shape (n_channels, n_samples) 

630 """ 

631 traces = np.zeros((self.n_channels, self.n_samples * self.upsampling)) 

632 counter = 0 

633 max_amp = 0 

634 while True: 

635 counter += 1 

636 if(counter % 1000 == 0): 

637 print(f"{counter:d}, {max_amp:.3g}, threshold = {self.threshold:.3g}") 

638 for iCh in range(self.n_channels): 

639 spec = self.noise.bandlimited_noise(self.min_freq, self.max_freq, self.n_samples * self.upsampling, 

640 self.sampling_rate * self.upsampling, 

641 self.amplitude, self.noise_type, time_domain=False) 

642 spec *= self.filt 

643 trace = fft.freq2time(spec, self.sampling_rate * self.upsampling) 

644 

645 if self.quantize: 

646 self._traces[iCh] = perfect_floor_comparator(trace, self.adc_n_bits, self.adc_ref_voltage) 

647 else: 

648 self._traces[iCh] = trace 

649 

650 shifts = np.zeros(self.n_channels, dtype=np.int) 

651 shifted_traces = copy.copy(traces) 

652 for shift1 in np.arange(-100, 100, 4, dtype=np.int): 

653 shifted_traces[1] = np.roll(traces[1], shift1) 

654 shifts[1] = shift1 

655 for shift2 in np.arange(-100, 100, 4, dtype=np.int): 

656 shifts[2] = shift2 

657 shifted_traces[2] = np.roll(traces[2], shift2) 

658 

659 for shift3 in np.arange(-100, 100, 4, dtype=np.int): 

660 shifts[3] = shift3 

661 shifted_traces[3] = np.roll(traces[3], shift3) 

662 phased_trace = np.zeros(self.n_samples * self.upsampling) 

663 for iCh in range(self.n_channels): 

664 phased_trace += shifted_traces[iCh] 

665 

666 # Create a sliding window 

667 coh_sum_squared = phased_trace ** 2 

668 num_frames = int(np.floor((len(phased_trace) - self.window) / self.step)) 

669 coh_sum_windowed = np.lib.stride_tricks.as_strided(coh_sum_squared, (num_frames, self.window), 

670 (coh_sum_squared.strides[0] * self.step, coh_sum_squared.strides[0])) 

671 squared_mean = np.sum(coh_sum_windowed, axis=1) / self.window 

672 max_amp = max(squared_mean.max(), max_amp) 

673 

674 if True in (squared_mean > self.threshold): 

675 logger.info(f"triggered at beam {shifts}") 

676 if(debug): 

677 import matplotlib.pyplot as plt 

678 fig, ax = plt.subplots(5, 1, sharex=True) 

679 for iCh in range(self.n_channels): 

680 ax[iCh].plot(traces[iCh]) 

681 logger.info(f"{traces[iCh].std():.2f}") 

682 ax[4].plot(phased_trace) 

683 fig.tight_layout() 

684 plt.show() 

685 return traces, phased_trace
