83def get_electric_field_energy_fluence( 

84 electric_field_trace, times, signal_window_mask=None, noise_window_mask=None, 

85 return_uncertainty=False, method="noise_subtraction", estimator_kwargs={}): 

86 """ 

87 Returns the energy fluence of each component of a 3-dimensional electric field trace. 

88 

89 Parameters 

90 ---------- 

91 electric_field_trace : numpy.ndarray 

92 The electric field trace to calculate the energy fluence for 

93 times : numpy.ndarray 

94 The time grid for the electric field trace 

95 signal_window_mask : numpy.ndarray (optional) 

96 A boolean mask that selects the signal window in which the energy fluence is calculated 

97 noise_window_mask : numpy.ndarray (optional) 

98 A boolean mask that selects the noise window. Only used if method is "noise_subtraction" 

99 return_uncertainty : bool (optional) 

100 If True, the uncertainty of the energy fluence is returned 

101 method : str (optional) 

102 The method to use for the energy fluence calculation. 

103 Can be either "noise_subtraction" or "rice". 

104 The Rice distribution method implementation is based on the code published by S. Martinelli et al.: https://arxiv.org/abs/2407.18654 

105 

106 Returns 

107 ------- 

108 signal_energy_fluence : numpy.ndarray 

109 The energy fluence of each component of the electric field trace 

110 signal_energy_fluence_error : numpy.ndarray 

111 The uncertainty of the energy fluences. Only returned if return_uncertainty is True 

112 

113 Other Parameters 

114 ---------------- 

115 estimator_kwargs : dict (optional) 

116 Additional keyword arguments passed to ``_get_noise_fluence_estimators`` and ``_get_signal_fluence_estimators`` functions. 

117 Only used if method is "rice". 

118 

119 """ 

120 

121 dt = times[1] - times[0] 

122 

123 if method == "noise_subtraction": 

124 if signal_window_mask is None: 

125 f_signal = np.sum(electric_field_trace ** 2, axis=1) 

126 else: 

127 f_signal = np.sum(electric_field_trace[:, signal_window_mask] ** 2, axis=1) 

128 

129 if noise_window_mask is not None and np.sum(noise_window_mask) > 0: 

130 f_noise = np.sum(electric_field_trace[:, noise_window_mask] ** 2, axis=1) 

131 f_signal -= f_noise * np.sum(signal_window_mask) / np.sum(noise_window_mask) 

132 f_signal[f_signal < 0] = 0 

133 

134 # calculate RMS noise for error estimation 

135 RMSNoise = np.sqrt(np.mean(electric_field_trace[:, noise_window_mask] ** 2, axis=1)) 

136 else: 

137 RMSNoise = None 

138 

139 signal_energy_fluence = f_signal * dt * conversion_factor_integrated_signal 

140 

141 # calculate error if RMSNoise is known: 

142 if RMSNoise is not None and return_uncertainty: 

143 signal_window_duration = sum(signal_window_mask) * dt if signal_window_mask is not None else len(times) * dt 

144 signal_energy_fluence_error = (4 * np.abs(signal_energy_fluence / conversion_factor_integrated_signal) * RMSNoise ** 2 * dt + 2 * signal_window_duration * RMSNoise ** 4 * dt) ** 0.5 * conversion_factor_integrated_signal 

145 else: 

146 signal_energy_fluence_error = np.zeros(3) 

147 

148 elif method.lower() == "rice": 

149 signal_energy_fluence = np.zeros(len(electric_field_trace)) 

150 signal_energy_fluence_error = np.zeros(len(electric_field_trace)) 

151 for i_pol in range(len(electric_field_trace)): 

152 noise_estimators, frequencies_window = _get_noise_fluence_estimators( 

153 trace = electric_field_trace[i_pol, :], 

154 times = times, 

155 signal_window_mask = signal_window_mask, 

156 **estimator_kwargs 

157 ) 

158 estimators, variances = _get_signal_fluence_estimators( 

159 trace = electric_field_trace[i_pol, :], 

160 times = times, 

161 signal_window_mask = signal_window_mask, 

162 noise_estimators = noise_estimators, 

163 **estimator_kwargs 

164 ) 

165 

166 #sample frequency (after the windowing) in MHz 

167 delta_f = frequencies_window[1] - frequencies_window[0] 

168 

169 #get the fluence of the trace summing up the frequency estimators and converting in eV/m^2 

170 fluence_freq = np.sum(estimators) * delta_f * conversion_factor_integrated_signal 

171 

172 if estimator_kwargs.get("truncate_negative_estimators") == "after_sum": 

173 fluence_freq = np.maximum(fluence_freq, 0) 

174 

175 #get the variance of the trace fluence summing up the frequency variances and converting in (eV/m^2)^2 

176 fluence_freq_variance = np.sum(variances) * (delta_f * conversion_factor_integrated_signal)**2 

177 

178 #get the fluence uncertainty as the root square of the variance 

179 fluence_freq_error = np.sqrt(fluence_freq_variance) 

180 

181 signal_energy_fluence[i_pol] = fluence_freq 

182 signal_energy_fluence_error[i_pol] = fluence_freq_error 

183 else: 

184 raise NotImplementedError(f"method '{method}' is not implemented, valid options are 'noise_subtraction' or 'rice'.") 

185 

186 if return_uncertainty: 

187 return signal_energy_fluence, signal_energy_fluence_error 

188 else: 

189 return signal_energy_fluence

192def _get_noise_fluence_estimators( 

193 trace, times, signal_window_mask, *, 

194 spacing_noise_signal=20*units.ns, relative_taper_width=0.142857143, 

195 use_median_value=False, **kwargs): 

196 """ 

197 Estimate the noise fluence from the trace. 

198 

199 Parameters 

200 ---------- 

201 trace : np.ndarray 

202 Trace to estimate the noise fluence from. 

203 times : np.ndarray 

204 Time grid for the trace. 

205 signal_window_mask : np.ndarray 

206 Boolean mask for the signal window. 

207 spacing_noise_signal : float (optional) 

208 Spacing between noise windows and signal window. Makes sure no signal leaks into the noise windows. (default: 20 ns, 

209 which should be enough for most applications, otherwise the signal_window_mask is too small) 

210 relative_taper_width : float (optional) 

211 Width of the taper region for the Tukey window relative to the full window length. (default: 0.142857143, 

212 which corresponds to 1/7 of the window length at each end) 

213 use_median_value : bool (optional) 

214 If True, the median of the squared spectra of the noise windows is used as estimator. Otherwise, the mean is used. (default: False) 

215 

216 Returns 

217 ------- 

218 estimators : np.ndarray 

219 Estimators for the noise fluence. 

220 frequencies_window : np.ndarray 

221 Frequencies corresponding to the estimators. 

222 """ 

223 

224 dt = times[1] - times[0] 

225 n_samples_window = sum(signal_window_mask) 

226 signal_start = times[signal_window_mask][0] - spacing_noise_signal 

227 signal_stop = times[signal_window_mask][-1] + spacing_noise_signal 

228 list_ffts_squared = [] 

229 

230 if signal_start < times[0]: 

231 signal_start = times[0] 

232 logger.warning("The signal window overlaps with the start of the trace. The efield pulse may be partialy outside the trace.") 

233 elif signal_stop > times[-1]: 

234 logger.warning("The signal window overlaps with the end of the trace. The efield pulse may be partialy outside the trace.") 

235 

236 #generate Tukey window 

237 window = scipy.signal.windows.tukey(n_samples_window, relative_taper_width * 2) 

238 

239 #loop over the trace defining noise windows (excluding the signal window) 

240 noise_start = times[0] 

241 while noise_start < times[-1]: 

242 

243 noise_stop = noise_start + n_samples_window * dt 

244 if noise_stop > times[-1]: 

245 break 

246 

247 elif (noise_stop <= signal_start and noise_start < signal_start) or (noise_stop > signal_stop and noise_start >= signal_stop): 

248 

249 #clipping the noise window (rounding is needed because noise_stop = noise_start + n_samples_window * dt has numerical uncertainties) 

250 mask_time = np.all([np.round(times, 5) >= np.round(noise_start, 5), np.round(times, 5) < np.round(noise_stop, 5)], axis=0) 

251 time_trace_clipped = trace[mask_time] 

252 

253 #applying the Tukey window 

254 windowed_trace = time_trace_clipped * window 

255 

256 #calculating the spectrum and frequencies 

257 frequencies_window = fft.freqs(n_samples_window, 1/dt) 

258 spectrum_window = np.abs(fft.time2freq(windowed_trace, 1/dt)) 

259 

260 list_ffts_squared.append(spectrum_window**2) 

261 noise_start = noise_stop 

262 

263 elif noise_stop > signal_start and noise_start <= signal_start: 

264 noise_start = signal_stop 

265 

266 else: 

267 logger.error("The noise window does not fulfill any of the conditions. This should not happen.") 

268 raise RuntimeError("The noise window does not fulfill any of the conditions. This should not happen.") 

269 

270 list_ffts_squared = np.array(list_ffts_squared, dtype=float) 

271 

272 if use_median_value: 

273 #robust estimator in presence of outliers from the noise windows 

274 estimators = np.median(list_ffts_squared, axis=0) / 1.405 #from chi2 distribution 

275 else: 

276 #it works well in presence of small number of outliers 

277 estimators=np.mean(list_ffts_squared, axis=0) 

278 

279 return estimators, frequencies_window

282def _get_signal_fluence_estimators( 

283 trace, times, signal_window_mask, noise_estimators, *, 

284 relative_taper_width=0.142857143, truncate_negative_estimators="before_sum", **kwargs): 

285 """ 

286 Estimate the signal fluence from the trace. 

287 

288 Parameters 

289 ---------- 

290 trace : np.ndarray 

291 Trace to estimate the signal fluence from. 

292 times : np.ndarray 

293 Time grid for the trace. 

294 signal_window_mask : np.ndarray 

295 Boolean mask for the signal window. 

296 noise_estimators : np.ndarray 

297 Estimators for the noise fluence. 

298 relative_taper_width : float (optional) 

299 Width of the taper region for the Tukey window relative to the full window length. (default: 0.142857143, 

300 which corresponds to 1/7 of the window length at each end) 

301 truncate_negative_estimators : str (optional) 

302 If "before_sum", negative estimators are set to zero before summing over frequencies, 

303 which is consistent with S. Martinelli et al.: https://arxiv.org/abs/2407.18654. 

304 If "after_sum", negative estimators are set to zero after summing over frequencies. Note 

305 that this is done in get_electric_field_energy_fluence. (default: "before_sum") 

306 

307 Returns 

308 ------- 

309 signal_estimators : np.ndarray 

310 Estimators for the signal fluence. 

311 variances : np.ndarray 

312 Variance of the signal fluence estimators. 

313 """ 

314 

315 dt = times[1] - times[0] 

316 n_samples_window = sum(signal_window_mask) 

317 signal_start = times[signal_window_mask][0] 

318 signal_stop = times[signal_window_mask][-1] + dt 

319 

320 #generate Tukey window 

321 window = scipy.signal.windows.tukey(n_samples_window, relative_taper_width * 2) 

322 

323 #clipping the signal window around the pulse position 

324 mask_time = np.all([times >= signal_start, times < signal_stop], axis=0) 

325 trace_clipped = trace[mask_time] 

326 

327 #applying the Tukey window 

328 windowed_trace = trace_clipped * window 

329 

330 #calculating the spectrum and frequencies 

331 spectrum_window = np.abs(fft.time2freq(windowed_trace, 1/dt)) 

332 

333 #signal estimator and variance for each frequency bin 

334 signal_estimators = spectrum_window**2 - noise_estimators 

335 if truncate_negative_estimators == "before_sum": 

336 signal_estimators[signal_estimators < 0] = 0 

337 variances = noise_estimators * (noise_estimators + 2*signal_estimators) 

338 

339 return signal_estimators, variances

342def get_stokes(trace_u, trace_v, window_samples=128, squeeze=True): 

343 """ 

344 Compute the stokes parameters for electric field traces 

345 

346 Parameters 

347 ---------- 

348 trace_u : 1d array (float) 

349 The u component of the electric field trace 

350 trace_v : 1d array (float) 

351 The v component of the electric field trace. 

352 The two components should have equal lengths, 

353 and the (u, v) coordinates should be perpendicular. 

354 Common choices are (theta, phi) or (vxB, vxvxB) 

355 window_samples : int | None, default: 128 

356 If not None, return a running average 

357 of the stokes parameters over ``window_samples``. 

358 If None, compute the stokes parameters over the 

359 whole trace (equivalent to ``window_samples=len(trace_u)``). 

360 squeeze : bool, default: True 

361 Only relevant if ``window_samples=None``. Squeezes 

362 out the second axis (which has a length of one) 

363 and returns an array of shape (4,) 

364 

365 Returns 

366 ------- 

367 stokes : 2d array of floats 

368 The stokes parameters I, Q, U, V. The shape of 

369 the returned array is ``(4, len(trace_u) - window_samples +1)``, 

370 i.e. stokes[0] returns the I parameter, 

371 stokes[1] corresponds to Q, and so on. 

372 

373 Examples 

374 -------- 

375 For an electric field defined in (eR, eTheta, ePhi) components, 

376 the stokes parameters can be given simply by: 

377 

378 .. code-block:: 

379 

380 get_stokes(electric_field.get_trace()[1], electric_field.get_trace()[2]) 

381 

382 To instead get the stokes parameters in vxB and vxvxB, we need to first obtain 

383 the appropriate electric field components 

384 

385 .. code-block:: 

386 

387 cs = radiotools.coordinatesystems.cstrafo(zenith, azimuth, magnetic_field_vector) 

388 

389 efield_trace_vxB_vxvxB = cs.transform_to_vxB_vxvxB( 

390 cs.transform_from_onsky_to_ground(efield.get_trace()) 

391 ) 

392 

393 """ 

394 

395 assert len(trace_u) == len(trace_v) 

396 h1 = scipy.signal.hilbert(trace_u) 

397 h2 = scipy.signal.hilbert(trace_v) 

398 stokes_i = np.abs(h1)**2 + np.abs(h2)**2 

399 stokes_q = np.abs(h1)**2 - np.abs(h2)**2 

400 uv = 2 * h1 * np.conjugate(h2) 

401 stokes_u = np.real(uv) 

402 stokes_v = np.imag(uv) 

403 stokes = np.array([stokes_i, stokes_q, stokes_u, stokes_v]) 

404 if window_samples == 1: # no need to average 

405 return stokes 

406 elif window_samples is None: 

407 window_samples = len(h1) 

408 

409 stokes = np.asarray([ 

410 scipy.signal.convolve(i, np.ones(window_samples), mode='valid') for i in stokes 

411 ]) 

412 stokes /= window_samples 

413 

414 if squeeze: 

415 return np.squeeze(stokes) 

416 

417 return stokes

440def get_split_trace_noise_RMS(trace, segments=4, lowest=2): 

441 """ 

442 Calculates the noise root mean square (RMS) of a given trace. 

443 

444 This method splits the trace into segments, 

445 then calculates the RMS of each segment, 

446 and then takes the mean of the lowest few segemts' RMS values. 

447 

448 Parameters 

449 ---------- 

450 trace: array of floats 

451 Array containing the trace 

452 segments: int 

453 Amount of segments to cut the trace int 

454 lowest: int 

455 Amount of lowest segment rms values to use when calculating the mean RMS end result 

456 

457 Returns 

458 ------- 

459 noise_root_mean_square: float 

460 The mean of the lowest few segments' RMS values 

461 """ 

462 split_array = np.array_split(trace, segments) 

463 split_array = np.array(split_array, dtype="object") #Objectify dtype to allow timetraces indivisible by amount of segments 

464 rms_of_splits = [np.std(split) for split in split_array] 

465 ordered_rmss = np.sort(rms_of_splits) 

466 lowest_rmss = ordered_rmss[:lowest] 

467 noise_root_mean_square = np.mean(lowest_rmss) 

468 

469 return noise_root_mean_square

472def get_signal_to_noise_ratio(trace, noise_rms, window_size=3): 

473 """ 

474 Computes the Signal to Noise Ratio (SNR) of a given trace. 

475 

476 The signal to noise ratio is calculated as the peak to peak amplitude 

477 within a given window size divided by twice the noise root mean square (RMS). 

478 

479 Parameters 

480 ---------- 

481 trace : array of floats 

482 Trace of a waveform 

483 noise_rms: float 

484 Noise root mean square (RMS) 

485 window_size: int 

486 Coincidence window size (default: 3) 

487 

488 Returns 

489 ------- 

490 signal_to_noise_ratio: float 

491 Signal to Noise Ratio (SNR) value 

492 """ 

493 if not window_size >= 2: 

494 logger.error(f"Window size must be greater-equal 2 (but is {window_size})") 

495 raise ValueError(f"Window size must be greater-equal 2 (but is {window_size})") 

496 

497 p2p = np.amax(peak_to_peak_amplitudes(trace, window_size)) 

498 signal_to_noise_ratio = p2p / (2 * noise_rms) 

499 

500 return signal_to_noise_ratio

503def get_root_power_ratio(trace, times, noise_rms): 

504 """ 

505 Computes the Root Power Ratio (RPR) of a given trace. 

506 

507 It compares the peak signal strength to the baseline noise. 

508 The waveform’s power is smoothed using a 25 ns sliding window, 

509 and the square root of this smoothed power gives the rolling root power. 

510 The RPR is the maximum root power divided by the noise RMS 

511 

512 Parameters 

513 ---------- 

514 trace: array of floats 

515 Trace of a waveform 

516 times: array of floats 

517 Times of a waveform 

518 noise_rms: float 

519 noise root mean square (RMS) 

520 

521 Returns 

522 ------- 

523 root_power_ratio: float 

524 Root Power Ratio (RPR) value 

525 """ 

526 

527 # Calculating RPR (Root Power Ratio) 

528 if noise_rms == 0: 

529 root_power_ratio = np.inf 

530 else: 

531 channel_wf = trace ** 2 

532 

533 # Calculate the smoothing window size based on sampling rate 

534 dt = times[1] - times[0] 

535 sum_win = 25 # Smoothing window in ns 

536 sum_win_idx = int(np.round(sum_win / dt)) # Convert window size to sample points 

537 

538 channel_wf = np.sqrt(scipy.ndimage.uniform_filter1d(channel_wf, size=sum_win_idx, mode='constant')) 

539 

540 # Find the maximum value of the smoothed waveform 

541 max_bin = np.argmax(channel_wf) 

542 max_val = channel_wf[max_bin] 

543 

544 root_power_ratio = max_val / noise_rms 

545 

546 return root_power_ratio

569def get_impulsivity(trace): 

570 """ 

571 Calculates the impulsivity of a signal (trace). 

572 

573 This function computes the impulsivity of a trace by first performing the Hilbert Transform 

574 to obtain an analytic signal (Hilbert envelope) and then determining the 

575 cumulative distribution function (CDF) of the square of the sorted envelope values i.e. power values based on their 

576 closeness to the maximum value. The average of the CDF is then scaled and returned 

577 as the impulsivity value. 

578 

579 Parameters 

580 ---------- 

581 trace: array of floats 

582 Trace of a waveform 

583 

584 Returns 

585 ------- 

586 impulsivity: float 

587 Impulsivity of the signal (scaled between 0 and 1) 

588 """ 

589 

590 envelope = get_hilbert_envelope(trace) 

591 maxv = np.argmax(envelope) 

592 envelope_indexes = np.arange(len(envelope)) ## just a list of indices the same length as the array 

593 closeness = list( 

594 np.abs(envelope_indexes - maxv) 

595 ) ## create an array containing index distance to max voltage (lower the value, the closer it is) 

596 

597 sorted_envelope = np.array([x for _, x in sorted(zip(closeness, envelope))]) 

598 cdf = np.cumsum(sorted_envelope**2) 

599 cdf = cdf / cdf[-1] 

600 

601 impulsivity = (np.mean(np.asarray([cdf])) * 2.0) - 1.0 

602 if impulsivity < 0: 

603 impulsivity = 0.0 

604 

605 return impulsivity

608def get_coherent_sum(trace_set, ref_trace, use_envelope = False): 

609 """ 

610 Generates the coherently-summed waveform (CSW) of a sets of traces. 

611 

612 This function finds the correlation between each trace from a set and the reference trace, 

613 then rolls all traces to align with the reference trace, 

614 and then add them up to get the CSW. 

615 

616 Parameters 

617 ---------- 

618 trace_set: 2-D array of floats 

619 Traces of multiple channel waveforms without the reference trace 

620 ref_trace: 1-D array of floats 

621 Trace of the reference channel 

622 use_envelope: bool 

623 See if users would like to find the correlation between envelopes or just normal traces (default: False) 

624 

625 Returns 

626 ------- 

627 sum_trace: 1-D array of floats 

628 CSW of the set of traces 

629 """ 

630 # Normalize: subtract mean, divide by std (z-score) 

631 def process(trace): 

632 if use_envelope: 

633 trace = get_hilbert_envelope(trace) 

634 return (trace - np.mean(trace, axis=-1, keepdims=True)) / np.std(trace, axis=-1, keepdims=True) 

635 

636 n_samples = len(ref_trace) 

637 ref_processed = process(ref_trace) 

638 

639 # Process all traces 

640 trace_set = np.stack(trace_set) # Make sure it's 2D 

641 traces_processed = process(trace_set) 

642 

643 sum_trace = np.copy(ref_trace) 

644 

645 lag_array = scipy.signal.correlation_lags(n_samples, n_samples, mode='full') 

646 

647 for i, trace in enumerate(trace_set): 

648 corr = scipy.signal.correlate(ref_processed, traces_processed[i], mode='full') / n_samples 

649 best_lag = lag_array[np.argmax(corr)] 

650 aligned_trace = np.roll(trace, best_lag) 

651 sum_trace += aligned_trace 

652 

653 return sum_trace

656def get_entropy(trace, n_hist_bins = 50): 

657 """ 

658 Calculates the shannon entropy (randomness measurement) of a trace. 

659 

660 Parameters 

661 ---------- 

662 trace: array of floats 

663 Trace of a waveform 

664 n_hist_bins: int 

665 Number of bins for the histogram (default: 50) 

666 

667 Returns 

668 ------- 

669 entropy: float 

670 Shannon entropy of the signal (trace) 

671 """ 

672 

673 # Step 1: Discretize the signal into bins 

674 # If density = True, the result is the value of the probability density function at the bin, 

675 # normalized such that the integral over the range is 1. 

676 hist, bin_edges = np.histogram(trace, bins = n_hist_bins, density = True) 

677 

678 # Step 2: Calculate the probability distribution (normalized) 

679 probabilities = hist / np.sum(hist) 

680 

681 # Step 3: Calculate Shannon Entropy 

682 # Using base = 2 for entropy in bits 

683 entropy = scipy.stats.entropy(probabilities, base = 2) 

684 

685 return entropy

760def get_variable_window_size_correlation(data_trace, template_trace, window_size, sampling_rate=3.2*units.GHz, return_time_difference=False, debug=False): 

761 """ 

762 Calculate the correlation between two traces using a variable window size and matrix multiplication 

763 

764 Parameters 

765 ---------- 

766 data_trace: array 

767 full trace of the data event 

768 template_trace: array 

769 full trace of the template 

770 window_size: float 

771 Size of the template window in nanoseconds 

772 sampling_rate: float 

773 sampling rate of the data and template trace 

774 return_time_difference: boolean 

775 if true, the time difference (for the maximal correlation value) between the starting of the data 

776 trace and the starting of the (cut) template trace is returned (returned time is in units.ns) 

777 debug: boolean 

778 if true, debug plots are created 

779 

780 Returns 

781 ------- 

782 correlation : array of floats 

783 time_diff : float, optional 

784 The time difference of the maximal correlation value. Returned only if ``return_time_difference==True`` 

785 """ 

786 # preparing the traces 

787 data_trace = np.asarray(data_trace, dtype=float) 

788 template_trace = np.asarray(template_trace, dtype=float) 

789 

790 # create the template window 

791 window_steps = int(window_size * sampling_rate) 

792 

793 max_amp = np.max(abs(template_trace)) 

794 max_amp_i = np.where(abs(template_trace) == max_amp)[0][0] 

795 lower_bound = int(max_amp_i - window_steps / 3) 

796 upper_bound = int(max_amp_i + 2 * window_steps / 3) 

797 template_trace = template_trace[lower_bound:upper_bound] 

798 

799 # zero padding on the data trace 

800 data_trace = np.append(np.zeros(len(template_trace) - 1), data_trace) 

801 data_trace = np.append(data_trace, np.zeros(len(template_trace) - 1)) 

802 

803 if debug: 

804 plot_data_trace = data_trace.copy() 

805 

806 # only calculate the correlation of the part of the trace where at least 10% of the maximum is visible (fastens the calculation) 

807 max_amp_data = np.max(abs(data_trace)) 

808 help_val = np.where(abs(data_trace) >= 0.1 * max_amp_data)[0] 

809 lower_bound_data = help_val[0] - (len(template_trace) - 1) 

810 upper_bound_data = help_val[len(help_val) - 1] + (len(template_trace) - 1) 

811 data_trace = data_trace[lower_bound_data:upper_bound_data] 

812 

813 # run the correlation using matrix multiplication 

814 dataMatrix = np.lib.stride_tricks.sliding_window_view(data_trace, len(template_trace)) 

815 corr_numerator = dataMatrix.dot(template_trace) 

816 norm_dataMatrix = np.linalg.norm(dataMatrix, axis=1) 

817 norm_template_trace = np.linalg.norm(template_trace) 

818 corr_denominator = norm_dataMatrix * norm_template_trace 

819 correlation = corr_numerator / corr_denominator 

820 

821 max_correlation = np.max(abs(correlation)) 

822 max_corr_i = np.where(abs(np.asarray(correlation)) == max_correlation)[0][0] 

823 

824 if return_time_difference: 

825 # calculate the time difference between the beginning of the template and data trace for the largest correlation value 

826 # time difference is given in ns 

827 time_diff = (max_corr_i + (lower_bound_data - len(template_trace))) / sampling_rate 

828 

829 logger.debug('max correlation: {}'.format(max_correlation)) 

830 if return_time_difference: 

831 logger.debug('time difference: {:.2f} ns'.format(time_diff)) 

832 

833 if debug: 

834 import matplotlib.pyplot as plt 

835 fig, axs = plt.subplots(2) 

836 axs[0].plot(correlation) 

837 axs[0].plot(np.array([np.where(abs(correlation) == max(abs(correlation)))[0][0]]), np.array([max_correlation]), marker="x", markersize=12, color='tab:red') 

838 axs[0].set_ylim(-1.1, 1.1) 

839 axs[0].set_ylabel(r"$\chi$") 

840 axs[0].set_xlabel('N') 

841 axs[1].plot(plot_data_trace, label='complete data trace') 

842 x_data = np.arange(0, len(data_trace), 1) 

843 x_data = x_data + lower_bound_data 

844 axs[1].plot(x_data, data_trace, label='scanned data trace') 

845 x_template = np.arange(0, len(template_trace), 1) 

846 x_template = x_template + max_corr_i + lower_bound_data 

847 axs[1].plot(x_template, template_trace, label='template') 

848 axs[1].set_xlabel('time') 

849 axs[1].set_ylabel('amplitude') 

850 axs[1].legend() 

851 fig.savefig('debug_plots_get_variable_window_size_correlation.png') 

852 

853 if return_time_difference: 

854 return correlation, time_diff 

855 else: 

856 return correlation
