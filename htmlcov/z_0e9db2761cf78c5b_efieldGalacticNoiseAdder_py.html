11class efieldGalacticNoiseAdder(channelGalacticNoiseAdder): 

12 r""" 

13 Class that simulates the noise produced by galactic radio emission 

14 

15 Uses the pydgsm package (https://github.com/telegraphic/pygdsm), which provides 

16 radio background data based on Oliveira-Costa et al. (2008) (https://arxiv.org/abs/0802.1525) 

17 and Zheng et al. (2016) (https://arxiv.org/abs/1605.04920) 

18 

19 The radio sky model is evaluated on a number of points above the horizon 

20 folded with the antenna response. Since evaluating every frequency individually 

21 would be too slow, the model is evaluated for a few frequencies and the log10 

22 of the brightness temperature is interpolated in between. 

23 

24 This module is largely equivalent to the channelGalacticNoiseAdder, but operates on ``ElectricField`` 

25 instead of ``Channel`` objects. 

26 

27 See Also 

28 -------- 

29 NuRadioReco.modules.channelGalacticNoiseAdder.channelGalacticNoiseAdder 

30 Similar class that directly adds the noise to ``Channel`` (voltage trace) objects 

31 instead of ``ElectricField``\s. 

32 """ 

33 

34 def __init__(self): 

35 super().__init__() 

36 

37 @register_run() 

38 def run( 

39 self, 

40 event, 

41 station, 

42 detector, 

43 passband=None, 

44 ): 

45 

46 """ 

47 Adds noise resulting from galactic radio emission to the field traces 

48 

49 Parameters 

50 ---------- 

51 event: Event object 

52 The event containing the station to whose channels noise shall be added 

53 station: Station object 

54 The station whose channels noise shall be added to 

55 detector: Detector object 

56 The detector description 

57 passband: list of float, optional 

58 Lower and upper bound of the frequency range in which noise shall be 

59 added. The default (no passband specified) is [10, 1000] MHz 

60 """ 

61 if self._channelGalacticNoiseAdder__noise_temperatures is None: # check if .begin has been called, give helpful error message if not 

62 msg = "efieldGalacticNoiseAdder was not initialized correctly. Maybe you forgot to call `.begin()`?" 

63 logger.error(msg) 

64 raise ValueError(msg) 

65 

66 # check that if all channels field.get_frequencies() are identical 

67 last_freqs = None 

68 for field in station.get_electric_fields(): 

69 if (last_freqs is not None) and ( 

70 not np.allclose(last_freqs, field.get_frequencies(), rtol=0, atol=0.1 * units.MHz)): 

71 logger.error("The frequencies of each field must be the same, but they are not!") 

72 return 

73 last_freqs = field.get_frequencies() 

74 

75 freqs = last_freqs 

76 if passband is None: 

77 passband = [10 * units.MHz, 1000 * units.MHz] 

78 

79 passband_filter = (freqs > passband[0]) & (freqs < passband[1]) 

80 

81 site_latitude, site_longitude = detector.get_site_coordinates(station.get_id()) 

82 station_time = station.get_station_time() 

83 

84 local_coordinates = get_local_coordinates((site_latitude, site_longitude), station_time, self._channelGalacticNoiseAdder__n_side) 

85 

86 n_ice = ice.get_refractive_index(-0.01, detector.get_site(station.get_id())) 

87 n_air = ice.get_refractive_index(depth=1, site=detector.get_site(station.get_id())) 

88 

89 field_spectra = {} 

90 for field in station.get_electric_fields(): 

91 field_spectra[field.get_unique_identifier()] = field.get_frequency_spectrum() 

92 

93 for i_pixel in range(healpy.pixelfunc.nside2npix(self._channelGalacticNoiseAdder__n_side)): 

94 azimuth = local_coordinates[i_pixel].az.rad 

95 zenith = np.pi / 2. - local_coordinates[i_pixel].alt.rad 

96 

97 if zenith > 90. * units.deg: 

98 continue 

99 

100 if n_ice != n_air: # consider signal reflection at ice surface 

101 t_theta = geometryUtilities.get_fresnel_t_p(zenith, n_ice, n_air) 

102 t_phi = geometryUtilities.get_fresnel_t_s(zenith, n_ice, n_air) 

103 fresnel_zenith = geometryUtilities.get_fresnel_angle(zenith, n_ice, n_air) 

104 else: # we are at an in-air site; no refraction 

105 t_theta = 1 

106 t_phi = 1 

107 fresnel_zenith = zenith 

108 

109 if fresnel_zenith is None: 

110 continue 

111 

112 temperature_interpolator = scipy.interpolate.interp1d( 

113 self._channelGalacticNoiseAdder__interpolation_frequencies, np.log10(self._channelGalacticNoiseAdder__noise_temperatures[:, i_pixel]), kind='quadratic') 

114 noise_temperature = np.power(10, temperature_interpolator(freqs[passband_filter])) 

115 

116 efield_amplitude = signal_processing.get_electric_field_from_temperature( 

117 freqs[passband_filter], noise_temperature, self.solid_angle) 

118 

119 # assign random phases to electric field 

120 noise_spectrum = np.zeros((3, freqs.shape[0]), dtype=complex) 

121 phases = self._channelGalacticNoiseAdder__random_generator.uniform(0, 2. * np.pi, len(efield_amplitude)) 

122 

123 noise_spectrum[1][passband_filter] = np.exp(1j * phases) * efield_amplitude 

124 noise_spectrum[2][passband_filter] = np.exp(1j * phases) * efield_amplitude 

125 

126 efield_noise_spec = np.zeros_like(noise_spectrum) 

127 

128 for field in station.get_electric_fields(): 

129 

130 field_pos = field.get_position() 

131 if field_pos[2] < 0: 

132 curr_t_theta = t_theta 

133 curr_t_phi = t_phi 

134 curr_fresnel_zenith = fresnel_zenith 

135 curr_n = n_ice 

136 else: # we are in air 

137 curr_t_theta = 1 

138 curr_t_phi = 1 

139 curr_fresnel_zenith = zenith 

140 curr_n = n_air 

141 

142 # calculate the phase offset in comparison to station center 

143 # consider additional distance in air & ice 

144 # assume for air & ice constant index of refraction 

145 dt = geometryUtilities.get_time_delay_from_direction( 

146 curr_fresnel_zenith, azimuth, field_pos, n=curr_n) 

147 if field_pos[2] < -5: 

148 logger.warning( 

149 "Galactic noise cannot be simulated accurately for deep in-ice channels. " 

150 "Coherence and arrival direction of noise are probably inaccurate.") 

151 

152 delta_phases = -2 * np.pi * freqs[passband_filter] * dt 

153 

154 # add random polarizations and phase to electric field 

155 polarizations = self._channelGalacticNoiseAdder__random_generator.uniform(0, 2. * np.pi, len(efield_amplitude)) 

156 

157 efield_noise_spec[1][passband_filter] = noise_spectrum[1][passband_filter] * np.exp( 

158 1j * delta_phases) * np.cos(polarizations) * curr_t_theta 

159 efield_noise_spec[2][passband_filter] = noise_spectrum[2][passband_filter] * np.exp( 

160 1j * delta_phases) * np.sin(polarizations) * curr_t_phi 

161 

162 # add noise spectrum from this to field freq spectrum (which is in on-sky CS -> no noise in R-direction) 

163 field_spectra[field.get_unique_identifier()] += efield_noise_spec 

164 

165 # Store updated fields spectra 

166 for field in station.get_electric_fields(): 

167 field.set_frequency_spectrum(field_spectra[field.get_unique_identifier()], "same")
