33def register_run(level=None): 

34 """ 

35 Decorator for run methods. This decorator registers the run methods. It allows to keep track of 

36 which module is executed in which order and with what parameters. Also the execution time of each 

37 module is tracked. 

38 """ 

39 

40 def run_decorator(run): 

41 

42 @wraps(run) 

43 def register_run_method(self, *args, **kwargs): 

44 

45 # the following if/else part finds out if this module operates on full events or on a specific station 

46 # In principle, different modules can be executed on different stations, so we keep it general and save the 

47 # modules station specific. 

48 # The logic is: If the first two arguments are event and station -> station module 

49 # if the first argument is an event and the second not a station -> event module 

50 # if the first argument is not an event -> reader module that creates events. In this case, the module 

51 # returns an event and we use this event to store the module information (but the module actually returns a 

52 # generator, so not sure how to access the event. 

53 evt = None 

54 station = None 

55 

56 signature = inspect.signature(run) 

57 parameters = signature.parameters 

58 # convert args to kwargs to facilitate easier bookkeeping 

59 keys = [key for key in parameters.keys() if key != 'self'] 

60 all_kwargs = {key: value for key, value in zip(keys, args)} 

61 all_kwargs.update(kwargs) # this silently overwrites positional args with kwargs, but this is probably okay as we still raise an error later 

62 

63 # include parameters with default values 

64 for key,value in parameters.items(): 

65 if key not in all_kwargs.keys(): 

66 if value.default is not inspect.Parameter.empty: 

67 all_kwargs[key] = value.default 

68 

69 store_kwargs = {} 

70 for idx, (key, value) in enumerate(all_kwargs.items()): 

71 # event should be the first argument 

72 if isinstance(value, NuRadioReco.framework.event.Event) and idx == 0: 

73 evt = value 

74 # station should be second argument 

75 elif isinstance(value, NuRadioReco.framework.base_station.BaseStation) and idx == 1: 

76 station = value 

77 elif isinstance(value, (detectors.detector_base.DetectorBase, detectors.rnog_detector.Detector)): 

78 pass # we don't try to store detectors 

79 else: # we try to store other arguments IF they are pickleable 

80 try: 

81 pickle.dumps(value, protocol=4) 

82 store_kwargs[key] = value 

83 except (TypeError, AttributeError): # object couldn't be pickled - we store the error instead 

84 store_kwargs[key] = TypeError(f"Argument of type {type(value)} could not be serialized") 

85 

86 if station is not None: 

87 module_level = "station" 

88 elif evt is not None: 

89 module_level = "event" 

90 else: 

91 module_level = "reader" 

92 

93 start = timer() 

94 

95 if module_level == "event": 

96 evt._register_module_event(self, self.__class__.__name__, store_kwargs) 

97 elif module_level == "station": 

98 evt._register_module_station(station.get_id(), self, self.__class__.__name__, store_kwargs) 

99 elif module_level == "reader": 

100 # not sure what to do... function returns generator, not sure how to access the event... 

101 pass 

102 

103 res = run(self, *args, **kwargs) 

104 

105 end = timer() 

106 

107 if self not in register_run_method.time: 

108 # keep track of timing of modules. We use the module instance as key to time different module instances separately. 

109 register_run_method.time[self] = 0 

110 register_run_method.time[self] += (end - start) 

111 

112 return res 

113 

114 register_run_method.time = {} 

115
