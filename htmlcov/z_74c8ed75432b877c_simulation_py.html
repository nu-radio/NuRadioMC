93def calculate_sim_efield( 

94 showers, station_id, channel_id, 

95 det, propagator, medium, config, 

96 min_efield_amplitude=None, 

97 distance_cut=None): 

98 """ 

99 Calculate the simulated electric field for a given shower and channel. 

100 

101 Parameters 

102 ---------- 

103 showers : list of showers 

104 A list of the showers that should be simulated (as defined in the input HDF5 file) 

105 station_id : int 

106 the station id, only needed to identify the correct channel in the detector description 

107 channel_id : int 

108 the channel id for which the electric field should be calculated as defined in the detector description 

109 det : Detector object 

110 the detector description defining all channels. 

111 propagator : propagator object 

112 the propagator that should be used to calculate the electric field from emitter to receiver, typically a ray tracer 

113 medium: medium object 

114 the medium in which the electric field is propagating, typically ice 

115 config : dict 

116 the NuRadioMC configuration dictionary (from the yaml file) 

117 min_efield_amplitude: float 

118 speedup cut: the minimum electric field amplitude, if all efields from all showers are below this threshold value 

119 the station will not be set as candidate station. 

120 

121 Returns 

122 ------- 

123 list of SimEfield objects 

124 A list of SimEfield objects, one for each shower and propagation solution 

125 

126 """ 

127 if distance_cut is not None: 

128 time_logger.start_time('distance cut') 

129 

130 vertex_positions = np.zeros((len(showers), 3)) 

131 shower_energies = np.zeros((len(showers))) 

132 for i, shower in enumerate(showers): 

133 vertex_positions[i] = shower.get_parameter(shp.vertex) 

134 shower_energies[i] = shower.get_parameter(shp.energy) 

135 

136 vertex_distances = np.linalg.norm(vertex_positions - vertex_positions[0], axis=1) 

137 time_logger.stop_time('distance cut') 

138 

139 logger.debug("Calculating electric field for station %d , channel %d from list of showers", station_id, channel_id) 

140 

141 sim_station = NuRadioReco.framework.sim_station.SimStation(station_id) 

142 sim_station.set_candidate(False) 

143 if min_efield_amplitude is None: 

144 sim_station.set_candidate(True) 

145 sim_station.set_is_neutrino() # naming not ideal, but this function defines in-ice emission (compared to in-air emission from air showers) 

146 

147 x2 = det.get_relative_position(station_id, channel_id) + det.get_absolute_position(station_id) 

148 dt = 1. / config['sampling_rate'] 

149 # rescale the number of samples to the internal (higher) sampling rate used in the simulation 

150 n_samples = det.get_number_of_samples(station_id, channel_id) / det.get_sampling_frequency(station_id, channel_id) / dt 

151 n_samples = int(np.ceil(n_samples / 2.) * 2) # round to nearest even integer 

152 

153 for iSh, shower in enumerate(showers): 

154 x1 = shower.get_parameter(shp.vertex) 

155 if distance_cut is not None: 

156 time_logger.start_time('distance cut') 

157 # Aggregate energy of showers which are close to the shower in question 

158 mask_shower_sum = np.abs(vertex_distances - vertex_distances[iSh]) < config['speedup']['distance_cut_sum_length'] 

159 shower_energy_sum = np.sum(shower_energies[mask_shower_sum]) 

160 

161 # Check with the aggregated energy whether the shower could be detected. 

162 # (This does not skip showers which are to close to a other shower!) 

163 if np.linalg.norm(x1 - x2) > distance_cut(shower_energy_sum): 

164 time_logger.stop_time('distance cut') 

165 continue 

166 

167 time_logger.stop_time('distance cut') 

168 

169 time_logger.start_time('ray tracing') 

170 logger.debug("Calculating electric field for shower %d and station %d, channel %d", shower.get_id(), station_id, channel_id) 

171 shower_direction = -1 * shower.get_axis() # We need the propagation direction here, so we multiply the shower axis with '-1' 

172 n_index = medium.get_index_of_refraction(x1) 

173 cherenkov_angle = np.arccos(1. / n_index) 

174 

175 if config['speedup']['redo_raytracing']: 

176 # force a recalculation even if solutions for the same geometry (e.g., from a previous 

177 # shower at the same position) are still stored in the propagator 

178 propagator.reset_solutions() 

179 

180 propagator.set_start_and_end_point(x1, x2) 

181 propagator.use_optional_function('set_shower_axis', shower_direction) 

182 propagator.find_solutions() 

183 time_logger.stop_time('ray tracing') 

184 

185 if not propagator.has_solution(): 

186 logger.debug("shower %d and station %d, channel %d from %s to %s does not have any ray tracing solution", 

187 shower.get_id(), station_id, channel_id, x1, x2) 

188 continue 

189 

190 n = propagator.get_number_of_solutions() 

191 logger.debug("found %d solutions for shower %d and station %d, channel %d from %s to %s", 

192 n, shower.get_id(), station_id, channel_id, x1, x2) 

193 

194 delta_Cs = np.zeros(n) 

195 viewing_angles = np.zeros(n) 

196 # loop through all ray tracing solution 

197 for iS in range(n): 

198 viewing_angles[iS] = hp.get_angle(shower_direction, propagator.get_launch_vector(iS)) 

199 delta_Cs[iS] = viewing_angles[iS] - cherenkov_angle 

200 

201 # discard event if delta_C (angle off cherenkov cone) is too large 

202 if min(np.abs(delta_Cs)) > config['speedup']['delta_C_cut']: 

203 logger.debug(f'delta_C too large, event unlikely to be observed, (min(Delta_C) = {min(np.abs(delta_Cs))/units.deg:.1f}deg), skipping event') 

204 continue 

205 

206 for iS in range(n): # loop through all ray tracing solution 

207 # skip individual channels where the viewing angle difference is too large 

208 # discard event if delta_C (angle off cherenkov cone) is too large 

209 if np.abs(delta_Cs[iS]) > config['speedup']['delta_C_cut']: 

210 logger.debug('delta_C too large, ray tracing solution unlikely to be observed, skipping ray tracing solution') 

211 continue 

212 

213 # TODO: Fill with previous values if RT was already performed 

214 time_logger.start_time('ray tracing (time)') 

215 wave_propagation_distance = propagator.get_path_length(iS) # calculate path length 

216 wave_propagation_time = propagator.get_travel_time(iS) # calculate travel time 

217 time_logger.stop_time('ray tracing (time)') 

218 if wave_propagation_distance is None or wave_propagation_time is None: 

219 logger.warning('travel distance or travel time could not be calculated, skipping ray tracing solution. ' 

220 f'Shower ID: {shower.get_id()} Station ID: {station_id} Channel ID: {channel_id}') 

221 continue 

222 

223 kwargs = {} 

224 # if the input file specifies a specific shower realization, or 

225 # if the shower was already simulated (e.g. for a different channel or ray tracing solution) 

226 # use that realization 

227 if config['signal']['model'] in ["ARZ2019", "ARZ2020"] and shower.has_parameter(shp.charge_excess_profile_id): 

228 kwargs['iN'] = shower[shp.charge_excess_profile_id] 

229 logger.debug(f"reusing shower {kwargs['iN']} ARZ shower library") 

230 elif config['signal']['model'] == "Alvarez2009" and shower.has_parameter(shp.k_L): 

231 kwargs['k_L'] = shower[shp.k_L] 

232 logger.debug(f"reusing k_L parameter of Alvarez2009 model of k_L = {kwargs['k_L']:.4g}") 

233 

234 time_logger.start_time('signal gen.') 

235 

236 spectrum, additional_output = askaryan.get_frequency_spectrum(shower[shp.energy], viewing_angles[iS], 

237 n_samples, dt, shower[shp.type], n_index, wave_propagation_distance, 

238 config['signal']['model'], seed=config['seed'], full_output=True, **kwargs) 

239 

240 # save shower realization to SimShower and hdf5 file 

241 if config['signal']['model'] in ["ARZ2019", "ARZ2020"]: 

242 if not shower.has_parameter(shp.charge_excess_profile_id): 

243 shower.set_parameter(shp.charge_excess_profile_id, additional_output['iN']) 

244 logger.debug(f"setting shower profile for ARZ shower library to i = {additional_output['iN']}") 

245 if not shower.has_parameter(shp.shower_maximum): 

246 shower_xmax = additional_output['profile_depth'][np.argmax(additional_output['profile_ce'])] 

247 shower.set_parameter(shp.shower_maximum, shower_xmax) 

248 

249 if config['signal']['model'] == "Alvarez2009": 

250 if not shower.has_parameter(shp.k_L): 

251 shower.set_parameter(shp.k_L, additional_output['k_L']) 

252 

253 polarization_direction_onsky = calculate_polarization_vector(shower_direction, propagator.get_launch_vector(iS), config) 

254 receive_vector = propagator.get_receive_vector(iS) 

255 eR, eTheta, ePhi = np.outer(polarization_direction_onsky, spectrum) 

256 time_logger.stop_time('signal gen.') 

257 

258 # this is common stuff which is the same between emitters and showers 

259 electric_field = NuRadioReco.framework.electric_field.ElectricField([channel_id], 

260 position=det.get_relative_position(station_id, channel_id), 

261 shower_id=shower.get_id(), ray_tracing_id=iS) 

262 electric_field.set_frequency_spectrum(np.array([eR, eTheta, ePhi]), 1. / dt) 

263 time_logger.start_time('prop. effects') 

264 electric_field = propagator.apply_propagation_effects(electric_field, iS) 

265 time_logger.stop_time('prop. effects') 

266 # Trace start time is equal to the interaction time relative to the first 

267 # interaction plus the wave travel time. 

268 if shower.has_parameter(shp.vertex_time): 

269 trace_start_time = shower[shp.vertex_time] + wave_propagation_time 

270 else: 

271 trace_start_time = wave_propagation_time 

272 

273 # We shift the trace start time so that the trace time matches the propagation time. 

274 # The centre of the trace corresponds to the instant when the signal from the shower 

275 # vertex arrives at the observer. The next line makes sure that the centre time 

276 # of the trace is equal to vertex_time + T (wave propagation time) 

277 trace_start_time -= 0.5 * electric_field.get_number_of_samples() / electric_field.get_sampling_rate() 

278 

279 zenith, azimuth = hp.cartesian_to_spherical(*receive_vector) 

280 electric_field.set_trace_start_time(trace_start_time) 

281 electric_field[efp.azimuth] = azimuth 

282 electric_field[efp.zenith] = zenith 

283 electric_field[efp.ray_path_type] = propagation.solution_types[propagator.get_solution_type(iS)] 

284 electric_field[efp.nu_vertex_distance] = wave_propagation_distance 

285 electric_field[efp.nu_vertex_propagation_time] = wave_propagation_time 

286 electric_field[efp.nu_viewing_angle] = viewing_angles[iS] 

287 #: electric field polarization in onsky-coordinates. 0 corresponds to polarization in e_theta, 90deg is polarization in e_phi 

288 electric_field[efp.polarization_angle] = np.arctan2(*polarization_direction_onsky[1:][::-1]) 

289 electric_field[efp.raytracing_solution] = propagator.get_raytracing_output(iS) 

290 electric_field[efp.launch_vector] = propagator.get_launch_vector(iS) 

291 

292 if min_efield_amplitude is not None: 

293 time_logger.start_time('candidate check') 

294 if np.max(np.abs(electric_field.get_trace())) > min_efield_amplitude: 

295 sim_station.set_candidate(True) 

296 time_logger.stop_time('candidate check') 

297 

298 sim_station.add_electric_field(electric_field) 

299 logger.debug( 

300 f"Added electric field to SimStation for shower {shower.get_id()} and station {station_id}, " 

301 f"channel {channel_id} with ray tracing solution {iS} and viewing angle {viewing_angles[iS] / units.deg:.1f}deg") 

302 

303 return sim_station

306def calculate_sim_efield_for_emitter( 

307 emitters, station_id, channel_id, 

308 det, propagator, medium, config, 

309 rnd, antenna_pattern_provider, 

310 min_efield_amplitude=None): 

311 """ 

312 Calculate the simulated electric field for a given shower and channel. 

313 

314 Parameters 

315 ---------- 

316 emitters : list of emitters 

317 A list of the emitters that should be simulated (as defined in the input HDF5 file) 

318 station_id : int 

319 the station id, only needed to identify the correct channel in the detector description 

320 channel_id : int 

321 the channel id for which the electric field should be calculated as defined in the detector description 

322 det : Detector object 

323 the detector description defining all channels. 

324 propagator : propagator object 

325 the propagator that should be used to calculate the electric field from emitter to receiver, typically a ray tracer 

326 medium: medium object 

327 the medium in which the electric field is propagating, typically ice 

328 config : dict 

329 the NuRadioMC configuration dictionary (from the yaml file) 

330 rnd: numpy random number generator 

331 the random number generator to be used for the simulation 

332 antenna_pattern_provider: antenna pattern provider object 

333 the antenna pattern provider object to be used for the simulation 

334 min_efield_amplitude: float 

335 speedup cut: the minimum electric field amplitude, if all efields from all showers are belwo this threshold value 

336 the station will not be set as candidate station. 

337 

338 Returns 

339 ------- 

340 list of SimEfield objects 

341 A list of SimEfield objects, one for each shower and propagation solution 

342 

343 """ 

344 logger.debug(f"Calculating electric field for station {station_id}, channel {channel_id} from list of emitters") 

345 

346 sim_station = NuRadioReco.framework.sim_station.SimStation(station_id) 

347 sim_station.set_is_neutrino() # naming not ideal, but this function defines in-ice emission (compared to in-air emission from air showers) 

348 sim_station.set_candidate(False) 

349 if min_efield_amplitude is None: 

350 sim_station.set_candidate(True) 

351 

352 x2 = det.get_relative_position(station_id, channel_id) + det.get_absolute_position(station_id) 

353 dt = 1. / config['sampling_rate'] 

354 # rescale the number of samples to the internal (higher) sampling rate used in the simulation 

355 n_samples = det.get_number_of_samples(station_id, channel_id) / det.get_sampling_frequency(station_id, channel_id) / dt 

356 n_samples = int(np.ceil(n_samples / 2.) * 2) # round to nearest even integer 

357 

358 for emitter in emitters: 

359 time_logger.start_time('ray tracing') 

360 x1 = emitter.get_parameter(ep.position) 

361 n_index = medium.get_index_of_refraction(x1) 

362 

363 if config['speedup']['redo_raytracing']: 

364 # force a recalculation even if solutions for the same geometry (e.g., from a previous 

365 # emitter at the same position) are still stored in the propagator 

366 propagator.reset_solutions() 

367 

368 propagator.set_start_and_end_point(x1, x2) 

369 propagator.find_solutions() 

370 time_logger.stop_time('ray tracing') 

371 

372 if not propagator.has_solution(): 

373 logger.debug(f"emitter {emitter.get_id()} and station {station_id}, " 

374 f"channel {channel_id} from {x1} to {x2} does not have any ray tracing solution") 

375 continue 

376 

377 n = propagator.get_number_of_solutions() 

378 for iS in range(n): # loop through all ray tracing solution 

379 time_logger.start_time('ray tracing (time)') 

380 # TODO: Fill with previous values if RT was already performed 

381 wave_propagation_distance = propagator.get_path_length(iS) # calculate path length 

382 wave_propagation_time = propagator.get_travel_time(iS) # calculate travel time 

383 time_logger.stop_time('ray tracing (time)') 

384 if wave_propagation_distance is None or wave_propagation_time is None: 

385 logger.warning('travel distance or travel time could not be calculated, skipping ray tracing solution. ' 

386 f'Emitter ID: {emitter.get_id()} Station ID: {station_id} Channel ID: {channel_id}') 

387 continue 

388 

389 # if the input file specifies a specific shower realization, or 

390 # if the shower was already simulated (e.g. for a different channel or ray tracing solution) 

391 # use that realization 

392 amplitude = emitter[ep.amplitude] 

393 emitter_model = emitter[ep.model] 

394 emitter_kwargs = {} 

395 emitter_kwargs["launch_vector"] = propagator.get_launch_vector(iS) 

396 for key in ep: 

397 if key.name not in ['amplitude', 'model', 'position']: 

398 if emitter.has_parameter(key): 

399 emitter_kwargs[key.name] = emitter[key] 

400 

401 time_logger.start_time('signal gen.') 

402 if emitter_model.startswith("efield_"): 

403 if emitter_model == "efield_idl1_spice": 

404 if emitter.has_parameter(ep.realization_id): 

405 emitter_kwargs['iN'] = emitter[ep.realization_id] 

406 else: 

407 emitter_kwargs['rnd'] = rnd 

408 

409 (eR, eTheta, ePhi), additional_output = emitter_signalgen.get_frequency_spectrum(amplitude, n_samples, dt, emitter_model, **emitter_kwargs, full_output=True) 

410 if emitter_model == "efield_idl1_spice": 

411 if not emitter.has_parameter(ep.realization_id): 

412 emitter.set_parameter(ep.realization_id, additional_output['iN']) 

413 logger.debug(f"setting emitter realization to i = {additional_output['iN']}") 

414 else: 

415 # the emitter fuction returns the voltage output of the pulser. We need to convole with the antenna response of the emitting antenna 

416 # to obtain the emitted electric field. 

417 # get emitting antenna properties 

418 antenna_model = emitter[ep.antenna_type] 

419 antenna_pattern = antenna_pattern_provider.load_antenna_pattern(antenna_model) 

420 ori = [emitter[ep.orientation_theta], emitter[ep.orientation_phi], 

421 emitter[ep.rotation_theta], emitter[ep.rotation_phi]] 

422 # source voltage given to the emitter 

423 voltage_spectrum_emitter = emitter_signalgen.get_frequency_spectrum(amplitude, n_samples, dt, 

424 emitter_model, **emitter_kwargs) 

425 # convolve voltage output with antenna response to obtain emitted electric field 

426 frequencies = np.fft.rfftfreq(n_samples, d=dt) 

427 zenith_emitter, azimuth_emitter = hp.cartesian_to_spherical(*propagator.get_launch_vector(iS)) 

428 VEL = antenna_pattern.get_antenna_response_vectorized(frequencies, zenith_emitter, azimuth_emitter, *ori) 

429 c = constants.c * units.m / units.s 

430 eTheta = VEL['theta'] * (-1j) * voltage_spectrum_emitter * frequencies * n_index / c 

431 ePhi = VEL['phi'] * (-1j) * voltage_spectrum_emitter * frequencies * n_index / c 

432 eR = np.zeros_like(eTheta) 

433 

434 # rescale amplitudes by 1/R, for emitters this is not part of the "SignalGen" class 

435 eTheta *= 1 / wave_propagation_distance 

436 ePhi *= 1 / wave_propagation_distance 

437 time_logger.stop_time('signal gen.') 

438 

439 # this is common stuff which is the same between emitters and showers. Make sure to do any changes to this code in both places 

440 electric_field = NuRadioReco.framework.electric_field.ElectricField([channel_id], 

441 position=det.get_relative_position(station_id, channel_id), 

442 shower_id=emitter.get_id(), ray_tracing_id=iS) 

443 electric_field.set_frequency_spectrum(np.array([eR, eTheta, ePhi]), 1. / dt) 

444 time_logger.start_time('prop. effects') 

445 electric_field = propagator.apply_propagation_effects(electric_field, iS) 

446 time_logger.stop_time('prop. effects') 

447 # Trace start time is equal to the emitter time in case one was defined 

448 # (relevant for multiple emitters per event group) 

449 if emitter.has_parameter(ep.time): 

450 trace_start_time = emitter[ep.time] + wave_propagation_time 

451 else: 

452 trace_start_time = wave_propagation_time 

453 

454 # We shift the trace start time so that the trace time matches the propagation time. 

455 # The centre of the trace corresponds to the instant when the signal from the shower 

456 # vertex arrives at the observer. The next line makes sure that the centre time 

457 # of the trace is equal to vertex_time + wave_propagation_time (wave propagation time) 

458 trace_start_time -= 0.5 * electric_field.get_number_of_samples() / electric_field.get_sampling_rate() 

459 

460 zenith, azimuth = hp.cartesian_to_spherical(*propagator.get_receive_vector(iS)) 

461 electric_field.set_trace_start_time(trace_start_time) 

462 electric_field[efp.azimuth] = azimuth 

463 electric_field[efp.zenith] = zenith 

464 electric_field[efp.ray_path_type] = propagation.solution_types[propagator.get_solution_type(iS)] 

465 electric_field[efp.nu_vertex_distance] = wave_propagation_distance 

466 electric_field[efp.nu_vertex_propagation_time] = wave_propagation_time 

467 electric_field[efp.raytracing_solution] = propagator.get_raytracing_output(iS) 

468 electric_field[efp.launch_vector] = propagator.get_launch_vector(iS) 

469 

470 if min_efield_amplitude is not None: 

471 time_logger.start_time('candidate check') 

472 if np.max(np.abs(electric_field.get_trace())) > min_efield_amplitude: 

473 sim_station.set_candidate(True) 

474 time_logger.stop_time('candidate check') 

475 

476 sim_station.add_electric_field(electric_field) 

477 

478 return sim_station

481def apply_det_response_sim( 

482 sim_station, det, config, 

483 detector_simulation_filter_amp=None, 

484 evt=None,

546def apply_det_response( 

547 evt, det, config, 

548 detector_simulation_filter_amp=None, 

549 add_noise=None, 

550 Vrms_per_channel=None, 

551 integrated_channel_response=None, 

552 noiseless_channels=None, 

553 detector_simulation_part2=None, 

554 channel_ids=None): 

555 """ 

556 Apply the detector response to the simulated electric field, i.e., the voltage traces 

557 seen by the readout system. This function combines all electric fields (from different showers and 

558 ray tracing solutions) of one detector channel/antenna. This includes the effect of the antenna response, the 

559 analog signal chain. The result is a list of Channel objects which are added to the 

560 Station object. 

561 

562 Parameters 

563 ---------- 

564 evt : NuRadioReco.framework.event.Event 

565 Event object containing all the showers/emitters and electric fields 

566 det : Detector object 

567 the detector description defining all channels. 

568 config : dict 

569 the NuRadioMC configuration dictionary (from the yaml file) 

570 detector_simulation_filter_amp: function (optional) 

571 a function that applies the filter and amplifier response to the electric field 

572 the arguments to the function are (event, station, detector) 

573 if not provided, the function `detector_simulation_part2` needs to be provided. 

574 add_noise : bool 

575 if True, noise is added to the channels 

576 Vrms_per_channel : dict 

577 the noise RMS per channel 

578 integrated_channel_response : dict 

579 the integrated channels response int(S21^2, dt) per channel. Corresponds to bandwidth for signal 

580 chains without amplification. This is used to normalize the noise level 

581 to the bandwidth it is generated for. 

582 noiseless_channels : dict (keys station_id) of list of ints 

583 the channels that should not have noise added 

584 detector_simulation_part2: function (optional) 

585 this function gives the user the full flexibility to implement all processing 

586 arguments to the function are (event, station, detector) 

587 channel_ids: list of ints 

588 the channel ids for which the detector response should be calculated. If None, all channels are used. 

589 

590 Returns nothing. The Channels are added to the Station object. 

591 """ 

592 time_logger.start_time('det. response') 

593 

594 if detector_simulation_filter_amp is None and detector_simulation_part2 is None: 

595 logger.error("No detector response function provided. Please provide either detector_simulation_filter_amp or detector_simulation_part2") 

596 raise ValueError("No detector response function provided. Please provide either detector_simulation_filter_amp or detector_simulation_part2") 

597 

598 station = evt.get_station() # will raise an error if there are more than one station, but this should never happen 

599 # convert efields to voltages at digitizer 

600 if detector_simulation_part2 is not None: 

601 detector_simulation_part2(evt, station, det, add_noise=add_noise) 

602 else: 

603 dt = 1. / (config['sampling_rate']) 

604 # start detector simulation 

605 

606 # convolve efield with antenna pattern and add cable delay (this is also done in the efieldToVoltageConverter 

607 # (unlike the efieldToVoltageConverterPEREFIELD)) 

608 efieldToVoltageConverter.run(evt, station, det, channel_ids=channel_ids) 

609 

610 if add_noise: 

611 max_freq = 0.5 / dt 

612 Vrms = {} 

613 for channel_id in det.get_channel_ids(station.get_id()): 

614 norm = integrated_channel_response[station.get_id()][channel_id] 

615 # normalize noise level to the bandwidth its generated for 

616 Vrms[channel_id] = Vrms_per_channel[station.get_id()][channel_id] / (norm / max_freq) ** 0.5 

617 

618 channelGenericNoiseAdder.run( 

619 evt, station, det, amplitude=Vrms, min_freq=0 * units.MHz, 

620 max_freq=max_freq, type='rayleigh', 

621 excluded_channels=noiseless_channels[station.get_id()]) 

622 

623 detector_simulation_filter_amp(evt, station, det) 

624 

625 time_logger.stop_time('det. response')

628def build_dummy_event(station_id, det, config): 

629 """ 

630 Builds a dummy event for simulation. 

631 

632 Parameters 

633 ---------- 

634 station_id : int 

635 The station ID. 

636 det : Detector object 

637 The detector description defining all channels. 

638 config : dict 

639 The NuRadioMC configuration dictionary (from the yaml file) 

640 

641 Returns 

642 ------- 

643 object: The built event object. 

644 """ 

645 

646 evt = NuRadioReco.framework.event.Event(0, 0) 

647 sim_station = NuRadioReco.framework.sim_station.SimStation(station_id) 

648 sim_station.set_is_neutrino() # naming not ideal, but this function defines in-ice emission (compared to in-air emission from air showers) 

649 

650 dt = 1. / config['sampling_rate'] 

651 # rescale the number of samples to the internal (higher) sampling rate used in the simulation 

652 channel_id = det.get_channel_ids(station_id)[0] 

653 n_samples = det.get_number_of_samples(station_id, channel_id) / det.get_sampling_frequency(station_id, channel_id) / dt 

654 n_samples = int(np.ceil(n_samples / 2.) * 2) # round to nearest even integer 

655 

656 for channel_id in det.get_channel_ids(station_id): 

657 electric_field = NuRadioReco.framework.electric_field.ElectricField([channel_id], 

658 det.get_relative_position(sim_station.get_id(), channel_id)) 

659 trace = np.zeros(n_samples) 

660 trace[n_samples // 2] = 100 * units.V # set a signal that should satisfy any trigger and speedup cuts 

661 trace[n_samples // 2 + 1] = -100 * units.V 

662 electric_field.set_trace(np.array([np.zeros(n_samples), trace, trace]), 1. / dt) 

663 electric_field.set_trace_start_time(0) 

664 electric_field[efp.azimuth] = 0 

665 electric_field[efp.zenith] = 100 * units.deg 

666 electric_field[efp.ray_path_type] = 0 

667 sim_station.add_electric_field(electric_field) 

668 

669 station = NuRadioReco.framework.station.Station(station_id) 

670 station.set_sim_station(sim_station) 

671 evt.set_station(station) 

672 return evt

675def build_NuRadioEvents_from_hdf5(fin, fin_attrs, idxs): 

676 """ 

677 Build NuRadioReco event structures from the input file 

678 

679 The function automatically determines if a particle or an emitter is simulated and builds the 

680 corresponding event structure. 

681 

682 Parameters 

683 ---------- 

684 fin : dict 

685 the input file data dictionary 

686 fin_attrs : dict 

687 the input file attributes dictionary 

688 idxs : list of ints 

689 the indices of the events that should be built 

690 

691 Returns 

692 ------- 

693 event_group : `NuRadioReco.framework.event.Event` 

694 an event group object containing the showers and particles or emitters 

695 the output should contain all relevant information from the hdf5 file (except the attributes) 

696 to perform a NuRadioMC simulation 

697 """ 

698 time_logger.start_time('event builder') 

699 

700 parent_id = idxs[0] 

701 event_group_id = fin['event_group_ids'][parent_id] 

702 event_group = NuRadioReco.framework.event.Event(event_group_id, parent_id) 

703 # add event generator info event 

704 for enum_entry in genattrs: 

705 if enum_entry.name in fin_attrs: 

706 event_group.set_parameter(enum_entry, fin_attrs[enum_entry.name]) 

707 

708 particle_mode = "simulation_mode" not in fin_attrs or fin_attrs['simulation_mode'] != "emitter" 

709 if particle_mode: # first case: simulation of a particle interaction which produces showers 

710 # there is only one primary particle per event group 

711 input_particle = NuRadioReco.framework.particle.Particle(event_group_id) 

712 input_particle[simp.flavor] = fin['flavors'][parent_id] 

713 input_particle[simp.energy] = fin['energies'][parent_id] 

714 input_particle[simp.interaction_type] = fin['interaction_type'][parent_id] 

715 input_particle[simp.inelasticity] = fin['inelasticity'][parent_id] 

716 input_particle[simp.vertex] = np.array([fin['xx'][parent_id], 

717 fin['yy'][parent_id], 

718 fin['zz'][parent_id]]) 

719 input_particle[simp.zenith] = fin['zeniths'][parent_id] 

720 input_particle[simp.azimuth] = fin['azimuths'][parent_id] 

721 input_particle[simp.inelasticity] = fin['inelasticity'][parent_id] 

722 input_particle[simp.n_interaction] = fin['n_interaction'][parent_id] 

723 input_particle[simp.shower_id] = fin['shower_ids'][parent_id] 

724 if fin['n_interaction'][parent_id] <= 1: 

725 # parents before the neutrino and outgoing daughters without shower are currently not 

726 # simulated. The parent_id is therefore at the moment only rudimentarily populated. 

727 input_particle[simp.parent_id] = None # primary does not have a parent 

728 input_particle[simp.vertex_time] = 0 

729 if 'vertex_times' in fin: 

730 input_particle[simp.vertex_time] = fin['vertex_times'][parent_id] 

731 else: 

732 logger.warning("The input file does not include vertex times, setting vertex time to zero. Vertices from the same event will not be time-ordered.") 

733 event_group.add_particle(input_particle) 

734 

735 # now loop over all showers and add them to the event group 

736 for idx in idxs: 

737 vertex_time = 0 

738 if 'vertex_times' in fin: 

739 vertex_time = fin['vertex_times'][idx] 

740 

741 # create NuRadioReco event structure 

742 sim_shower = NuRadioReco.framework.radio_shower.RadioShower(fin['shower_ids'][idx]) 

743 # save relevant neutrino properties 

744 sim_shower[shp.zenith] = fin['zeniths'][idx] 

745 sim_shower[shp.azimuth] = fin['azimuths'][idx] 

746 sim_shower[shp.energy] = fin['shower_energies'][idx] 

747 sim_shower[shp.flavor] = fin['flavors'][idx] 

748 sim_shower[shp.interaction_type] = fin['interaction_type'][idx] 

749 sim_shower[shp.n_interaction] = fin['n_interaction'][idx] 

750 sim_shower[shp.vertex] = np.array([fin['xx'][idx], fin['yy'][idx], fin['zz'][idx]]) 

751 sim_shower[shp.vertex_time] = vertex_time 

752 sim_shower[shp.type] = fin['shower_type'][idx] 

753 if 'shower_realization_ARZ' in fin: 

754 sim_shower[shp.charge_excess_profile_id] = fin['shower_realization_ARZ'][idx] 

755 if 'shower_realization_Alvarez2009' in fin: 

756 sim_shower[shp.k_L] = fin['shower_realization_Alvarez2009'][idx] 

757 

758 # TODO direct parent does not necessarily need to be the primary in general, but full 

759 # interaction chain is currently not populated in the input files. 

760 sim_shower[shp.parent_id] = event_group_id 

761 logger.debug(f"Adding shower {sim_shower.get_id()} to event group {event_group.get_id()}") 

762 event_group.add_sim_shower(sim_shower) 

763 

764 else: # emitter mode: simulation of one or several artificial emitters 

765 for idx in idxs: 

766 emitter_obj = NuRadioReco.framework.sim_emitter.SimEmitter(fin['shower_ids'][idx]) # shower_id is equivalent to emitter_id in this case 

767 emitter_obj[ep.position] = np.array([fin['xx'][idx], fin['yy'][idx], fin['zz'][idx]]) 

768 emitter_obj[ep.model] = fin['emitter_model'][idx] 

769 emitter_obj[ep.amplitude] = fin['emitter_amplitudes'][idx] 

770 for key in ep: 

771 if not emitter_obj.has_parameter(key): 

772 if 'emitter_' + key.name in fin: 

773 emitter_obj[key] = fin['emitter_' + key.name][idx] 

774 event_group.add_sim_emitter(emitter_obj) 

775 

776 time_logger.stop_time('event builder') 

777 

778 return event_group

814def calculate_polarization_vector(shower_axis, launch_vector, config): 

815 """ calculates the polarization vector in spherical coordinates (eR, eTheta, ePhi) 

816 

817 Parameters 

818 ---------- 

819 shower_axis: array-like 

820 shower axis in cartesian coordinates 

821 launch_vector: array-like 

822 launch vector in cartesian coordinates 

823 config: dict 

824 configuration dictionary 

825 

826 Returns 

827 ------- 

828 array-like 

829 polarization vector in spherical coordinates (eR, eTheta, ePhi) 

830 """ 

831 if config['signal']['polarization'] == 'auto': 

832 polarization_direction = np.cross(launch_vector, np.cross(shower_axis, launch_vector)) 

833 polarization_direction /= np.linalg.norm(polarization_direction) 

834 cs = cstrans.cstrafo(*hp.cartesian_to_spherical(*launch_vector)) 

835 return cs.transform_from_ground_to_onsky(polarization_direction) 

836 

837 if config['signal']['polarization'] == 'custom': 

838 ePhi = float(config['signal']['ePhi']) 

839 eTheta = (1 - ePhi ** 2) ** 0.5 

840 v = np.array([0, eTheta, ePhi]) 

841 return v / np.linalg.norm(v) 

842 

843 msg = f"{config['signal']['polarization']} for config.signal.polarization is not a valid option" 

844 logger.error(msg) 

845 raise ValueError(msg)

868def calculate_particle_weight(event_group, idx, cfg, fin=None): 

869 """ 

870 Calculate the (survival) propability of a neutrino reaching the simulation volume. 

871 

872 Depending on the config settings, the weights is used from the input file, or 

873 calulated based on the Earth attenuation and cross section model defined in the config 

874

876 ---------- 

877 event_group : EventGroup 

878 The event group containing the particle. 

879 idx : int 

880 The index of the particle in the event group. 

881 cfg : dict 

882 The configuration parameters. 

883 fin : dictionary 

884 the data part of the hdf5 input file 

885 

886 Returns 

887 ------- 

888 float 

889 The weight of the particle. 

890

917 phi_nu=primary[simp.azimuth]) 

918 # all entries for the event for this primary get the calculated primary's weight 

919 return primary[simp.weight]

922def group_into_events(station, event_group, particle_mode, split_event_time_diff, 

923 zerosignal=False): 

924 """ 

925 Group the signals from a station into multiple events based on signal arrival times. 

926 

927 Parameters 

928 ---------- 

929 station : NuRadioReco.framework.station.Station 

930 The station object containing the signals. 

931 event_group : NuRadioMC.framework.event.Event 

932 The event group object containing all showers or emitters and other meta attributes 

933 particle_mode : bool 

934 Flag indicating whether the events are a particle simulations or not (i.e. an emitter simulation) 

935 split_event_time_diff : float 

936 The time difference threshold for splitting events. 

937 zerosignal : bool, optional 

938 Flag indicating whether to zero out the signals. Default is False. 

939 

940 Returns 

941 ------- 

942 events : list of NuRadioReco.framework.event.Event 

943 The list of events created from the grouped signals. 

944 """ 

945 time_logger.start_time('group into events') 

946 

947 event_group_id = event_group.get_run_number() 

948 start_times = [] 

949 channel_identifiers = [] 

950 for channel in station.get_sim_station().iter_channels(): 

951 channel_identifiers.append(channel.get_unique_identifier()) 

952 start_times.append(channel.get_trace_start_time()) 

953 

954 start_times = np.array(start_times) 

955 start_times_sort = np.argsort(start_times) 

956 delta_start_times = np.diff(start_times[start_times_sort]) # this array is sorted in time 

957 split_event_time_diff = float(split_event_time_diff) 

958 

959 iSplit = np.atleast_1d(np.squeeze(np.argwhere(delta_start_times > split_event_time_diff))) 

960 n_sub_events = len(iSplit) + 1 

961 if n_sub_events > 1: 

962 logger.info(f"splitting event group id {event_group_id} into {n_sub_events} " 

963 f"sub events because time separation larger than {split_event_time_diff/units.ns}ns") 

964 

965 tmp_station = copy.deepcopy(station) 

966 events = [] 

967 for iEvent in range(n_sub_events): 

968 iStart = 0 

969 iStop = len(channel_identifiers) 

970 if n_sub_events > 1: 

971 if iEvent > 0: 

972 iStart = iSplit[iEvent - 1] + 1 

973 if iEvent < n_sub_events - 1: 

974 iStop = iSplit[iEvent] + 1 

975 indices = start_times_sort[iStart: iStop] 

976 if n_sub_events > 1: 

977 tmp = "" 

978 for start_time in start_times[indices]: 

979 tmp += f"{start_time/units.ns:.0f}, " 

980 tmp = tmp[:-2] + " ns" 

981 logger.info(f"creating event {iEvent} of event group {event_group_id} ranging " 

982 f"from {iStart} to {iStop} with indices {indices} corresponding to signal times of {tmp}") 

983 

984 evt = NuRadioReco.framework.event.Event(event_group_id, iEvent) # create new event 

985 if particle_mode: 

986 # add MC particles that belong to this (sub) event to event structure 

987 # add only primary for now, since full interaction chain is not typically in the input hdf5s 

988 evt.add_particle(event_group.get_primary()) # add primary particle to event 

989 

990 # copy over generator information from temporary event to event 

991 for enum_entry in genattrs: 

992 if event_group.has_parameter(enum_entry): 

993 evt.set_parameter(enum_entry, event_group.get_parameter(enum_entry)) 

994 

995 station = NuRadioReco.framework.station.Station(tmp_station.get_id()) 

996 sim_station = NuRadioReco.framework.sim_station.SimStation(tmp_station.get_id()) 

997 sim_station.set_is_neutrino() 

998 tmp_sim_station = tmp_station.get_sim_station() 

999 shower_ids_of_sub_event = [] 

1000 for iCh in indices: 

1001 ch_uid = channel_identifiers[iCh] 

1002 shower_id = ch_uid[1] 

1003 if shower_id not in shower_ids_of_sub_event: 

1004 shower_ids_of_sub_event.append(shower_id) 

1005 sim_station.add_channel(tmp_sim_station.get_channel(ch_uid)) 

1006 efield_uid = (tuple([ch_uid[0]]), ch_uid[1], ch_uid[2]) # the efield unique identifier has as first parameter an array of the channels it is valid for 

1007 for efield in tmp_sim_station.get_electric_fields(): 

1008 if efield.get_unique_identifier() == efield_uid: 

1009 sim_station.add_electric_field(efield) 

1010 logger.debug(f"adding sim efield {efield_uid} to sim station") 

1011 

1012 if particle_mode: 

1013 # add showers that contribute to this (sub) event to event structure 

1014 for shower_id in shower_ids_of_sub_event: 

1015 evt.add_sim_shower(event_group.get_sim_shower(shower_id)) 

1016 else: 

1017 for shower_id in shower_ids_of_sub_event: 

1018 evt.add_sim_emitter(event_group.get_sim_emitter(shower_id)) 

1019 

1020 station.set_sim_station(sim_station) 

1021 station.set_station_time(event_group.get_event_time()) 

1022 evt.set_station(station)

1741 def add_filtered_noise_to_channels(self, evt, station, channel_ids): 

1742 """ 

1743 Add noise to the traces of the channels in the event. 

1744 This function is used to add noise to the traces of the non-trigger channels. 

1745 The traces of the non-trigger channels already have the detector response applied to them. 

1746 Hence we add "filtered" noise, i.e., noise which is based through the same filter seperatly. 

1747 """ 

1748 time_logger.start_time('noise (non-trigger channels)') 

1749 station_id = station.get_id() 

1750 for channel_id in channel_ids: 

1751 channel = station.get_channel(channel_id) 

1752 

1753 if channel_id in self._noiseless_channels[station_id]: 

1754 continue 

1755 

1756 ff = channel.get_frequencies() 

1757 filt = np.ones_like(ff, dtype=complex) 

1758 for i, (name, instance, kwargs) in enumerate(evt.iter_modules(station_id)): 

1759 if hasattr(instance, "get_filter"): 

1760 filt *= instance.get_filter(ff, station_id, channel_id, self._det, **kwargs) 

1761 

1762 noise = channelGenericNoiseAdder.bandlimited_noise_from_spectrum( 

1763 channel.get_number_of_samples(), channel.get_sampling_rate(), 

1764 spectrum=filt, amplitude=self._Vrms_per_channel[station.get_id()][channel_id], 

1765 type='rayleigh', time_domain=False) 

1766 

1767 channel.set_frequency_spectrum(channel.get_frequency_spectrum() + noise, channel.get_sampling_rate()) 

1768 

1769 time_logger.stop_time('noise (non-trigger channels)')

1799 def _is_in_fiducial_volume(self, pos): 

1800 """ Checks if pos is in fiducial volume """ 

1801 

1802 for check_attr in ['fiducial_zmin', 'fiducial_zmax']: 

1803 if not check_attr in self._fin_attrs: 

1804 logger.warning("Fiducial volume not defined. Return True") 

1805 return True 

1806 

1807 pos = copy.deepcopy(pos) - np.array([self._fin_attrs.get("x0", 0), self._fin_attrs.get("y0", 0), 0]) 

1808 

1809 if not (self._fin_attrs["fiducial_zmin"] < pos[2] < self._fin_attrs["fiducial_zmax"]): 

1810 return False 

1811 

1812 if "fiducial_rmax" in self._fin_attrs: 

1813 radius = np.sqrt(pos[0] ** 2 + pos[1] ** 2) 

1814 return self._fin_attrs["fiducial_rmin"] < radius < self._fin_attrs["fiducial_rmax"] 

1815 elif "fiducial_xmax" in self._fin_attrs: 

1816 return (self._fin_attrs["fiducial_xmin"] < pos[0] < self._fin_attrs["fiducial_xmax"] and 

1817 self._fin_attrs["fiducial_ymin"] < pos[1] < self._fin_attrs["fiducial_ymax"]) 

1818 else: 

1819 raise ValueError("Could not contruct fiducial volume from input file.")
