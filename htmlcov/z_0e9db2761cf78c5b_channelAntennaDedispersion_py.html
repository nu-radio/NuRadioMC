10class channelAntennaDedispersion: 

11 """ 

12 The goal of this module is to unfold the group delay (i.e., dedispersing the channel signal) introduced by the 

13 antenna at a point where the direction of the incoming signal is not known. Hence, we choose the most sensitive 

14 direction (and polarisation) for each antenna as a proxy. The goal is to improve the signal-to-noise ratio on 

15 the channel level. 

16 

17 For a proper reconstruction of the electric field, please look at the voltage2EFieldConverter module. 

18 """ 

19 def __init__(self, additional_antennas=None): 

20 """ 

21 Parameters 

22 ---------- 

23 additional_antennas: dict, optional 

24 A dictionary with antenna names as keys and the most sensitive direction (zenith, azimuth) relative to the antenna orientation 

25 as values to add to the default list of antennas. 

26 """ 

27 

28 self._provider = AntennaPatternProvider() 

29 

30 # relative to the antenna orientation 

31 self.antennas_most_sensitive_directions = { 

32 "LPDA": [0, 0], # the sensitive direction of an LPDA is the boresight direction 

33 "bicone": [90 * units.deg, 0], # the sensitive direction of a dipole is perpendicular to its orientation 

34 "vpol": [90 * units.deg, 0], # the sensitive direction of a dipole is perpendicular to its orientation 

35 "hpol": [0, 0], # the sensitive direction of a dipole is along it's direction 

36 } 

37 

38 if additional_antennas is not None: 

39 if not isinstance(additional_antennas, dict): 

40 raise ValueError("To add antennas provide a dictionary with antenna names as keys and direction arrays as values") 

41 for ant_name, direction in additional_antennas.items(): 

42 if not isinstance(direction, (list, np.ndarray)) or len(direction) != 2: 

43 raise ValueError(f"Antenna direction for {ant_name} requires zenith and azimuth") 

44 self.antennas_most_sensitive_directions[ant_name] = direction 

45 

46 

47 def _get_response(self, det, station_id, channel_id, ff): 

48 antenna_name = det.get_antenna_model(station_id, channel_id) 

49 antenna = self._provider.load_antenna_pattern(antenna_name) 

50 zen_ori, az_ori, zen_rot, az_rot = det.get_antenna_orientation(station_id, channel_id) 

51 

52 match = next( 

53 (key_name for key_name in self.antennas_most_sensitive_directions if key_name.lower() in antenna_name.lower()), None 

54 ) 

55 if match is None: 

56 raise AttributeError(f"Antenna name {antenna_name} can't be interpreted. " 

57 f"Available names are: {list(self.antennas_most_sensitive_directions.keys())}") 

58 

59 zen = zen_ori + self.antennas_most_sensitive_directions[match][0] 

60 az = az_ori + self.antennas_most_sensitive_directions[match][1] 

61 

62 VEL = antenna.get_antenna_response_vectorized(ff, zen, az, zen_ori, az_ori, zen_rot, az_rot) 

63 polarization = "phi" 

64 if np.sum(np.abs(VEL['theta'])) > np.sum(np.abs(VEL['phi'])): 

65 polarization = "theta" 

66 

67 response = np.exp(1j * np.angle(VEL[polarization])) 

68 return response

74 Unfold the group delay introduced by the antenna for each channel without proper 

75 knowledge of the incoming signal direction. 

76 

77 Parameters 

78 ---------- 

79 evt: Event 

80 The event containing the station and channel information.

84 The detector containing the antenna information. 

85 debug: bool, optional 

86 If True, plots the trace before and after dedispersion for debugging purposes. Default is False. 

87 """
