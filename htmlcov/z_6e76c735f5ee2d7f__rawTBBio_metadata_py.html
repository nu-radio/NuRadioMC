1""" 

2Module to read in calibration metadata from LOFAR TBB files. 

3 

4Adapted from https://github.com/Bhare8972/LOFAR-LIM/blob/master/LoLIM/IO/metadata.py; 

5original description included below. 

6Most users will not want to use this module directly, but instead use the 

7`readLOFARData <NuRadioReco.modules.io.LOFAR.readLOFARData>` module, which converts the 

8TBB data to the NuRadio ``Event`` format, taking care to use the correct units etc. 

9Note that to minimize changes, **this** module adheres to the LOFAR internal units (seconds / Hz) 

10rather than using the NuRadio unit system. 

11 

12Original description 

13-------------------- 

14This module reads in calibration metadata from file in the early phases of LOFAR. In the future this should be replaced 

15by reading the metadata from the files. 

16 

17.. moduleauthor:: Sander ter Veen <s.terveen@astro.ru.nl> 

18 

19Modified by Brian Hare for use with LOFAR for Lightning Imaging.

50def mapAntennasetKeyword(antenna_set): 

51 """ 

52 Ugly fix to map correct antenna names in input to wrong antenna names 

53 for metadata module. 

54 """ 

55 

56 # Strip whitespace 

57 antenna_set = antenna_set.strip() 

58 

59 allowed = ["LBA_OUTER", "LBA_INNER", 

60 "LBA_SPARSE_EVEN", "LBA_SPARSE_ODD", 

61 "LBA_X", "LBA_Y", 

62 "HBA", "HBA_0", "HBA_1"] 

63 

64 incorrect = { 

65 "LBA_INNER": "LBA_INNER", 

66 "LBA_OUTER": "LBA_OUTER", 

67 "LBA_SPARSE0": "LBA_SPARSE_EVEN", 

68 "LBA_SPARSE1": "LBA_SPARSE_ODD", 

69 "HBA_ZERO": "HBA_0", 

70 "HBA_ONE": "HBA_1", 

71 "HBA_DUAL": "HBA", 

72 "HBA_JOINED": "HBA", 

73 "HBA_ZERO_INNER": "HBA_0", # Only true for core stations 

74 "HBA_ONE_INNER": "HBA_1", # Only true for core stations 

75 "HBA_DUAL_INNER": "HBA", # Only true for core stations 

76 "HBA_JOINED_INNER": "HBA", 

77 } # Only true for core stations 

78 

79 if antenna_set in incorrect: 

80 antenna_set = incorrect[antenna_set] 

81 elif antenna_set == "HBA_BOTH": 

82 # This keyword is also wrong but present in file headers 

83 print("Keyword " + antenna_set + " does not comply with ICD, mapping...") 

84 antenna_set = "HBA" 

85 

86 assert antenna_set in allowed, f"Antenna set {antenna_set} is not allowed!" 

87 

88 return antenna_set

91def getItrfAntennaPosition(station, antenna_set, metadata_dir): 

92 """ 

93 Returns the antenna positions of all the antennas in the station 

94 in ITRF coordinates for the specified antennaset. 

95 station can be the name or id of the station. 

96 

97 Parameters 

98 ---------- 

99 station : int or str 

100 Name or id of the station. e.g. "CS302" or 142 

101 antenna_set : {LBA_INNER, LBA_OUTER, LBA_X, LBA_Y, LBA_SPARSE0, LBA_SPARSE1, HBA_0, HBA_1, HBA} 

102 Antenna set used for this station 

103 metadata_dir: str 

104 Path to the directory containing the LOFAR static metadata. 

105 

106 """ 

107 # Check station id type 

108 if isinstance(station, int): 

109 # Convert a station id to a station name 

110 station = SId_to_Sname[station] 

111 

112 antenna_set = mapAntennasetKeyword(antenna_set) 

113 

114 if "LBA" in antenna_set: 

115 antennatype = "LBA" 

116 elif "HBA" in antenna_set: 

117 antennatype = "HBA" 

118 

119 # Obtain filename of antenna positions 

120 filename = ( 

121 metadata_dir 

122 + "/lofar/StaticMetaData/AntennaFields/" 

123 + station 

124 + "-AntennaField.conf" 

125 ) 

126 

127 # Open file 

128 f = open(filename, "r") 

129 

130 if station[0:2] != "CS": 

131 if "HBA" in antenna_set: 

132 antenna_set = "HBA" 

133 

134 # Find position of antennaset in file 

135 str_line = "" 

136 while antennatype != str_line.strip(): 

137 str_line = f.readline() 

138 if len(str_line) == 0: 

139 # end of file reached, no data available 

140 assert False 

141 

142 # Find the location of the station. Antenna locations are relative to this 

143 str_line = f.readline() 

144 str_split = str_line.split() 

145 stationX = float(str_split[2]) 

146 stationY = float(str_split[3]) 

147 stationZ = float(str_split[4]) 

148 

149 str_line = f.readline() 

150 

151 # Get number of antennas and the number of directions 

152 nrantennas = int(str_line.split()[0]) 

153 nrdir = int(str_line.split()[4]) 

154 

155 antenna_positions = np.empty((2 * nrantennas, nrdir), dtype=np.double) 

156 for i in range(nrantennas): 

157 line = f.readline().split() 

158 

159 antenna_positions[2 * i, 0] = float(line[0]) + stationX 

160 antenna_positions[2 * i, 1] = float(line[1]) + stationY 

161 antenna_positions[2 * i, 2] = float(line[2]) + stationZ 

162 

163 antenna_positions[2 * i + 1, 0] = float(line[3]) + stationX 

164 antenna_positions[2 * i + 1, 1] = float(line[4]) + stationY 

165 antenna_positions[2 * i + 1, 2] = float(line[5]) + stationZ 

166 

167 if antennatype == "LBA": 

168 # There are three types of feed 

169 # H for HBA 

170 # h for lbh 

171 # l for lbl 

172 feed = {"CS": {}, "RS": {}, "DE": {}} 

173 feed["CS"]["LBA_SPARSE_EVEN"] = "24hhll" 

174 feed["CS"]["LBA_SPARSE_ODD"] = "24llhh" 

175 feed["CS"]["LBA_X"] = "48hl" 

176 feed["CS"]["LBA_Y"] = "48lh" 

177 feed["CS"]["LBA_INNER"] = "96h" 

178 feed["CS"]["LBA_OUTER"] = "96l" 

179 feed["RS"]["LBA_SPARSE_EVEN"] = "24hhll" 

180 feed["RS"]["LBA_SPARSE_ODD"] = "24llhh" 

181 feed["RS"]["LBA_X"] = "48hl" 

182 feed["RS"]["LBA_Y"] = "48lh" 

183 feed["RS"]["LBA_INNER"] = "96h" 

184 feed["RS"]["LBA_OUTER"] = "96l" 

185 feed["DE"]["LBA"] = "192h" 

186 if station[0:2] == "CS" or "RS": 

187 feedsel = feed[station[0:2]][antenna_set] 

188 nrset = int(feedsel.split("l")[0].split("h")[0].split("H")[0]) 

189 feeds = "" 

190 feedsel = feedsel[len(str(nrset)) :] 

191 for i in range(nrset): 

192 feeds += feedsel 

193 

194 indexselection = [] 

195 for i in range(len(feeds)): 

196 if feeds[i] == "l": 

197 # The 'l' feeds are the last 96 numbers of the total list 

198 indexselection.append(i + 96) 

199 elif feeds[i] == "h": 

200 # The 'h' feeds are the first 96 numbers of the total list 

201 indexselection.append(i) 

202 else: 

203 # This selection is not yet supported 

204 assert False 

205 antenna_positions = antenna_positions[indexselection] 

206 

207 return antenna_positions

210def getStationPositions(station, antenna_set, coordinate_system, metadata_dir): 

211 """ 

212 Returns the antenna positions of all the antennas in the station 

213 relative to the station center for the specified antenna set. 

214 station can be the name or id of the station. 

215 

216 Parameters 

217 ---------- 

218 

219 station : int or str 

220 Name or id of the station. e.g. "CS302" or 142 

221 antenna_set : {LBA_INNER, LBA_OUTER, LBA_X, LBA_Y, LBA_SPARSE0, LBA_SPARSE1, HBA_0, HBA_1, HBA} 

222 Antenna set used for this station 

223 coordinate_system : {WGS84, ITRF} 

224 The coordinate system to use when returning antenna positions (see also Notes section).\ 

225 metadata_dir : str 

226 Path to the directory containing the LOFAR static metadata. 

227 

228 Notes 

229 ----- 

230 For the coordinate system, when using "WGS84", the function returns the postions as a Numpy array containing 

231 [lat, lon, alt] of antenna positions. Else, when using the ITRF coordinate system, is return the positions as 

232 [X, Y, Z]. 

233 

234 """ 

235 

236 # Check if requested antennaset is known 

237 assert coordinate_system in ["WGS84", "ITRF"] 

238 

239 # Check station id type 

240 if isinstance(station, int): 

241 # Convert a station id to a station name 

242 station = SId_to_Sname[station] 

243 

244 antenna_set = mapAntennasetKeyword(antenna_set) 

245 

246 # Obtain filename of antenna positions 

247 if "WGS84" in coordinate_system: 

248 filename = ( 

249 metadata_dir 

250 + "/lofar/StaticMetaData/AntennaArrays/" 

251 + station 

252 + "-AntennaArrays.conf" 

253 ) 

254 else: 

255 filename = ( 

256 metadata_dir 

257 + "/lofar/StaticMetaData/AntennaFields/" 

258 + station 

259 + "-AntennaField.conf" 

260 ) 

261 

262 # Open file 

263 f = open(filename, "r") 

264 

265 if "LBA" in antenna_set: 

266 antenna_set = "LBA" 

267 

268 if station[0:2] != "CS": 

269 if "HBA" in antenna_set: 

270 antenna_set = "HBA" 

271 

272 # Find position of antennaset in file 

273 str_line = "" 

274 while antenna_set != str_line.strip(): 

275 str_line = f.readline() 

276 if len(str_line) == 0: 

277 # end of file reached, no data available 

278 print("Antenna set not found in calibration file", filename) 

279 return None 

280 

281 # Skip name and station reference position 

282 str_line = f.readline().split() 

283 

284 A = float(str_line[2]) # lon in WGS84, X in ITRF 

285 B = float(str_line[3]) # lat in WGS84, Y in ITRF 

286 C = float(str_line[4]) # alt in WGS84, Z in ITRF 

287 

288 return np.array([A, B, C])

291def convertITRFToLocal(itrfpos, metadata_dir, phase_center=None, ref_lat_lon=None, out=None): 

292 """ 

293 

294 Parameters 

295 ---------- 

296 itrfpos : list or np.ndarray 

297 The ITRF positions as 1D numpy array, or list of positions as a 2D array 

298 metadata_dir: str 

299 Path to the directory containing the LOFAR static metadata. 

300 phase_center: list or np.ndarray, default=None 

301 The origin of the coordinate system, in ITRF. Default is the coordinates of station CS002. 

302 ref_lat_lon: list or np.ndarray, default=None 

303 The rotation of the coordinate system, i.e. the [lat, lon] (in degrees) on the Earth which defines "UP". 

304 If not specified, the coordinates of CS002 will be used. 

305 out: np.ndarray, default=None 

306 If given, the output will be stored in this array. Otherwise, a new array will be created and returned. 

307 Cannot be same array as itrfpos 

308 

309 Notes 

310 ----- 

311 Function returns a 2D numpy array (even if input is 1D). 

312 

313 """ 

314 if ref_lat_lon is None: 

315 ref_lat_lon = [52.91512249, 6.869837540] 

316 

317 if phase_center is None: 

318 phase_center = getStationPositions("CS002", "LBA_OUTER", coordinate_system="ITRF", metadata_dir=metadata_dir) 

319 # ($LOFARSOFT/data/lofar/StaticMetaData/AntennaFields/CS002-AntennaField.conf) 

320 if out is itrfpos: 

321 logger.error( 

322 "The `out` array cannot be same as `itrfpos` in convertITRFToLocal." 

323 ) 

324 raise ValueError 

325 

326 lat = np.deg2rad(ref_lat_lon[0]) 

327 lon = np.deg2rad(ref_lat_lon[1]) 

328 arg0 = np.array( 

329 [-np.sin(lon), -np.sin(lat) * np.cos(lon), np.cos(lat) * np.cos(lon)] 

330 ) 

331 arg1 = np.array( 

332 [np.cos(lon), -np.sin(lat) * np.sin(lon), np.cos(lat) * np.sin(lon)] 

333 ) 

334 arg2 = np.array([0.0, np.cos(lat), np.sin(lat)]) 

335 

336 if out is None: 

337 ret = np.empty(itrfpos.shape, dtype=np.double) 

338 else: 

339 ret = out 

340 

341 ret[:] = np.outer(itrfpos[..., 0] - phase_center[0], arg0) 

342 ret += np.outer(itrfpos[..., 1] - phase_center[1], arg1) 

343 ret += np.outer(itrfpos[..., 2] - phase_center[2], arg2) 

344 

345 return ret

348def getStationPhaseCalibration( 

349 station, antenna_set, metadata_dir, file_location=None 

350): 

351 """ 

352 Read phase calibration data for a station. 

353 

354 Parameters 

355 ---------- 

356 

357 station : int or str 

358 Name or id of the station. e.g. "CS302" or 142 

359 antenna_set : {LBA_INNER, LBA_OUTER, LBA_X, LBA_Y, LBA_SPARSE0, LBA_SPARSE1, HBA_0, HBA_1, HBA} 

360 Antenna set used for this station 

361 metadata_dir : str 

362 Path to the directory containing the LOFAR static metadata. 

363 file_location : str, default=None 

364 The path to the LOFAR calibration tables. If None, it is assumed to be in the /lofar/StaticMetaData/CalTables 

365 directory relative to `xMetaData_directory`. 

366 

367 Returns 

368 ------- 

369 

370 The weights for 512 subbands. 

371 

372 Examples 

373 -------- 

374 

375 >>> getStationPhaseCalibration("CS002","LBA_OUTER",) 

376 array([[ 1.14260161 -6.07397622e-18j, 1.14260161 -6.05283530e-18j, 

377 1.14260161 -6.03169438e-18j, ..., 1.14260161 +4.68675289e-18j, 

378 1.14260161 +4.70789381e-18j, 1.14260161 +4.72903474e-18j], 

379 [ 0.95669876 +2.41800591e-18j, 0.95669876 +2.41278190e-18j, 

380 0.95669876 +2.40755789e-18j, ..., 0.95669876 -2.41017232e-19j, 

381 0.95669876 -2.46241246e-19j, 0.95669876 -2.51465260e-19j], 

382 [ 0.98463207 +6.80081617e-03j, 0.98463138 +6.89975906e-03j, 

383 0.98463069 +6.99870187e-03j, ..., 0.98299670 +5.71319125e-02j, 

384 0.98299096 +5.72306908e-02j, 0.98298520 +5.73294686e-02j], 

385 ..., 

386 [ 1.03201290 +7.39535744e-02j, 1.03144532 +8.14880844e-02j, 

387 1.03082273 +8.90182487e-02j, ..., -0.82551740 -6.23731331e-01j, 

388 -0.82094046 -6.29743206e-01j, -0.81631975 -6.35721497e-01j], 

389 [ 1.12370332 -1.15296909e-01j, 1.12428451 -1.09484545e-01j, 

390 1.12483564 -1.03669252e-01j, ..., -0.92476286 +6.48703460e-01j, 

391 -0.92810503 +6.43912711e-01j, -0.93142239 +6.39104744e-01j], 

392 [ 1.10043006 -6.18995646e-02j, 1.10075250 -5.58731668e-02j, 

393 1.10104193 -4.98450938e-02j, ..., -1.01051042 +4.40052904e-01j, 

394 -1.01290481 +4.34513198e-01j, -1.01526883 +4.28960464e-01j]]) 

395 

396 >>> getStationPhaseCalibration(122,"LBA_OUTER",) 

397 Calibration data not yet available. Returning 1 

398 array([[ 1.+0.j, 1.+0.j, 1.+0.j, ..., 1.+0.j, 1.+0.j, 1.+0.j], 

399 [ 1.+0.j, 1.+0.j, 1.+0.j, ..., 1.+0.j, 1.+0.j, 1.+0.j], 

400 [ 1.+0.j, 1.+0.j, 1.+0.j, ..., 1.+0.j, 1.+0.j, 1.+0.j], 

401 ..., 

402 [ 1.+0.j, 1.+0.j, 1.+0.j, ..., 1.+0.j, 1.+0.j, 1.+0.j], 

403 [ 1.+0.j, 1.+0.j, 1.+0.j, ..., 1.+0.j, 1.+0.j, 1.+0.j], 

404 [ 1.+0.j, 1.+0.j, 1.+0.j, ..., 1.+0.j, 1.+0.j, 1.+0.j]]) 

405 

406 """ 

407 

408 # Return mode nr depending on observation mode 

409 antennasetToMode = { 

410 "LBA_OUTER": "LBA_OUTER-10_90", 

411 "LBA_INNER": "LBA_INNER-10_90", 

412 "HBA": "HBA-110_190", 

413 "HBA_0": "HBA-110_190", 

414 "HBA_1": "HBA-110_190", 

415 } 

416 

417 antenna_set = mapAntennasetKeyword(antenna_set) 

418 

419 if antenna_set not in antennasetToMode.keys(): 

420 raise KeyError("Not a valid antennaset " + antenna_set) 

421 

422 mode_name = antennasetToMode[antenna_set] 

423 if not isinstance(station, str): 

424 # Convert a station id to a station name 

425 station = SId_to_Sname[station] 

426 

427 stationNr = station[2:] 

428 

429 # filename 

430 if file_location is None: 

431 file_location = metadata_dir + "/lofar/StaticMetaData/CalTables" 

432 

433 filename = file_location + "/CalTable-" + stationNr + "-" + mode_name + ".dat" 

434 with open(filename, "rb") as fin: 

435 # Test for header record above raw data - present in newer caltables (starting 2012) 

436 line = fin.readline().decode() 

437 if "HeaderStart" in line: 

438 while not "HeaderStop" in line: 

439 line = fin.readline().decode() 

440 else: # no header present, seek to starting position 

441 fin.seek(0) 

442 

443 data = np.fromfile(fin, dtype=np.double) 

444 

445 data.resize(512, 96, 2) 

446 

447 complexdata = np.empty(shape=(512, 96), dtype=complex) 

448 complexdata.real = data[:, :, 0] 

449 complexdata.imag = data[:, :, 1] 

450 

451 return complexdata.transpose()

454def convertPhase_to_Timing(phase_calibration, sample_time=5.0e-9): 

455 """ 

456 Given the phase calibration of the 512 LOFAR subbands, 

457 such as the output of getStationPhaseCalibration, 

458 return the timing callibration of each antenna. 

459 Not sure how well this works with HBA antennas. 

460 Sample time should be seconds per sample. Default is 5 ns 

461 """ 

462 phases = np.angle(phase_calibration) 

463 delays = (phases[:, 1] - phases[:, 0]) * (1024 / (2 * np.pi)) * sample_time 

464 return delays

467# Functions for previously known clock offsets. Only used for compatibility with past data! 

468def getClockCorrectionFromParsetAddition(metadata_dir): 

469 parsetFilename = ( 

470 metadata_dir + "/lofar/station_clock_offsets/StationCalibration.parset" 

471 ) 

472 

473 offsetDictX = {} 

474 offsetDictY = {} 

475 

476 infile = open(parsetFilename, "r") 

477 for line in infile: 

478 s = line.split("=") 

479 value = s[1] 

480 params = s[0].split(".") 

481 thisStation = params[2][0:5] 

482 thisAntennaSet = params[3] 

483 thisFilter = params[4] 

484 thisValueType = params[5] 

485 thisPolarization = params[6][0] 

486 

487 if ( 

488 thisAntennaSet == "LBA_OUTER" 

489 and thisFilter == "LBA_30_90" 

490 and thisValueType == "delay" 

491 ): 

492 if thisPolarization == "X": 

493 offsetDictX[thisStation] = float(value) 

494 elif thisPolarization == "Y": 

495 offsetDictY[thisStation] = float(value) 

496 else: 

497 raise ValueError("Wrong!") 

498 infile.close() 

499 

500 offsetDictCombined = {} 

501 

502 for key in offsetDictX.keys(): 

503 combined = 0.5 * (offsetDictX[key] + offsetDictY[key]) 

504 offsetDictCombined[key] = combined 

505 

506 return offsetDictCombined
