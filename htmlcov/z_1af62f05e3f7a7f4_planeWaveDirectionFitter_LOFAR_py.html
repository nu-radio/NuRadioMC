20def average_direction(event, detector, mode='normal'): 

21 """ 

22 Calculate the average direction for an event based on the plane wave directions of the individual stations. 

23 

24 Parameters 

25 ---------- 

26 event : Event object 

27 The event for which to calculate the average direction. 

28 detector : Detector object 

29 The detector for which to calculate the average direction. 

30 mode : str, default='normal' 

31 The mode to use for the calculation. Can be 'normal' (just raw mean) or 'weighted' 

32 (with number of good antennas as weight per station). 

33 

34 Returns 

35 ------- 

36 avg_zenith : float 

37 The average zenith angle for the event. 

38 avg_azimuth : float 

39 The average azimuth angle for the event. 

40 """ 

41 zeniths = [] 

42 azimuths = [] 

43 num_good_antennas = [] 

44 for station in event.get_stations(): 

45 if station.get_parameter(stationParameters.triggered): 

46 flagged_channels = station.get_parameter(stationParameters.flagged_channels) 

47 num_good_antennas.append( 

48 detector.get_number_of_channels(station.get_id()) - len(flagged_channels) 

49 ) 

50 zeniths.append(station.get_parameter(stationParameters.cr_zenith)) 

51 azimuths.append(station.get_parameter(stationParameters.cr_azimuth)) 

52 

53 zeniths = np.array(zeniths) 

54 azimuths = np.array(azimuths) 

55 num_good_antennas = np.array(num_good_antennas) 

56 

57 # Calculate the average direction:  

58 if mode == 'normal': 

59 avg_zenith = np.mean(zeniths) 

60 avg_azimuth = np.mean(azimuths) 

61 elif mode == 'weighted': 

62 avg_zenith = np.sum(zeniths * num_good_antennas) / np.sum(num_good_antennas) 

63 avg_azimuth = np.sum(azimuths * num_good_antennas) / np.sum(num_good_antennas) 

64 else: 

65 raise ValueError(f"Unknown mode: {mode}") 

66 

67 return avg_zenith, avg_azimuth

127 def _get_timelags(station, channel_ids_dominant_pol): 

128 """ 

129 Get timing differences between signals in antennas with respect to some reference antenna (the first one 

130 in the list of ids). The peak is determined using the Hilbert envelope after resampling the trace with 

131 `resample_factor`. 

132 

133 Parameters 

134 ---------- 

135 station : Station object 

136 The station for which to get the time lags 

137 channel_ids_dominant_pol : list of int 

138 The list of channel ids to return the time lags for (usually the dominant polarisation) 

139 

140 Returns 

141 ------- 

142 timelags : np.ndarray 

143 The timelags (in internal units) for each channel in the list, with respect to the first one 

144 """ 

145 # Get the signal time found by stationPulseFinder 

146 timelags = [] 

147 for channel_id in channel_ids_dominant_pol: 

148 timelags.append(station.get_channel(channel_id).get_parameter(channelParameters.signal_time)) 

149 

150 timelags -= timelags[0] # get timelags wrt 1st antenna 

151 

152 return np.asarray(timelags)

154 @staticmethod 

155 def _direction_horizontal_array(positions: np.ndarray, times: np.ndarray, ignore_z_coordinate=False): 

156 r""" 

157 --- adapted from pycrtools.modules.scrfind --- 

158 Given N antenna positions, and (pulse) arrival times for each antenna, 

159 get a direction of arrival (azimuth, zenith) assuming a source at infinity (plane wave). 

160 

161 Here, we find the direction assuming all antennas are placed in the z=0 plane. 

162 If all antennas are co-planar, the best-fitting solution can be found using a 2D-linear fit. 

163 We find the best-fitting A and B in: 

164 

165 .. math:: 

166 

167 t = A x + B y + C 

168 

169 where t is the array of times; x and y are arrays of coordinates of the antennas. 

170 The C is the overall time offset in the data, that has to be subtracted out. 

171 The optimal value of C has to be determined in the fit process (it's not just the average time, 

172 nor the time at antenna 0). 

173 

174 This is done using :mod:`numpy.linalg.lstsq`. 

175 

176 The (azimuth, zenith) follows from: 

177 

178 .. math:: 

179 

180 A = \sin(\mathrm{zenith}) \cos(\mathrm{azimuth}) 

181 

182 B = \sin(\mathrm{zenith}) \sin(\mathrm{azimuth}) 

183 

184  

185 Parameters 

186 ---------- 

187 positions : np.ndarray 

188 Positions (x,y,z) of the antennas (shape: (N_antennas, 3)) 

189 times : array, float 

190 Measured pulse arrival times for all antennas 

191 

192 Returns 

193 ------- 

194 zenith : float 

195 Zenith in the [0, 2pi] interval (given in internal units) 

196 azimuth : float 

197 Azimuth in the [0, 2pi] interval (given in internal units) 

198 """ 

199 

200 # make x, y arrays out of the input position array 

201 x = positions[:, 0] 

202 y = positions[:, 1] 

203 

204 # now a crude test for nonzero z-input, |z| > 0.5 

205 z = positions[:, 2] 

206 if not ignore_z_coordinate and max(abs(z)) > 0.5: 

207 raise ValueError("Input values of z are nonzero ( > 0.5) !") 

208 

209 M = np.vstack([x, y, np.ones(len(x))]).T # says the linalg.lstsq doc 

210 

211 A, B, C = np.linalg.lstsq(M, lightspeed * times, rcond=None)[0] 

212 

213 zenith = np.arcsin(np.sqrt(A**2 + B**2)) # TODO: this can result in RuntimeWarning - why? 

214 azimuth = np.arctan2(-B, -A) # note minus sign as we want the direction of the _incoming_ vector (from the sky, not towards it) 

215 

216 return np.mod(zenith * units.rad, 360 * units.deg), np.mod(azimuth * units.rad, 360 * units.deg)

218 @register_run() 

219 def run(self, event, detector): 

220 """ 

221 Run the plane wave fit for the given event and detector. 

222 

223 Parameters 

224 ---------- 

225 event : Event object 

226 The event for which to run the plane wave fit. 

227 detector : Detector object 

228 The detector for which to run the plane wave fit. 

229 """ 

230 for station in event.get_stations(): 

231 if not station.get_parameter(stationParameters.triggered): 

232 self.logger.debug(f"Station CS{station.get_id():03d} did not trigger, skipping...") 

233 continue 

234 self.logger.debug(f"Running over station CS{station.get_id():03d}") 

235 

236 # get LORA initial guess for the direction 

237 lora_zenith = event.get_hybrid_information().get_hybrid_shower("LORA").get_parameter(showerParameters.zenith) 

238 lora_azimuth = event.get_hybrid_information().get_hybrid_shower("LORA").get_parameter(showerParameters.azimuth) 

239 

240 # Get all group IDs which are still present in the station 

241 station_channel_group_ids = set([channel.get_group_id() for channel in station.iter_channels()]) 

242 

243 # Get the dominant polarisation orientation as calculated by stationPulseFinder 

244 dominant_orientation = station.get_parameter(stationParameters.cr_dominant_polarisation) 

245 

246 # Collect the positions of 'good' antennas 

247 good_channel_pair_ids = np.zeros((len(station_channel_group_ids), 2), dtype=int) 

248 relative_position_array = np.zeros((len(station_channel_group_ids), 3)) 

249 good_amp_or_snr = np.zeros(len(station_channel_group_ids), dtype=bool) 

250 for ind, channel_group_id in enumerate(station_channel_group_ids): 

251 relative_position_array[ind] = detector.get_relative_position(station.get_id(), channel_group_id) 

252 for channel in station.iter_channel_group(channel_group_id): 

253 if np.all(detector.get_antenna_orientation(station.get_id(), channel.get_id()) == dominant_orientation): 

254 good_channel_pair_ids[ind, 0] = channel.get_id() 

255 else: 

256 good_channel_pair_ids[ind, 1] = channel.get_id() 

257 

258 # Check if dominant channel has acceptable SNR or acceptable amplitude (if desired) 

259 channel = station.get_channel(good_channel_pair_ids[ind, 0]) 

260 if self.__min_amp is None: 

261 if channel.get_parameter(channelParameters.SNR) > self.__cr_snr: 

262 good_amp_or_snr[ind] = True 

263 else: 

264 if np.max(np.abs(channel.get_trace())) >= self.__min_amp: 

265 good_amp_or_snr[ind] = True 

266 

267 num_good_antennas = np.sum(good_amp_or_snr) 

268 mask_good_antennas = np.full(num_good_antennas, True) 

269 

270 # the dominant antennas are good_antennas[:, 0] 

271 good_antennas = good_channel_pair_ids[good_amp_or_snr] 

272 position_array = relative_position_array[good_amp_or_snr] 

273 

274 # iteratively do the plane wave fit and remove outliers (controlled by rmsfactor) 

275 # until the number of good antennas remains constant 

276 niter = 0 

277 zenith, azimuth = lora_zenith, lora_azimuth 

278 while niter < self.__max_iter: # TODO: maybe add additional condition? 

279 niter += 1 

280 # if only three antennas (or less) remain, fit should not be trusted as it always has a solution (fails) 

281 if num_good_antennas < self.__min_number_good_antennas: 

282 self.logger.warning(f"Only {num_good_antennas:d} good antennas remaining!") 

283 self.logger.error(f"Too few good antennas for direction fit!") 

284 break 

285 

286 # update arrays to use only previously found "good" antennas: 

287 position_array = position_array[mask_good_antennas] 

288 good_antennas = good_antennas[mask_good_antennas] 

289 

290 # get time lags from the dominant antennas only 

291 times = self._get_timelags(station, good_antennas[:, 0]) 

292 

293 goodpositions = position_array 

294 goodtimes = times 

295 

296 zenith, azimuth = self._direction_horizontal_array(goodpositions, goodtimes, 

297 self.__ignore_non_horizontal_array) 

298 

299 # get residuals 

300 expected_delays = geometric_delay_far_field( 

301 goodpositions, hp.spherical_to_cartesian(zenith / units.rad, azimuth / units.rad) 

302 ) 

303 expected_delays -= expected_delays[0] # get delays wrt 1st antenna 

304 

305 residual_delays = goodtimes - expected_delays 

306 

307 # Debug plots if required 

308 if self.__debug: 

309 self.debug_plots( 

310 event, expected_delays, good_antennas, niter, position_array, residual_delays, station, times 

311 ) 

312 

313 if np.isnan(zenith) or np.isnan(azimuth): 

314 self.logger.error( 

315 'Plane wave fit returns NaN. I will try to recover by setting zenith and azimuth ' 

316 'to the LORA estimate and recalculating the residual delays.' 

317 ) 

318 

319 zenith = lora_zenith 

320 azimuth = lora_azimuth 

321 

322 expected_delays = geometric_delay_far_field( 

323 goodpositions, hp.spherical_to_cartesian(zenith / units.rad, azimuth / units.rad) 

324 ) 

325 expected_delays -= expected_delays[0] 

326 residual_delays = goodtimes - expected_delays 

327 

328 bins = int( 

329 (residual_delays.max() - residual_delays.min()) * lightspeed / 

330 (position_array[:, 0].max() - position_array[:, 0].min()) 

331 ) 

332 hist, edges = np.histogram(residual_delays, bins=max(bins, 1)) 

333 

334 max_time = np.argmax(hist) 

335 self.logger.debug(f"histogram filled: {hist}") 

336 self.logger.debug(f"edges: {edges}") 

337 self.logger.debug(f"maximum at: {max_time}") 

338 

339 upper = edges[min(max_time + 2, len(edges) - 1)] 

340 lower = edges[max(max_time - 1, 0)] 

341 

342 self.logger.debug(f"Selecting between lower {lower} and upper {upper}") 

343 mask_good_antennas = (residual_delays > lower) & (residual_delays < upper) 

344 else: 

345 # remove > k-sigma outliers and iterate 

346 spread = np.std(residual_delays) 

347 k = self.__rmsfactor 

348 mask_good_antennas = abs(residual_delays - np.mean(residual_delays)) < k * spread 

349 # gives subset of 'good_antennas' that is 'good' after this iteration 

350 

351 self.logger.debug(f"station {station.get_id()}:") 

352 self.logger.debug(f"iteration {niter:d}:") 

353 self.logger.debug(f'azimuth = {np.rad2deg(azimuth):.3f}, zenith = {np.rad2deg(zenith):.3f}') 

354 self.logger.debug(f'number of good antennas = {num_good_antennas:d}') 

355 

356 # if the next iteration has the same number of good antennae the while loop will be terminated 

357 if len(good_antennas[mask_good_antennas]) == num_good_antennas: 

358 break 

359 else: 

360 num_good_antennas = len(good_antennas[mask_good_antennas]) 

361 

362 self.logger.status( 

363 f"Azimuth (counterclockwise wrt to East) and zenith for station CS{station.get_id():03d}:" 

364 ) 

365 self.logger.status(f"{azimuth / units.deg}, {zenith / units.deg}") 

366 

367 self.logger.status( 

368 f"Azimuth (clockwise wrt to North) and elevation for station CS{station.get_id():03d}:" 

369 ) 

370 self.logger.status(f"{90 - azimuth / units.deg}, {90 - zenith / units.deg}") 

371 

372 # Set stationParameters.zenith/azimuth because voltageToEfieldConverter uses these to convert 

373 # NOTE: these can be the LORA direction (in case the fit failed) 

374 station.set_parameter(stationParameters.zenith, zenith) 

375 station.set_parameter(stationParameters.azimuth, azimuth) 

376 

377 # Only set reconstructed direction if it is not identical to the LORA direction 

378 if not (zenith == lora_zenith and azimuth == lora_azimuth): 

379 self.logger.info( 

380 f"The fit for station CS{station.get_id():03d} seems to have failed." 

381 f"I will not set the cr_zenith and cr_azimuth station parameters, but you can" 

382 f"still unfold the voltages to electric fields with the LORA direction as this" 

383 f"is saved in the zenith and azimuth station parameters." 

384 ) 

385 station.set_parameter(stationParameters.cr_zenith, zenith) 

386 station.set_parameter(stationParameters.cr_azimuth, azimuth) 

387 

388 # flag channels that were not used in the fit 

389 station_flagged_channels = station.get_parameter(stationParameters.flagged_channels) 

390 

391 for channel_id in good_channel_pair_ids.flatten().tolist(): 

392 if channel_id not in good_antennas.flatten(): 

393 # TODO: this flag is not always correct, as channels excluded by SNR are also flagged 

394 station_flagged_channels[channel_id].append("planewavefit_timing_outlier") 

395 

396 station.set_parameter(stationParameters.flagged_channels, station_flagged_channels)
