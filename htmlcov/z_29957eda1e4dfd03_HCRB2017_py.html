34def get_time_trace(energy, theta, N, dt, is_em_shower, n_index, R, LPM=True, a=None): 

35 """ 

36 returns the Askaryan pulse in the time domain 

37 

38 Parameters 

39 ---------- 

40 energy : float 

41 energy of the shower 

42 theta: float 

43 viewangle: angle between shower axis (neutrino direction) and the line 

44 of sight between interaction and detector 

45 N : int 

46 number of samples in the time domain 

47 dt: float 

48 time bin width, i.e. the inverse of the sampling rate 

49 is_em_shower: bool 

50 true if EM shower, false otherwise 

51 n: float 

52 index of refraction at interaction vertex 

53 R: float 

54 distance from vertex to observer 

55 LPM: bool (default True) 

56 enable/disable LPD effect 

57 a: float or None (default Nont) 

58 if variable set, the shower width is manually set to this value 

59 """ 

60 

61 freqs = np.fft.rfftfreq(N, dt) 

62 eR, eTheta = _get_E_omega(freqs, energy, R, theta, n_index, is_em_shower, LPM, a=a) 

63 traceR = np.fft.irfft(eR) / dt 

64 traceTheta = np.fft.irfft(eTheta) / dt 

65 return np.array([traceR, traceTheta, np.zeros_like(traceTheta)])

102def _get_eta(k, _askaryanDepthA, _askaryanR, _askaryanTheta): 

103 return k * _askaryanDepthA**2 / _askaryanR * np.sin(_askaryanTheta)**2

106def _get_Iff(ff, n_index, _askaryanDepthA, _askaryanR, _askaryanTheta): 

107 COS_THETA_C = 1. / n_index 

108 k = _get_k(ff, n_index) 

109 eta = _get_eta(k, _askaryanDepthA, _askaryanR, _askaryanTheta) 

110 re_d = 1 - 3 * eta**2 * np.cos(_askaryanTheta) / np.sin(_askaryanTheta)**2 * \ 

111 (np.cos(_askaryanTheta) - COS_THETA_C) / (1 + eta**2) 

112 im_d = -eta - 3 * eta**3 * np.cos(_askaryanTheta) / np.sin(_askaryanTheta)**2 * \ 

113 (np.cos(_askaryanTheta) - COS_THETA_C) / (1 + eta**2) 

114 denom = re_d + 1j * im_d 

115 re_power = -0.5 * (k * _askaryanDepthA)**2 * (np.cos(_askaryanTheta) - COS_THETA_C)**2 / (1 + eta**2) 

116 im_power = -eta * 0.5 * (k * _askaryanDepthA)**2 * (np.cos(_askaryanTheta) - COS_THETA_C)**2 / (1 + eta**2) 

117 power = re_power + 1j * im_power 

118 return np.exp(power) / denom**0.5

121def _get_E_omega(ff, E, R, theta, n_index, EM=True, 

122 LPM=True, use_form_factor=True, 

123 _rho0=1. / (np.sqrt(2.0 * np.pi) * 0.03 * units.m), 

124 a=None, fudge_LPM=False): 

125 """ 

126 calculates the frequncy spectrum of an Askaryan pulse. Do not use this function directly, 

127 use get_frequency_spectrum() instead 

128 

129 Parameters 

130 ---------- 

131 ff: np.array of floats 

132 array of frequencies 

133 E: float 

134 shower energy 

135 R: float 

136 distance from vertex to observer 

137 theta: float 

138 viewing angle 

139 n_index: float 

140 index of refraction at the shower 

141 EM: bool (default True) 

142 switch between EM and had. showers 

143 LPM: bool (default True) 

144 enable/disable LPD effect 

145 use_form_factor: bool (default True) 

146 use form factor 

147 _rho0: float 

148 the value of rho0 

149 a: float or None (default Nont) 

150 if variable set, the shower width is manually set to this value 

151 fudge_LPM: bool (default False) 

152 if True, the shower width parameterization of LPM showers is rescaled to match 

153 the Greisen parameterization at energies below the E_LPM, i.e., at energies where the LPM effect is negligible 

154 

155 Returns 

156 ------- 

157 eR: array of floats 

158 eR component of electric field in frequency domain 

159 eTheta: array of floats 

160 eTheta component of electric field in frequency domain 

161 

162 """ 

163 

164 _Nmax, _askaryanDepthA = get_N_AskDepthA(E, EM, LPM, fudge_LPM=fudge_LPM) 

165 if(a is not None): 

166 _askaryanDepthA = a 

167 COS_THETA_C = 1. / n_index 

168 k = _get_k(ff, n_index) 

169 eta = _get_eta(k, _askaryanDepthA, R, theta) 

170 I_FF = _get_Iff(ff, n_index, _askaryanDepthA, R, theta) 

171 nu = speed_of_light * k / (2.0 * np.pi) 

172 logger.debug("a {}, nmax {}, R {}".format(_askaryanDepthA, _Nmax, R)) 

173 norm = 2.52e-7 * 1e3 * _askaryanDepthA * _Nmax * nu / R / NORM # the additional *1e3 comes from putting all the units in the constant of Eq. (10). The left side of the equation is in MHz, whereas the right side is in GHz 

174 # Kinematic factor, psi...checked JCH March 8th, 2016...fixed missing sin(theta) 

175 psi = np.sin(theta) * np.sin(k * R) + 1j * (-np.sin(theta) * np.cos(k * R)) 

176 # radial component (imaginary part is zero)...checked JCH March 8th, 2016 

177 rComp_num = -(np.cos(theta) - COS_THETA_C) / np.sin(theta) 

178 rComp = I_FF * norm * psi * rComp_num 

179 # theta component (has real and imaginary parts)...checked JCH March 8th, 2016 

180 thetaComp_num = 1 + eta**2 / (1 + eta)**2 * COS_THETA_C / np.sin(theta)**2 * (np.cos(theta) - COS_THETA_C) + \ 

181 1j * (-eta / (1 + eta)**2 * COS_THETA_C / np.sin(theta)**2 * (np.cos(theta) - COS_THETA_C)) 

182 thetaComp = I_FF * norm * psi * thetaComp_num 

183 logger.debug("IFF[0] {:.2g}, norm {:.2g}, psi[0] {:.2g}, thetaComp_num {:.2g}".format(I_FF[1], norm[1], psi[1], thetaComp_num[1])) 

184 

185 if use_form_factor: 

186 a = k / _rho0 

187 b = np.sin(theta) / (2.0 * np.pi)**0.5 

188 atten = (1 + a**2 * b**2)**-1.5 

189 rComp *= atten 

190 thetaComp *= atten 

191 

192 return rComp, thetaComp

198def get_N_AskDepthA(E, EM=True, LPM=True, fudge_LPM=False): 

199 """ 

200 calculates the Gaussian width (sigma) of the shower profile using the 

201 Greisen profile for EM showers and the Gaisser-Hillas profile for HAD showers. 

202 If the LPM flag is activated, for EM shower of the shower width of 10.1103/PhysRevD.82.074017 is used. 

203 

204 Please note that the parameterization of the shower width for LPM showers is not compatible with the Greisen 

205 parameterization event at regimes where the LPM effect is negligible!!! 

206 

207 Parameters 

208 ---------- 

209 E: float 

210 the energy of the shower 

211 EM: bool (default True) 

212 switch between EM and had. showers 

213 LPM: bool (default True) 

214 enable/disable LPD effect 

215 fudge_LPM: bool (default False) 

216 if True, the shower width parameterization of LPM showers is rescaled to match 

217 the Greisen parameterization at energies below the E_LPM, i.e., at energies where the LPM effect is negligible 

218 """ 

219 if EM: 

220 E_CRIT = 0.073 * units.GeV # GeV 

221 max_x = 5000.0 # maximum number of radiation lengths 

222 dx = 0.01 # small enough bin in depth for our purposes. 

223 x_start = 0.01 # starting radiation length 

224 # Greissen EM shower profile from Energy E in GeV. 

225 

226 x = np.arange(x_start, max_x, dx) 

227 a = 0.31 / (np.log(E / E_CRIT))**0.5 

228 b = x 

229 c = 1.5 * x 

230 d = np.log((3 * x) / (x + 2 * np.log(E / E_CRIT))) 

231 nx = a * np.exp(b - c * d) 

232 

233 else: # hadronic shower profile 

234 # Gaisser-Hillas hadronic shower parameterization 

235 max_x = 200000.0 * units.g /units.cm**2 # maximum depth in g/cm^2 

236 dx = 1.0 * units.g /units.cm**2 # small enough bin in depth for our purposes. 

237 x_start = dx # depth in g/cm^2 

238 S0 = 0.11842 

239 X0 = 39.562 * units.g /units.cm**2 # g/cm^2 

240 l = 113.03 * units.g /units.cm**2# g/cm^2 

241 Ec = 0.17006 * units.GeV # GeV 

242 Xmax = X0 * np.log(E / Ec) 

243 x = np.arange(x_start, max_x, dx) 

244 a = S0 * E / Ec * (Xmax - l) / Xmax * np.exp(Xmax / l - 1) 

245 b = pow(x / (Xmax - l), Xmax / l) 

246 c = np.exp(-x / l) 

247 nx = a * b * c 

248 # find location of maximum, and charge excess from Fig. 5.9, compare in cm not m. 

249 n_max_position = np.argmax(nx) 

250 n_max = np.max(nx) 

251 if EM: 

252 excess = 0.09 + dx * n_max_position * ICE_RAD_LENGTH / ICE_DENSITY / 100. 

253 else: 

254 excess = 0.09 + dx * n_max_position / ICE_DENSITY * 1.0e-2 

255 Nmax = excess * n_max / 1000.0 

256 logger.debug("Nmax {}, excess {}, n_max {}".format(Nmax, excess, n_max)) 

257 

258 # We want to perform a fit for the regions with an excess charge 10% close to the maximum 

259 fit_region_cut = 0.95 

260 cut_left = np.argwhere((nx[:n_max_position] / nx[n_max_position]) > fit_region_cut)[0][0] 

261 cut_right = np.argwhere((nx[n_max_position:] / nx[n_max_position]) < fit_region_cut)[0][0]+n_max_position 

262 fit_width = cut_right-cut_left 

263 max_vicinity = nx[n_max_position-fit_width:n_max_position+fit_width]/nx[n_max_position] 

264 x_fit = np.arange(0, len(max_vicinity), 1) 

265 sigma = curve_fit(gauss, x_fit, max_vicinity)[0] 

266 if EM: 

267 _askaryanDepthA = dx * sigma[2] / ICE_DENSITY * ICE_RAD_LENGTH 

268 else: 

269 _askaryanDepthA = dx * sigma[2]/ ICE_DENSITY 

270 logger.debug("a (before LPM = {}".format(_askaryanDepthA)) 

271 

272 E_LPM = 3e14 * units.eV 

273 if(EM and LPM): 

274 if((E > E_LPM) or not fudge_LPM): # only apply LPM correction in regimes where it is relevant 

275 p1 = -2.8564e2 

276 p2 = 7.8140e1 

277 p3 = -8.3893 

278 p4 = 4.4175e-1 

279 p5 = -1.1382e-2 

280 p6 = 1.1493e-4 

281 e = np.log10(E/units.eV) # log_10 of Energy in eV 

282 log10_shower_depth = p1 + p2 * e + p3 * e**2 + p4 * e**3 + p5 * e**4 + p6 * e**5 

283 a = 10.0**log10_shower_depth * 0.5 # adjust shower wiedth to be just the sigma parameter of a Gaussian 

284 

285 if(fudge_LPM): 

286 # normalize to Greisen parameterization at LPM energy 

287 a_Greisen = get_N_AskDepthA(E_LPM, EM=True, LPM=False)[1] 

288 a /= a_Greisen 

289 

290 # Right here, record the reduction in n_max_position that I don't believe in. 

291 if _strictLowFreqLimit: 

292 logger.debug("strict_lowfeq Nmax = {:.2g}, a= {} priora = {}".format(Nmax, a, _askaryanDepthA/units.m, Nmax)) 

293 Nmax = Nmax / (a / _askaryanDepthA) 

294 _askaryanDepthA = a 

295 logger.debug("a = {:.2f}m, Nmax = {}".format(_askaryanDepthA/units.m, Nmax)) 

296 return Nmax, _askaryanDepthA
